<PageHeader slug="search-box" title="Search Box" component="search-box">
  Accepts a query and submits a search - with a shortcut hint, as a large hero search with a scope, a pill, or an icon that expands on focus. Part of the <DocLink href="website-search-navigation.html">Search & Navigation</DocLink> blocks.
</PageHeader>

<SkillPanel component="search-box">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;search&gt;</code> with a <code>&lt;form method="get"&gt;</code>: a visually hidden label, a search icon, an <code>&lt;input type="search" name="q"&gt;</code> (<code>enterkeyhint="search"</code>), a Kbd shortcut hint and submit.</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="input.html">Input</DocLink>, <DocLink href="kbd.html">Kbd</DocLink>, <DocLink href="select.html">Select</DocLink>, <DocLink href="button.html">Button</DocLink>.</SkillText>
  </SkillField>

```html example label="Default" hint="A <search> landmark around a GET form: hidden label, icon, input type=search with enterkeyhint, a ⌘K hint and submit - works without script, the URL is shareable." previewStyle="align-items:stretch;"
<search class="mk-search-box">
  <form action="#" method="get" role="search">
    <label class="sr-only" for="sb-1">Search</label>
    <span class="mk-search-box-field"><i data-lucide="search"></i><input class="input" id="sb-1" type="search" name="q" placeholder="Search docs, tasks and people" autocomplete="off" enterkeyhint="search"><kbd class="kbd">⌘K</kbd></span>
    <button class="btn" type="submit">Search</button>
  </form>
</search>
```

```html example label="Large" hint="data-variant='large': a hero search on a raised surface with a scope Select." previewStyle="align-items:stretch;"
