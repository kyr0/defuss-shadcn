<SkillPanel component="pricing">
  <SkillField label="Native basis">
<SkillText><code>display: grid; align-items: stretch</code> keeps plan cards equal height across content differences. The billing toggle is a radio group — <code>&lt;input type="radio"&gt;</code> in <code>&lt;label&gt;</code>s — and <code>:has(input:checked)</code> swaps every plan's yearly/monthly price rows. Keyboard, screen readers, and form semantics are browser-provided.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/input/radio" title="exclusive choice with free keyboard support"><code>&lt;input type="radio"&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="checked radio flips every price panel from CSS"><code>:has()</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/align-items" title="equal-height plan cards"><code>align-items: stretch</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/font-variant-numeric#tabular-nums" title="tabular price digits"><code>font-variant-numeric</code></ApiPill>
  </SkillRow>
