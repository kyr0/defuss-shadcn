  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-dumbbell');
    const TEXT = '#1D2A23', MUTED = '#7C7F6E', BURG = '#A51C30', GREEN = '#2F5D46', LINE = 'rgba(29,42,35,.28)';
    const G = { left: 250, right: 190, top: 320, bottom: 150 };
    // [district, 2019, 2025, burden 2019 %, burden 2025 %]
    const rows = [['Old Town', 14.2, 19.8, 30, 36], ['Riverside', 12.6, 18.9, 27, 36], ['University', 11.8, 16.4, 33, 40], ['Harbor', 9.4, 15.1, 22, 31], ['Midtown', 10.9, 14.3, 25, 32], ['Suburbs', 8.9, 11.2, 20, 23], ['Industrial', 7.1, 10.6, 17, 22]];
    const pct = (r) => Math.round((r[2] / r[1] - 1) * 100);
    const byLevel = rows.slice().sort((a, b) => b[2] - a[2]);
    const byGrowth = rows.slice().sort((a, b) => pct(b) - pct(a));
    const cats = (list) => ({ type: 'category', inverse: true, data: list.map((r) => r[0]), axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 700 } });
    const euro = { type: 'value', min: 6, max: 22, interval: 4, axisLabel: { color: MUTED, formatter: (v) => '€' + v } };

    const dumbbell = (list, hot) => ({
      grid: G, xAxis: euro, yAxis: cats(list),
      series: [
        { id: 'base', type: 'bar', stack: 'd', barWidth: 5, silent: true, data: list.map((r) => ({ name: r[0], value: r[1] })), itemStyle: { color: 'transparent' } },
        { id: 'span', type: 'bar', stack: 'd', barWidth: 5, silent: true, data: list.map((r) => ({ name: r[0], value: +(r[2] - r[1]).toFixed(1), itemStyle: { color: r[0] === hot ? BURG : LINE } })) },
        { id: 'before', type: 'scatter', symbolSize: 24, data: list.map((r) => ({ name: r[0], value: [r[1], r[0]] })), itemStyle: { color: '#F4EEDC', borderColor: GREEN, borderWidth: 3 } },
        { id: 'after', type: 'scatter', symbolSize: 24, data: list.map((r) => ({ name: r[0], value: [r[2], r[0]], itemStyle: { color: !hot || r[0] === hot ? BURG : GREEN } })),
          label: { show: true, position: 'right', distance: 18, formatter: (p) => '+' + pct(list[p.dataIndex]) + '%', color: TEXT, fontWeight: 700 } },
      ],
    });

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        gap: dumbbell(byLevel, null),
        growth: dumbbell(byGrowth, 'Harbor'),
        change: {
          grid: G, xAxis: { type: 'value', max: 70, interval: 10, axisLabel: { color: MUTED, formatter: '{value}%' } }, yAxis: cats(byGrowth),
          series: [{ id: 'span', type: 'bar', barWidth: '56%',
            data: byGrowth.map((r) => ({ name: r[0], value: pct(r), itemStyle: { color: r[0] === 'Harbor' || r[0] === 'Industrial' ? BURG : GREEN } })),
            label: { show: true, position: 'right', formatter: '+{c}%', color: TEXT, fontWeight: 700 } }],
        },
        burden: {
          legend: { top: 300, right: 190, textStyle: { color: TEXT } },
          grid: { left: 150, right: 120, top: 370, bottom: 150 },
          xAxis: { type: 'category', data: byLevel.map((r) => r[0]), axisLabel: { interval: 0, color: TEXT, fontWeight: 700 } },
          yAxis: { type: 'value', max: 45, interval: 15, axisLabel: { color: MUTED, formatter: '{value}%' } },
          series: [
            { id: 'before', name: '2019', type: 'bar', barGap: '8%', data: byLevel.map((r) => ({ name: r[0], value: r[3] })), itemStyle: { color: LINE } },
            { id: 'after', name: '2025', type: 'bar', data: byLevel.map((r) => ({ name: r[0], value: r[4] })), itemStyle: { color: BURG },
              label: { show: true, position: 'top', formatter: '{c}%', color: TEXT, fontWeight: 700 },
              markLine: { symbol: 'none', silent: true, lineStyle: { color: TEXT, type: 'dashed', width: 2 }, label: { formatter: '30% rule', color: TEXT, position: 'insideEndTop' }, data: [{ yAxis: 30 }] } },
          ],
        },
      },
    });
  });
</script>
