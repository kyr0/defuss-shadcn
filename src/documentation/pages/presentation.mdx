<PageHeader slug="presentation" title="Presentation" component="presentation">
  The HTML presentation primitive: a <strong>fixed 1600×900 design surface</strong> scaled uniformly to the mount (responsive <em>viewer</em>, fixed-coordinate <em>slide</em> - no fluid layout inside slides), keyboard/touch/fullscreen navigation, presenter notes, and the <DocLink href="motion.html">Motion</DocLink> entrance vocabulary replayed per slide. Zero dependencies beyond Defuss.

<script>
  // all.js (df$) boots after example scripts in the sandbox - the chart mount
  // and the showcase drive happen in the load handler, never at parse time.
  // Chart colors come off the deck's own --presentation-* surface (a slide's
  // art direction is fixed by design); the artboard is 1600×900, so chart
  // text runs at 26–28px - ECharts' 12px default is illegible at this scale.
  window.addEventListener('load', () => {
    const deck = document.getElementById('qbr-deck');
    const cs = getComputedStyle(deck);
    const ink = cs.getPropertyValue('--presentation-ink-foreground').trim() || '#e9eef8';
    const accent = cs.getPropertyValue('--presentation-accent').trim() || '#38bdf8';

    // slides stay full-size while hidden, so the chart mounts fine on load
    df$.shadcn.chart.mount(document.getElementById('qbr-revenue-chart'), {
      aria: { enabled: true },
      grid: { left: 110, right: 60, top: 60, bottom: 70 },
      xAxis: {
        type: 'category', data: ['Q1', 'Q2', 'Q3', 'Q4'],
        axisLine: { show: false }, axisTick: { show: false },
        axisLabel: { color: ink, fontSize: 28, fontWeight: 700 },
      },
      yAxis: {
        type: 'value', max: 8,
        axisLabel: { color: ink, fontSize: 26, formatter: '${value}M' },
        splitLine: { lineStyle: { color: 'rgba(233,238,248,.14)' } },
      },
      series: [{
        type: 'bar',
        data: [4.2, 5.1, 6.3, { value: 7.4, itemStyle: { opacity: 0.55 } }],
        itemStyle: { color: accent, borderRadius: [12, 12, 0, 0] },
        label: { show: true, position: 'top', color: ink, fontSize: 28, fontWeight: 700, formatter: '${c}M' },
      }],
    });

    // the product demo plays only while its slide is on stage: the shared
    // data-active MutationObserver drives the shipped .mk-showcase State API
    const demoSlide = document.getElementById('demo');
    const showcase = demoSlide.querySelector('.mk-showcase');
    const syncDemo = () => showcase.api.setState(demoSlide.hasAttribute('data-active') ? 'playing' : 'default');
    new MutationObserver(syncDemo).observe(demoSlide, { attributes: true, attributeFilter: ['data-active'] });
    syncDemo(); // the observer only fires on CHANGE - cover a deep-linked #demo
  });
</script>

  // The race STARTS when its slide activates and STOPS when it deactivates —
  // no timers burning on hidden slides. df$.shadcn.chart.mount owns init/resize/theme.
  window.addEventListener('load', () => {
    const raceSlide = document.getElementById('race-slide');
    const names = ['North', 'West', 'Central', 'Coast', 'Metro', 'Rural'];
    const frames = [
      [32, 27, 25, 21, 18, 14],
      [35, 31, 28, 23, 22, 17],
      [38, 33, 29, 28, 24, 20],
      [41, 36, 35, 30, 27, 22],
      [45, 42, 38, 31, 30, 25],
      [48, 46, 41, 34, 33, 28],
    ];
    const option = (i) => ({
      grid: { left: 110, right: 80, top: 40, bottom: 40 },
      xAxis: { type: 'value', max: 'dataMax' },
      yAxis: { type: 'category', inverse: true, data: names,
               axisLine: { show: false }, axisTick: { show: false } },
      series: [{ type: 'bar', realtimeSort: true, data: frames[i],
                 label: { show: true, position: 'right', valueAnimation: true } }],
    });

    let idx = 0;
    let timer = null;
    const race = df$.shadcn.chart.mount(document.getElementById('race-chart'), option(0));
    const stop = () => { clearInterval(timer); timer = null; };
    const start = () => {
      stop();
      idx = 0;
      race.setOption(option(0), true);
      timer = setInterval(() => {
        idx = (idx + 1) % frames.length;
        race.setOption(option(idx));
      }, 1300);
    };
    const sync = () => (raceSlide.hasAttribute('data-active') ? start() : stop());
    new MutationObserver(sync).observe(raceSlide, { attributes: true, attributeFilter: ['data-active'] });
    sync();
  });
</script>

  <p class="text-sm text-muted-foreground m-0 mb-3">Named states via the shared State API, driven per instance through the bound <code>api</code>:</p>
  <ul class="text-sm text-muted-foreground m-0 mb-3 ps-5">
    <li><code>default</code> - the authored surface; <code>setState('default', \{ index \})</code> activates slide N and <code>getState().config.slide</code> reports the live index</li>
    <li><code>notes</code> - presenter notes of the active slide visible (<code>data-notes</code>)</li>
    <li><code>fullscreen</code> - mount pinned to the viewport (<code>data-fullscreen</code>): native fullscreen or the fixed-overlay fallback</li>
  </ul>
  <p class="text-sm text-muted-foreground">The demo element carries <code>data-state-demo</code>; <code>bun run screenshots</code> drives it via <code>el.api.setState(name)</code>.</p>

Machine contract - verified against <a class="link" href="../schemas/presentation.schema.json"><code>presentation.schema.json</code></a> by <code>bun run verify</code>:

```states
| State | Type | Values | Default | Description |
| --- | --- | --- | --- | --- |
| `slide` | `number` | `—` | `0` | Active slide index - `setState('default', { index })`; mirrored as `data-current-slide` and updated by keyboard/click/hash. |
| `notes` | `boolean` | `true`, `false` | `false` | Presenter notes of the active slide visible (`data-notes` on the mount). |
| `fullscreen` | `boolean` | `true`, `false` | `false` | Mount pinned to the viewport (`data-fullscreen`): native fullscreen or the honest fixed-overlay fallback. |
```
