<PageHeader slug="payment-form" title="Payment Form" component="payment-form">
  Collects or selects a payment method - card, PayPal or bank transfer as tabs, the card fields with the right keyboards and autofill, or a saved card. Part of the <DocLink href="website-shop.html">Shop</DocLink> blocks.
</PageHeader>

<SkillPanel component="payment-form">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;fieldset&gt;</code>: methods as radio "tabs" (<code>value="card|paypal|bank"</code>), one panel per method (<code>data-method</code>) shown with <code>:has()</code>; the card panel has number, expiry, security code and name with <code>autocomplete="cc-…"</code> and <code>inputmode="numeric"</code>.</SkillText>
  </SkillField>

<SkillText><DocLink href="input.html">Input</DocLink>, <DocLink href="label.html">Label</DocLink>, <DocLink href="checkbox.html">Checkbox</DocLink>, <DocLink href="radio.html">Radio</DocLink>, <DocLink href="button.html">Button</DocLink>.</SkillText>
