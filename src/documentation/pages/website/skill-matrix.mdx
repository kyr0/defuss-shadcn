<PageHeader slug="skill-matrix" title="Skill Matrix" component="skill-matrix">
  What someone can do, in groups - each skill a name, its level as a native meter and a note; the levels as bars or as five dots, or the skills as tags without levels. Part of the <DocLink href="website-resume-portfolio.html">Resume &amp; Portfolio</DocLink> blocks.
</PageHeader>

<SkillPanel component="skill-matrix">
  <SkillField label="Native basis">
<SkillText>Groups of skills, each a <code>&lt;section&gt;</code> with a heading and a <code>&lt;ul&gt;</code>; a skill is its name (a <code>&lt;label&gt;</code> for its meter), its level as a native <code>&lt;meter min="0" max="5"&gt;</code> and a note.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/meter" title="a scalar value in a known range"><code>&lt;meter&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/label" title="names the meter with the skill"><code>&lt;label for&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/::-moz-meter-bar" title="the track and the fill, themed"><code>::-webkit-meter-bar</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/mask" title="the fill as five whole dots"><code>mask</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_grid_layout/Subgrid" title="names, levels and notes line up"><code>subgrid</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/print-color-adjust" title="the levels survive printing"><code>print-color-adjust</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-skill-matrix</code>
    <code>.mk-skill-matrix-group</code>
    <code>.mk-skill-matrix-title</code>
    <code>.mk-skill-matrix-list</code>
    <code>.mk-skill-matrix-skill</code>
    <code>.mk-skill-matrix-name</code>
    <code>.mk-skill-matrix-level</code>
    <code>.mk-skill-matrix-note</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant="dots"</code> (each level five dots), <code>data-variant="stars"</code> (five stars, half steps as half stars) and <code>data-variant="tags"</code> (no levels, the skills as pills).</SkillText>
  </SkillField>
</SkillPanel>
