<h2 id="template" class="h2-display">The template</h2>

<DocText>The Comments blocks composed into one page - the source is the page. A comment's time is its permalink, a route of <DocLink href="routing.html">defuss-dom-router</DocLink> (<code>#/comments/d-3</code>) that highlights it, and the sort order is a query parameter (<code>#/?sort=oldest</code>); the language switch translates the frame with <DocLink href="translation.html">defuss-i18n</DocLink> - the times come from <code>Intl.RelativeTimeFormat</code>, the comments stay as written.</DocText>

<script type="module">
  // routing and translation: defuss-dom-router owns the address, defuss-i18n the language
  import { createRouter } from 'https://cdn.jsdelivr.net/npm/defuss-dom-router@0.1.1/dist/index.js';
  import {
    createDomI18n,
    createI18n,
    resolveLocale,
  } from 'https://cdn.jsdelivr.net/npm/defuss-i18n@0.2.0/dist/core.js';

  // df$ comes with the component bundle - a full-screen page loads it after this module
  addEventListener('DOMContentLoaded', async () => {
    const site = df$('#site-comments').get(0);

    // -- translation: the markup is the catalog (the templates at the end of the page)
    const locale = createI18n({
      locale: navigator.languages.map((tag) => resolveLocale(['en', 'de'], tag, [])).find(Boolean) ?? 'en',
    });
    createDomI18n(df$).bind(site, locale, {
      values: () => ({
        // how long ago, from the page's "now" - Intl.RelativeTimeFormat in the language
        ...Object.fromEntries(
          [
            ['ago1', -2, 'hour'],
            ['ago2', -1, 'hour'],
            ['ago3', -40, 'minute'],
            ['ago4', -3, 'hour'],
          ].map(([key, n, unit]) => [
            key,
            new Intl.RelativeTimeFormat(locale.locale, { numeric: 'auto' }).format(n, unit),
          ]),
        ),
      }),
    });
    const language = df$('[data-app-locale]');
    language.val(locale.locale).on('change', () => locale.setLocale(language.val()));

    // -- routing
    const routes = [
      { id: 'thread', path: '/' },
      { id: 'comment', path: '/comments/:id' },
      { id: 'missing', path: '*' },
    ];
    // ?sort= orders the thread (top: likes, newest / oldest: time); a permalink highlights its comment
    const show = (to) => {
      const sort = new URLSearchParams(to.search).get('sort') ?? 'top';
      df$('[data-comments-sort]').val(sort);
      const thread = df$('[data-comments-thread]');
      const items = [...thread.children('.mk-comment-item')];
      const at = (c) => Date.parse(c.querySelector('time').getAttribute('datetime'));
      items.sort((a, b) =>
        sort === 'top' ? b.dataset.likes - a.dataset.likes : sort === 'newest' ? at(b) - at(a) : at(a) - at(b),
      );
      // moved to the front in reverse, so they read in order; the reply form stays last
      for (const item of items.reverse()) thread.prepend(item);
      df$('#site-comments .mk-comment-item').each((_i, c) => {
        c.toggleAttribute('data-highlighted', to.routeId === 'comment' && c.id === to.params.id);
      });
    };
    // the sort select navigates: the order is part of the address
    df$('[data-comments-sort]').on('change', (event) => {
      const href = '#/?sort=' + event.currentTarget.value;
      if (router.getSnapshot().current) router.navigate(href);
      else location.hash = href.slice(1);
    });
    const router = createRouter({ mode: 'hash', routes, prepare: ({ to }) => ({ commit: () => show(to) }) });
    router.attachLinks(site);
    if ((await router.start()).status === 'rejected') {
      // no address the router may own - a docs preview is an about:srcdoc frame: the same routes match its hash
      const preview = createRouter({ mode: 'hash', routes, baseUrl: 'https://preview.invalid/' });
      const follow = () => show(preview.resolve(location.hash.startsWith('#/') ? location.hash : '#/'));
      addEventListener('hashchange', follow);
      follow();
    }
  });
</script>
