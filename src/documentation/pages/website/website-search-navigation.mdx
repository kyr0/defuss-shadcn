<PageHeader slug="website-search-navigation" title="Search & Navigation" introStyle="max-width:42rem;">
  Blocks for finding and moving around: the search box and its suggestions, the results summary and results, filter bars and sidebars, active filters, sort and view controls, pagination and load more, tags, category menus, breadcrumbs, on-page contents and quick links. Every block is CSS-only and built from the library's components and tokens.
</PageHeader>

<h2 id="template" class="h2-display">The template</h2>

<DocText>The Search & Navigation blocks composed into one page - the source is the page. The search is its address: <code>q</code>, <code>category</code>, <code>sort</code>, <code>page</code> and <code>tag</code> are query parameters of one route of <DocLink href="routing.html">defuss-dom-router</DocLink> (<code>#/?category=design&amp;page=3</code>) - every link shows where it goes from here, Back steps through the searches; the language switch translates the page with <DocLink href="translation.html">defuss-i18n</DocLink>, the counts formatted for the language.</DocText>

  // routing and translation: defuss-dom-router owns the address, defuss-i18n the language
  import { createRouter } from 'https://cdn.jsdelivr.net/npm/defuss-dom-router@0.1.1/dist/index.js';
  import { createDomI18n, createI18n } from 'https://cdn.jsdelivr.net/npm/defuss-i18n@0.2.0/dist/core.js';

  // df$ comes with the component bundle - a full-screen page loads it after this module
  addEventListener('DOMContentLoaded', async () => {
    // light or dark: the switch writes the page's dark class (every look has both palettes); it follows the class when
    // something else sets it (the docs card follows the site's mode)
    const siteMode = df$('#site-search [data-site-mode]');
    const syncMode = () => siteMode.prop('checked', document.documentElement.classList.contains('dark'));
    siteMode.on('change', () => {
      document.documentElement.classList.toggle('dark', siteMode.prop('checked'));
      document.dispatchEvent(new CustomEvent('defuss-theme-change'));
    });
    new MutationObserver(syncMode).observe(document.documentElement, { attributes: true, attributeFilter: ['class'] });
    syncMode();

    const site = df$('#site-search').get(0);

    // the search the address names - what the page shows when a parameter is missing
    const DEFAULTS = { q: 'calm sofware', category: '', sort: '', page: '2', tag: '' };
    const TOTAL = 248;
    const PER_PAGE = 12;
    let query = new URLSearchParams();
    const get = (key) => query.get(key) ?? DEFAULTS[key];
    // -- translation: the markup is the catalog (the templates at the end of the page)
    const locale = createI18n({
      locale: 'en', // the demos start in English; the language select offers German
    });
    const texts = createDomI18n(df$).bind(site, locale, {
      getState: () => query,
      values: (state) => ({
        q: get('q'),
        total: locale.formatNumber(TOTAL),
        from: locale.formatNumber((Number(get('page')) - 1) * PER_PAGE + 1),
        to: locale.formatNumber(Math.min(TOTAL, Number(get('page')) * PER_PAGE)),
      }),
    });
    const language = df$('[data-app-locale]');
    language.val(locale.locale).on('change', () => locale.setLocale(language.val()));

    // -- the look: one of the site's designs. An element with data-look-set takes that look's attributes (a shader's
    // effect, colors or contrast); a shader reads its colors again on defuss-theme-change
    const lookRoot = df$('#site-search').get(0);
    const look = df$('#site-search [data-site-look]');
    const applyLook = (name) => {
      lookRoot.dataset.look = name;
      df$('#site-search [data-look-set]').each((_i, el) => {
        const sets = JSON.parse(el.dataset.lookSet);
        for (const [attr, value] of Object.entries({ ...sets.default, ...sets[name] })) el.setAttribute(attr, value);
      });
      document.dispatchEvent(new CustomEvent('defuss-theme-change'));
    };
    look.val(lookRoot.dataset.look).on('change', () => applyLook(look.val()));

    // -- routing
    const routes = [
      { id: 'search', path: '/' },
      { id: 'missing', path: '*' },
    ];
    // one parameter changed, the rest kept (a new search, category or sort starts at page 1); an empty q is a
    // cleared search, not the default one
    const hrefWith = (key, value) => {
      const next = new URLSearchParams(query);
      if (value || key === 'q') next.set(key, value);
      else next.delete(key);
      if (key !== 'page') next.delete('page');
      return '#/' + (next.size ? '?' + next : '');
    };
    const go = (href) => {
      if (router.getSnapshot().current) router.navigate(href);
      else location.hash = href.slice(1);
    };
    // the address is the search: every link shows where it goes from here, the controls show the parameters
    const show = (to) => {
      query = new URLSearchParams(to.search);
      df$('#site-search [data-query]').each((_i, link) => {
        const [key, value] = link.dataset.query.split('=');
        link.setAttribute('href', hrefWith(key, value));
        if (get(key) === value && key !== 'q') link.setAttribute('aria-current', 'page');
        else link.removeAttribute('aria-current');
      });
      // the Pagination component shows the page (its State API)
      if (pager.api.getState().config.activePage !== Number(get('page')))
        pager.api.setState('default', { activePage: Number(get('page')) });
      df$('[data-search-sort]').val(get('sort'));
      df$('#sb-2').val(query.get('q') ?? '');
      df$('[data-search-suggest]').get(0).hidden = get('q') !== 'calm sofware';
      texts.refresh();
    };
    df$('[data-search-sort]').on('change', (event) => go(hrefWith('sort', event.currentTarget.value)));
    // the Pagination component owns its links (it claims their clicks): the page it lands on goes into the address
    const pager = df$('[data-search-pager]').get(0);
    pager.store.subscribe(({ config }) => {
      if (String(config.activePage) !== get('page')) go(hrefWith('page', String(config.activePage)));
    });
    df$('[data-search-form]').on('submit', (event) => {
      event.preventDefault();
      go(hrefWith('q', df$('#sb-2').val().trim()));
    });
    const router = createRouter({ mode: 'hash', routes, prepare: ({ to }) => ({ commit: () => show(to) }) });
    router.attachLinks(site);
    if ((await router.start()).status === 'rejected') {
      // no address the router may own - a docs preview is an about:srcdoc frame: the same routes match its hash
      const preview = createRouter({ mode: 'hash', routes, baseUrl: 'https://preview.invalid/' });
      const follow = () => show(preview.resolve(location.hash.startsWith('#/') ? location.hash : '#/'));
      addEventListener('hashchange', follow);
      follow();
    }
  });
