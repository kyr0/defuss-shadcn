<h2 id="template" class="h2-display">The template</h2>

<DocText>The Blog blocks composed into one page - the source is the page. The categories and the archive months are routes of <DocLink href="routing.html">defuss-dom-router</DocLink> (<code>#/category/design</code>, <code>#/archive/2026/09</code>) that filter the posts, and the language switch translates the frame with <DocLink href="translation.html">defuss-i18n</DocLink> - the dates are formatted for the language, the posts stay as written.</DocText>

<script type="module">
  // routing and translation: defuss-dom-router owns the address, defuss-i18n the language
  import { createRouter } from 'https://cdn.jsdelivr.net/npm/defuss-dom-router@0.1.1/dist/index.js';
  import {
    createDomI18n,
    createI18n,
    resolveLocale,
  } from 'https://cdn.jsdelivr.net/npm/defuss-i18n@0.2.0/dist/core.js';

  // df$ comes with the component bundle - a full-screen page loads it after this module
  addEventListener('DOMContentLoaded', async () => {
    const site = df$('#site-blog').get(0);

    // -- translation: the markup is the catalog (the templates at the end of the page)
    const locale = createI18n({
      locale: navigator.languages.map((tag) => resolveLocale(['en', 'de'], tag, [])).find(Boolean) ?? 'en',
    });
    createDomI18n(df$).bind(site, locale, {
      values: () => ({
        d1001: locale.formatDate(Date.UTC(2026, 9, 1), { dateStyle: 'medium', timeZone: 'UTC' }),
        d0924: locale.formatDate(Date.UTC(2026, 8, 24), { dateStyle: 'medium', timeZone: 'UTC' }),
      }),
    });
    const language = df$('[data-app-locale]');
    language.val(locale.locale).on('change', () => locale.setLocale(language.val()));

    // -- routing
    const routes = [
      { id: 'all', path: '/' },
      { id: 'category', path: '/category/:category' },
      { id: 'month', path: '/archive/:year/:month' },
      { id: 'missing', path: '*' },
    ];
    // a category or a month filters the posts; its link is the current one
    const show = (to) => {
      const month = to.routeId === 'month' ? to.params.year + '-' + to.params.month : null;
      const category = to.routeId === 'category' ? to.params.category : null;
      let shown = 0;
      df$('#site-blog [data-category]').each((_i, post) => {
        post.hidden = (!!category && post.dataset.category !== category) || (!!month && post.dataset.month !== month);
        if (!post.hidden) shown++;
      });
      df$('#site-blog [data-blog-empty]').get(0).hidden = shown > 0;
      df$('#site-blog nav [data-router-link]').each((_i, link) => {
        if (link.getAttribute('href') === '#' + (to.routeId === 'missing' ? '/' : to.path))
          link.setAttribute('aria-current', 'page');
        else link.removeAttribute('aria-current');
      });
      // the archive opens the month's year
      if (month) df$('#site-blog [data-blog-archive] a[aria-current]').closest('details').prop('open', true);
    };
    const router = createRouter({ mode: 'hash', routes, prepare: ({ to }) => ({ commit: () => show(to) }) });
    router.attachLinks(site);
    if ((await router.start()).status === 'rejected') {
      // no address the router may own - a docs preview is an about:srcdoc frame: the same routes match its hash
      const preview = createRouter({ mode: 'hash', routes, baseUrl: 'https://preview.invalid/' });
      const follow = () => show(preview.resolve(location.hash.startsWith('#/') ? location.hash : '#/'));
      addEventListener('hashchange', follow);
      follow();
    }
  });
</script>
