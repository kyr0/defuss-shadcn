  // df$ comes with the component bundle - a full-screen page loads it after this module
  addEventListener('DOMContentLoaded', () => {
    // light or dark: the switch writes the page's dark class (every look has both palettes); it follows the class when
    // something else sets it (the docs card follows the site's mode)
    const siteMode = df$('#in-agency [data-site-mode]');
    const syncMode = () => siteMode.prop('checked', document.documentElement.classList.contains('dark'));
    siteMode.on('change', () => {
      document.documentElement.classList.toggle('dark', siteMode.prop('checked'));
      document.dispatchEvent(new CustomEvent('defuss-theme-change'));
    });
    new MutationObserver(syncMode).observe(document.documentElement, { attributes: true, attributeFilter: ['class'] });
    syncMode();

    // the look: one of the studio's three identities; the mesh reads its token colours again on defuss-theme-change
    const root = df$('#in-agency').get(0);
    const look = df$('#in-agency [data-site-look]');
    look.val(root.dataset.look).on('change', () => {
      root.dataset.look = look.val();
      document.dispatchEvent(new CustomEvent('defuss-theme-change'));
    });

    // the cases: Next closes this case and opens the next one
    df$('#in-agency [data-ag-next]').on('click', (event) => {
      const button = event.currentTarget;
      df$(button).closest('dialog').get(0).close();
      df$(`#${button.dataset.agNext}`).get(0).showModal();
    });

    // the brief: what a mix of services usually takes, from the last 60 projects
    const SERVICES = {
      brand: { name: 'a brand identity', weeks: [10, 16], from: 60 },
      web: { name: 'a website', weeks: [8, 14], from: 40 },
      product: { name: 'a digital product', weeks: [12, 24], from: 90 },
      pack: { name: 'packaging', weeks: [8, 12], from: 35 },
      system: { name: 'a design system', weeks: [10, 16], from: 70 },
      motion: { name: 'motion and film', weeks: [4, 8], from: 25 },
    };
    const brief = df$('#in-agency [data-ag-brief]');
    const listOf = (names) =>
      names.length < 2 ? names.join('') : `${names.slice(0, -1).join(', ')} and ${names[names.length - 1]}`;
    const estimate = () => {
      const fd = new FormData(brief.get(0));
      const picked = fd.getAll('service').map((s) => SERVICES[String(s)]);
      const budget = Number(fd.get('budget') ?? 0);
      if (!picked.length) {
        df$('#in-agency [data-ag-weeks]').text('-');
        df$('#in-agency [data-ag-budget]').text('-');
        df$('#in-agency [data-ag-team]').text('-');
        df$('#in-agency [data-ag-fit]').text('Tick what you need - the estimate comes from our last 60 projects.');
        return '';
      }
      // overlapping work: the longest service sets the pace, every other one adds two weeks
      const extra = (picked.length - 1) * 2;
      const low = Math.max(...picked.map((s) => s.weeks[0])) + extra;
      const high = Math.max(...picked.map((s) => s.weeks[1])) + extra;
      const from = Math.round(picked.reduce((sum, s) => sum + s.from, 0) * (picked.length > 1 ? 0.85 : 1));
      df$('#in-agency [data-ag-weeks]').text(`${low}–${high} weeks`);
      df$('#in-agency [data-ag-budget]').text(`from €${from}k`);
      df$('#in-agency [data-ag-team]').text(`${Math.min(3 + picked.length, 8)} people`);
      const mix = listOf(picked.map((s) => s.name));
      const fit = !budget
        ? `${mix[0].toUpperCase()}${mix.slice(1)}. Pick a budget and we tell you whether it fits.`
        : budget >= from
          ? `${mix[0].toUpperCase()}${mix.slice(1)} fits your budget.`
          : `Your budget is below what ${mix} usually takes - we will suggest what to do first.`;
      df$('#in-agency [data-ag-fit]').text(fit);
      return mix;
    };
    brief.on('change', estimate);
    estimate();
    // a service card ticks its box in the brief, then its link scrolls there
    df$('#in-agency [data-ag-service]').on('click', (event) => {
      df$(`#in-agency [name="service"][value="${event.currentTarget.dataset.agService}"]`).prop('checked', true);
      estimate();
    });
    const sent = df$('#ag-sent');
    brief.on('submit', (event) => {
      event.preventDefault();
      const fd = new FormData(event.currentTarget);
      const mix = estimate();
      const name = String(fd.get('name') ?? '')
        .trim()
        .split(/\s+/)[0];
      df$('#in-agency [data-ag-sent-name]').text(name ? `, ${name}` : '');
      df$('#in-agency [data-ag-sent-what]').text(
        `${fd.get('company')}${mix ? `: ${mix}` : ''} - ${String(fd.get('when')).toLowerCase()}.`,
      );
      sent.get(0).showModal();
    });
    sent.on('close', () => {
      brief.get(0).reset();
      estimate();
    });

    // the jobs: one application dialog, named after the role that opened it
    const apply = df$('#ag-apply');
    df$('#in-agency [data-ag-role]').on('click', (event) => {

      apply.attr('data-sent', null);
      apply.find('form').get(0).reset();
