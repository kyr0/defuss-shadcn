  // df$ comes with the component bundle - a full-screen page loads it after this module
  addEventListener('DOMContentLoaded', () => {
    // light or dark: the switch writes the page's dark class (every look has both palettes); it follows the class when
    // something else sets it (the docs card follows the site's mode)
    const siteMode = df$('#pf-minimal [data-site-mode]');
    const syncMode = () => siteMode.prop('checked', document.documentElement.classList.contains('dark'));
    siteMode.on('change', () => {
      document.documentElement.classList.toggle('dark', siteMode.prop('checked'));
      document.dispatchEvent(new CustomEvent('defuss-theme-change'));
    });
    new MutationObserver(syncMode).observe(document.documentElement, { attributes: true, attributeFilter: ['class'] });
    syncMode();
  });
</script>

<DocText>Built from <DocLink href="resume-header.html">Resume Header</DocLink> (compact) with its status pill and <DocLink href="resume-entry.html">Resume Entry</DocLink>. The page sets its own paper palette (with a dark twin) and two typefaces; the quiet comes from the narrow measure, the details from type alone - a drop cap, outline figures and an outline initial, small caps, accent underlines. The ruled paper is a <DocLink href="surface.html">Surface</DocLink> drifting slowly (still under reduced motion); <code>shape-circle</code> is a <DocLink href="shapes.html">Shapes</DocLink> utility; <DocLink href="motion.html">Motion</DocLink>'s stagger fades the sections in.</DocText>
