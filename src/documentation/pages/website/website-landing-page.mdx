<PageHeader slug="website-landing-page" title="Landing Page" introStyle="max-width:42rem;">
  The sections of a product or marketing page, top to bottom: the site header, a hero, product showcase and logos, section headers to introduce the parts, features, testimonials, stats, pricing, blog teasers, FAQ, contact, newsletter and the footer. Every block is CSS-only and built from the library's components and tokens.
</PageHeader>

<h2 id="template" class="h2-display">The template</h2>

<DocText>The Landing Page blocks composed into one page - the source is the page. The header's sections are routes of <DocLink href="routing.html">defuss-dom-router</DocLink> (<code>#/pricing</code>) that scroll to them - a deep link opens the page there, Back to top is <code>#/</code>. The blog posts (<code>#/blog/context</code>), the products (<code>#/product/analytics</code>) and the legal pages (<code>#/legal/privacy</code>) are pages of their own, filled from one data block in English and German; Book a demo and Get in touch open dialogs, and the links to the blog, the company, the docs and the sign-up lead to those sites. The language switch in the header translates the page with <DocLink href="translation.html">defuss-i18n</DocLink>.</DocText>

        view.find('.lp-legal-nav a').each((_i, link) => {
          if (link.getAttribute('href') === '#/legal/' + slug) link.setAttribute('aria-current', 'page');
          else link.removeAttribute('aria-current');
        });
      },
    };
    const render = () => opened && RENDER[opened.kind](opened.slug);
    // a language change writes the open page again
    locale.subscribe(render);

    // -- routing
    const routes = [
      { id: 'top', path: '/' },
      { id: 'post', path: '/blog/:slug' },
      { id: 'product', path: '/product/:slug' },
      { id: 'legal', path: '/legal/:slug' },
      { id: 'section', path: '/:section' },
      { id: 'missing', path: '*' },
    ];
    const SECTIONS = ['features', 'pricing', 'faq', 'customers', 'testimonials', 'blog'];
    const motion = matchMedia('(prefers-reduced-motion: reduce)').matches ? 'auto' : 'smooth';
    let first = true;
    // a page route shows its page in place of the landing sections; a section route scrolls to its section and
    // marks its header link; #/ is the top
    // where a link went, in words: the heading of the section or the page, read as shown (so in the language shown)
    const announce = (el) => {
      const heading = el && df$(el).find('h1, h2, h3, .mk-logos-caption, .mk-testimonial-context').get(0);
      df$('[data-lp-announce]').text(heading ? heading.textContent.replace(/\s+/g, ' ').trim() : '');
    };
    const show = (to) => {
      const kind = PAGES[to.routeId]?.[to.params.slug] ? to.routeId : null;
      opened = kind && { kind, slug: to.params.slug };
      queueMicrotask(() =>
        announce(
          kind
            ? df$('[data-lp-view="' + kind + '"]').get(0)
            : to.routeId === 'section' && SECTIONS.includes(to.params.section)
              ? df$('#lp-' + to.params.section).get(0)
              : null,
        ),
      );
      df$('#top [data-lp-home]').each((_i, part) => {
        part.hidden = !!kind;
      });
      pageView.prop('hidden', !kind);
      df$('#top [data-lp-view]').each((_i, view) => {
        view.hidden = view.dataset.lpView !== kind;
      });
      if (kind) {
        render();
        df$('.mk-header-nav [data-router-link]').each((_i, link) => {
          if (link.getAttribute('href') === location.hash) link.setAttribute('aria-current', 'true');
          else link.removeAttribute('aria-current');
        });
        scrollTo({ top: 0, behavior: 'instant' });
        first = false;
        return;
      }
      const section = to.routeId === 'section' && SECTIONS.includes(to.params.section) ? to.params.section : null;
      df$('.mk-header-nav [data-router-link]').each((_i, link) => {
        if (link.getAttribute('href') === '#/' + section) link.setAttribute('aria-current', 'true');
        else link.removeAttribute('aria-current');
      });
      // the page opens where its address says - at once ('auto' would take the page's smooth scroll-behavior) and
      // once its images have laid it out (a scroll before the load lands short); a click glides there
      const behavior = first ? 'instant' : motion;
      const go = () => {
        if (section)
          df$('#lp-' + section)
            .get(0)
            .scrollIntoView({ behavior, block: 'start' });
        else if (!first) scrollTo({ top: 0, behavior });
      };
      if (first && section && document.readyState !== 'complete') addEventListener('load', go, { once: true });
      else go();
      first = false;
    };
    // the page scrolls itself (to a section) - the router's own scroll to the top would stop that glide
    const router = createRouter({
      mode: 'hash',
      scroll: 'manual',
      routes,
      prepare: ({ to }) => ({ commit: () => show(to) }),
    });
    router.attachLinks(document);
    if ((await router.start()).status === 'rejected') {
      // no address the router may own - a docs preview is an about:srcdoc frame: the same routes match its hash
      const preview = createRouter({ mode: 'hash', routes, baseUrl: 'https://preview.invalid/' });
      const follow = () => show(preview.resolve(location.hash.startsWith('#/') ? location.hash : '#/'));
      addEventListener('hashchange', follow);
      follow();
    }
  });
</script>
