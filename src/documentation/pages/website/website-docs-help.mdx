        <h1 class="mk-docs-content-title" id="dc-title" data-docs-title>Offline sync</h1>
        <p class="mk-docs-content-lead" data-docs-article
          >How Acme keeps working without a connection, and what happens when two people edit the same thing.</p
        >
        <div class="mk-docs-content-meta" data-docs-article
          ><span><span data-i18n-target="dh-5-1">Updated</span> <time datetime="2026-09-28">28 Sep 2026</time></span
          ><span data-i18n-target="dh-6-1">6 min read</span
          ><a href="#"><i data-lucide="pencil"></i> <span data-i18n-target="dh-7-1">Edit this page</span></a></div
        >
        <p class="mk-docs-content-lead" data-docs-stub data-i18n-target="stub" hidden
          >This page has no article in this demo - Offline sync has one.</p
        >
        <div class="mk-docs-content-prose" data-docs-article>
          <p
            >Every change you make is written to your device first. When you are back online, Acme sends the changes in
            the order you made them - nothing waits for the network.</p
          >
          <h2 id="dc-enable">Turn it on</h2>
          <p
            >Offline sync is on by default for new workspaces. For older ones, enable it in
            <code>Settings → Sync</code> or with the CLI:</p
          >
          <pre><code>acme workspace sync --offline on</code></pre>
          <aside class="mk-docs-content-callout" role="note"
            ><i data-lucide="info"></i
            ><p
              ><strong>Note:</strong> the first sync downloads the whole workspace - on a slow connection this can take
              a few minutes.</p
            ></aside
          >
          <h2 id="dc-conflicts">Conflicts</h2>
          <p
            >When two people change the same field offline, the later change wins and the earlier one is kept in the
            history:</p
          >
          <ul
            ><li>Text merges character by character.</li
            ><li>Fields like status keep the last value.</li
            ><li>Deletions always win over edits.</li></ul
          >
          <table
            ><thead
              ><tr><th>Change</th><th>Resolution</th></tr></thead
            ><tbody
              ><tr><td>Both edit a paragraph</td><td>Merged</td></tr
              ><tr><td>Both set a status</td><td>Latest wins</td></tr
              ><tr><td>One deletes, one edits</td><td>Deleted</td></tr></tbody
            ></table
          >
          <aside class="mk-docs-content-callout" data-tone="warning" role="note"
            ><i data-lucide="triangle-alert"></i
            ><p><strong>Warning:</strong> clearing your browser data before syncing deletes unsynced changes.</p></aside
          >
          <aside class="mk-docs-content-callout" data-tone="tip" role="note"
            ><i data-lucide="lightbulb"></i
            ><p><strong>Tip:</strong> the cloud icon in the header shows how many changes are waiting.</p></aside
          >

  // routing and translation: defuss-dom-router owns the address, defuss-i18n the language
  import { createRouter } from 'https://cdn.jsdelivr.net/npm/defuss-dom-router@0.1.1/dist/index.js';
  import {
    createDomI18n,
    createI18n,
    resolveLocale,
  } from 'https://cdn.jsdelivr.net/npm/defuss-i18n@0.2.0/dist/core.js';

  // df$ comes with the component bundle - a full-screen page loads it after this module
  addEventListener('DOMContentLoaded', async () => {
    const site = df$('#site-docs').get(0);

    // -- translation: the markup is the catalog (the templates at the end of the page)
    const locale = createI18n({
      locale: navigator.languages.map((tag) => resolveLocale(['en', 'de'], tag, [])).find(Boolean) ?? 'en',
    });
    createDomI18n(df$).bind(site, locale);
    const language = df$('[data-app-locale]');
    language.val(locale.locale).on('change', () => locale.setLocale(language.val()));

    // -- routing
    const routes = [
      { id: 'home', path: '/' },
      { id: 'page', path: '/docs/:page' },
      { id: 'missing', path: '*' },
    ];
    // the docs pages in the navigation's order
    const links = [...df$('.mk-docs-navigation a[data-router-link]')];
    const label = (link) => link.firstChild.textContent.trim();
    // the page's link is current; the breadcrumb, the title and the pager follow it; only Offline sync has an article
    let shown = null;
    const show = (to) => {
      shown = to;
      const href = '#/docs/' + (to.routeId === 'page' ? to.params.page : 'offline-sync');
      const at = Math.max(
        0,
        links.findIndex((link) => link.getAttribute('href') === href),
      );
      const link = links[at];
      links.forEach((l) => (l === link ? l.setAttribute('aria-current', 'page') : l.removeAttribute('aria-current')));
      // its section: the heading above its list (a summary of a group, or the heading paragraph)
      const group = link.closest('details')?.querySelector('summary') ?? df$('.mk-docs-navigation-heading').get(0);
      df$('[data-docs-crumb-section]').text(group.textContent.trim());
      df$('[data-docs-crumb]').text(label(link));
      df$('[data-docs-title]').text(label(link));
      const article = link.getAttribute('href') === '#/docs/offline-sync';
      df$('[data-docs-article]').each((_i, part) => {
        part.hidden = !article;
      });
      df$('[data-docs-stub]').get(0).hidden = article;
      for (const [rel, step] of [
        ['prev', -1],
        ['next', 1],
      ]) {
        const near = links[(at + step + links.length) % links.length];
        const pager = df$('.mk-docs-content-pager a[rel="' + rel + '"]');
        pager.attr('href', near.getAttribute('href'));
        pager.find('[data-docs-near]').text(label(near));
      }
    };
    // the section in the breadcrumb is read from its heading again once the language changed
    site.addEventListener('defuss-i18n:change', () => shown && show(shown));
    const router = createRouter({ mode: 'hash', routes, prepare: ({ to }) => ({ commit: () => show(to) }) });
    router.attachLinks(site);
    if ((await router.start()).status === 'rejected') {
      // no address the router may own - a docs preview is an about:srcdoc frame: the same routes match its hash
      const preview = createRouter({ mode: 'hash', routes, baseUrl: 'https://preview.invalid/' });
      const follow = () => show(preview.resolve(location.hash.startsWith('#/') ? location.hash : '#/'));
      addEventListener('hashchange', follow);
      follow();
    }
  });
