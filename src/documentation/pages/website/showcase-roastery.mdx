  // the look: one of the site's designs. An element with data-look-set takes that look's attributes; a shader reads
  // its colors again on defuss-theme-change
  addEventListener('DOMContentLoaded', () => {
    // light or dark: the switch writes the page's dark class (every look has both palettes); it follows the class when
    // something else sets it (the docs card follows the site's mode)
    const siteMode = df$('#sc-roastery [data-site-mode]');
    const syncMode = () => siteMode.prop('checked', document.documentElement.classList.contains('dark'));
    siteMode.on('change', () => {
      document.documentElement.classList.toggle('dark', siteMode.prop('checked'));
      document.dispatchEvent(new CustomEvent('defuss-theme-change'));
    });
    new MutationObserver(syncMode).observe(document.documentElement, { attributes: true, attributeFilter: ['class'] });
    syncMode();

    // the subscription form: sending shows the thank-you (a demo - nothing leaves the page); closing resets it
    df$('#sc-roastery [data-sc-form]').each((_i, form) => {
      const dialog = df$(form).closest('dialog');
      df$(form).on('submit', (event) => {
        event.preventDefault();
        dialog.attr('data-sent', '');
        dialog.find('.sc-done').get(0).focus();
      });
      dialog.on('close', () => {
        dialog.attr('data-sent', null);
        form.reset();
      });
    });

    const root = df$('#sc-roastery').get(0);
    const look = df$('#sc-roastery [data-site-look]');
    look.val(root.dataset.look).on('change', () => {
      root.dataset.look = look.val();
      df$('#sc-roastery [data-look-set]').each((_i, el) => {
        const sets = JSON.parse(el.dataset.lookSet);
        for (const [attr, value] of Object.entries({ ...sets.default, ...sets[root.dataset.look] }))
          el.setAttribute(attr, value);
      });
      document.dispatchEvent(new CustomEvent('defuss-theme-change'));
    });
  });
</script>
