  Blocks for a newsroom, a press page or a changelog front: the header with its topics, a ticker of the latest headlines and the stories themselves - as the lead, as rows with thumbnails, or as a compact list. Every block is CSS-only and built from the library's components and tokens.

<h2 id="template" class="h2-display">The template</h2>

<DocText>The News blocks composed into one page - the source is the page. The section links are routes of <DocLink href="routing.html">defuss-dom-router</DocLink> (<code>#/section/engineering</code>) that show a section's stories, and the language switch translates the frame with <DocLink href="translation.html">defuss-i18n</DocLink> - the dates are formatted for the language, the stories stay as written.</DocText>

<script type="module">
  // routing and translation: defuss-dom-router owns the address, defuss-i18n the language
  import { createRouter } from 'https://cdn.jsdelivr.net/npm/defuss-dom-router@0.1.1/dist/index.js';
  import {
    createDomI18n,
    createI18n,
    resolveLocale,
  } from 'https://cdn.jsdelivr.net/npm/defuss-i18n@0.2.0/dist/core.js';

  // df$ comes with the component bundle - a full-screen page loads it after this module
  addEventListener('DOMContentLoaded', async () => {
    const site = df$('#site-news').get(0);

    // -- translation: the markup is the catalog (the templates at the end of the page)
    const locale = createI18n({
      locale: navigator.languages.map((tag) => resolveLocale(['en', 'de'], tag, [])).find(Boolean) ?? 'en',
    });
    createDomI18n(df$).bind(site, locale, {
      values: () => ({
        today: locale.formatDate(Date.UTC(2026, 9, 1), { dateStyle: 'full', timeZone: 'UTC' }),
        d1001: locale.formatDate(Date.UTC(2026, 9, 1), { dateStyle: 'medium', timeZone: 'UTC' }),
        d0928: locale.formatDate(Date.UTC(2026, 8, 28), { dateStyle: 'medium', timeZone: 'UTC' }),
        d0927: locale.formatDate(Date.UTC(2026, 8, 27), { dateStyle: 'medium', timeZone: 'UTC' }),
      }),
    });
    const language = df$('[data-app-locale]');
    language.val(locale.locale).on('change', () => locale.setLocale(language.val()));

    // -- routing
    const routes = [
      { id: 'front', path: '/' },
      { id: 'section', path: '/section/:topic' },
      { id: 'missing', path: '*' },
    ];
    // a section shows its stories only; its link is the current one
    const show = (to) => {
      const topic = to.routeId === 'section' ? to.params.topic : null;
      df$('#site-news [data-topic]').each((_i, story) => {
        story.hidden = !!topic && story.dataset.topic !== topic;
      });
      df$('#site-news nav [data-router-link]').each((_i, link) => {
        if (link.getAttribute('href') === '#' + (topic ? to.path : '/')) link.setAttribute('aria-current', 'page');
        else link.removeAttribute('aria-current');
      });
    };
    const router = createRouter({ mode: 'hash', routes, prepare: ({ to }) => ({ commit: () => show(to) }) });
    router.attachLinks(site);
    if ((await router.start()).status === 'rejected') {
      // no address the router may own - a docs preview is an about:srcdoc frame: the same routes match its hash
      const preview = createRouter({ mode: 'hash', routes, baseUrl: 'https://preview.invalid/' });
      const follow = () => show(preview.resolve(location.hash.startsWith('#/') ? location.hash : '#/'));
      addEventListener('hashchange', follow);
      follow();
    }
  });
</script>
