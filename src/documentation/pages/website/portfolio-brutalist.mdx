      });
      page.find('[data-p-shot]').each((_i, figure) => {

          .attr('alt', alt);
        if (caption) df$(figure).find('figcaption').text(caption);
      });
      page.find('[rel="prev"]').attr('href', '#/work/' + prev);
      page.find('[data-p-prev]').text(PROJECTS[prev].title);
      page.find('[rel="next"]').attr('href', '#/work/' + next);

      if (slug) fill(slug);

    // a section of the home view: the home view shows again (a page hides it) and the section scrolls into view - on
