<script>
  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-waterfall');
    const INK = '#F8E8B6', MUTED = '#8F856D', GOLD = '#E2B857', TEAL = '#5AB7A5', THREAD = '#F2D675';
    const G = { left: 150, right: 130, top: 320, bottom: 150 };

    // the waterfall transform: invisible baseline + signed bars + running total
    const steps = [['Tickets', 6.4], ['Streaming window', -4.8], ['Concessions', 3.1], ['Energy', -1.9], ['Premium seats', 2.7], ['Private events', 1.6]];
    const labels = ['2024', ...steps.map((s) => s[0]), '2025'];
    let run = 48.2;
    const base = [0], gain = [48.2], loss = ['-'], totals = [48.2];
    for (const [, d] of steps) {
      if (d >= 0) { base.push(run); gain.push(d); loss.push('-'); run += d; }
      else { run += d; base.push(run); gain.push('-'); loss.push(-d); }
      run = Math.round(run * 10) / 10;
      totals.push(run);
    }
    base.push(0); gain.push(run); loss.push('-'); totals.push(run);
    const named = (arr) => arr.map((v, i) => ({ name: labels[i], value: v }));
    const money = (v) => (v === '-' ? '' : v);

    const bridgeAxes = {
      grid: G,
      xAxis: { type: 'category', data: labels, axisLabel: { interval: 0, width: 160, overflow: 'break', lineHeight: 24, fontSize: 20, color: INK, fontWeight: 700 } },
      yAxis: { type: 'value', min: 0, max: 60, interval: 15, axisLabel: { color: MUTED } },
    };
    const bridge = [
      { id: 'base', type: 'bar', stack: 'w', silent: true, data: named(base), itemStyle: { color: 'transparent' } },
      { id: 'gain', type: 'bar', stack: 'w', barWidth: '58%', data: named(gain),
        itemStyle: { color: (p) => (p.dataIndex === 0 || p.dataIndex === labels.length - 1 ? INK : TEAL) },
        label: { show: true, position: 'top', formatter: (p) => money(p.value), color: INK, fontWeight: 700 } },
      { id: 'loss', type: 'bar', stack: 'w', barWidth: '58%', data: named(loss), itemStyle: { color: GOLD },
        label: { show: true, position: 'bottom', formatter: (p) => (p.value === '-' ? '' : '−' + p.value), color: INK, fontWeight: 700 } },
    ];

    // ranked movers: the SAME series ids and item names, now horizontal
    const ranked = steps.slice().sort((a, b) => Math.abs(b[1]) - Math.abs(a[1]));

    df$.shadcn.chart.deck(deck, {
      base: { color: [TEAL, GOLD, INK, MUTED, THREAD], tooltip: { trigger: 'item' } },
      states: {
        bridge: { ...bridgeAxes, series: bridge },
        thread: { ...bridgeAxes, series: [...bridge,
          { id: 'total', type: 'line', data: named(totals), z: 5, symbolSize: 16, step: false,
            lineStyle: { color: THREAD, width: 4 }, itemStyle: { color: THREAD, borderColor: '#07111F', borderWidth: 3 } }] },
        movers: {
          grid: { left: 330, right: 170, top: 320, bottom: 140 },
          xAxis: { type: 'value', min: -8, max: 8, axisLabel: { color: MUTED, formatter: (v) => (v > 0 ? '+' : v < 0 ? '−' : '') + Math.abs(v) } },
          yAxis: { type: 'category', inverse: true, data: ranked.map((s) => s[0]), axisLine: { show: false }, axisLabel: { color: INK, fontWeight: 700 } },
          series: [
            { id: 'gain', type: 'bar', stack: 'm', barWidth: '62%', data: ranked.map(([n, d]) => ({ name: n, value: d > 0 ? d : '-' })),
              itemStyle: { color: TEAL }, label: { show: true, position: 'right', formatter: (p) => (p.value === '-' ? '' : '+' + p.value), color: INK, fontWeight: 700 } },
            { id: 'loss', type: 'bar', stack: 'm', barWidth: '62%', data: ranked.map(([n, d]) => ({ name: n, value: d < 0 ? d : '-' })),
              itemStyle: { color: GOLD }, label: { show: true, position: 'left', formatter: (p) => (p.value === '-' ? '' : '−' + Math.abs(p.value)), color: INK, fontWeight: 700 } },
          ],
        },
        mix: {
          series: [{ id: 'gain', type: 'pie', radius: ['30%', '54%'], center: ['50%', '64%'],
            color: [TEAL, GOLD, THREAD, '#9FC9C0', MUTED],
            itemStyle: { borderColor: '#07111F', borderWidth: 4, borderRadius: 6 },
            label: { formatter: '{b}\n€{c} M', color: INK, fontWeight: 700, lineHeight: 34 },
            labelLine: { length: 26, length2: 36, lineStyle: { color: MUTED } },
            data: [['Tickets', 29.1], ['Concessions', 14.6], ['Premium seats', 6.2], ['Private events', 3.4], ['Advertising', 2.0]].map(([n, v]) => ({ name: n, value: v })) }],
        },
      },
    });
  });
