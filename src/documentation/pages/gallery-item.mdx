<PageHeader slug="gallery-item" title="Gallery Item" component="gallery-item">
  Presents one image with a caption and enlarged-view action - the picture opens a large view in a native modal dialog. Part of the <DocLink href="website-media.html">Media</DocLink> blocks.
</PageHeader>

<SkillPanel component="gallery-item">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;figure&gt;</code>: a <code>&lt;button commandfor command="show-modal"&gt;</code> around the thumbnail, a <code>&lt;figcaption&gt;</code>, and a <code>&lt;dialog&gt;</code> with the large picture, its caption and a close button in a <code>&lt;form method="dialog"&gt;</code>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/figure" title="media with its caption"><code>&lt;figure&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/dialog" title="native dialog; method=&quot;dialog&quot; forms close it"><code>&lt;dialog&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/button#command" title="a button opens a dialog declaratively"><code>commandfor / command</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/@starting-style" title="entry animations"><code>@starting-style</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/object-fit" title="pictures crop instead of stretching"><code>object-fit</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-gallery-item</code>
    <code>.mk-gallery-item-open</code>
    <code>.mk-gallery-item-zoom</code>
    <code>.mk-gallery-item-caption</code>
    <code>.mk-gallery-item-viewer</code>
    <code>.mk-gallery-item-viewer-inner</code>
    <code>.mk-gallery-item-close</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant</code>: <code>overlay</code>, <code>framed</code>.</SkillText>
  </SkillField>
</SkillPanel>
