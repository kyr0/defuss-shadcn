<script>
  document.addEventListener('DOMContentLoaded', () => {
    df$('[data-share="copy"]').on('click', async () => {
      await navigator.clipboard?.writeText(location.href).catch(() => {});
      df$('#share-status').text('Link copied');
      setTimeout(() => df$('#share-status').text(''), 2000);
    });
  });
</script>
