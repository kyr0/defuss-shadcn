      yAxis: { type: 'value', min: -0.5, max: 9.5, inverse: true, show: false },
    };
    const waffle = (col) => {
      let at = 0;
      return {
        ...board,
        legend: { orient: 'vertical', left: 1010, top: 'middle', icon: 'rect', itemWidth: 26, itemHeight: 26, itemGap: 26, textStyle: { color: TEXT, fontWeight: 700 },
          formatter: (n) => n + '  ' + platforms.find((p) => p[0] === n)[col] },
        series: platforms.map(([name, color, ...counts]) => {
          const n = counts[col - 2];
          const data = Array.from({ length: n }, (_, i) => [(at + i) % 10, Math.floor((at + i) / 10)]);
          at += n;
          return { id: name, name, type: 'scatter', dataGroupId: name, symbol: 'rect', symbolSize: 38, data, itemStyle: { color },
            universalTransition: { enabled: true, divideShape: 'split' } };
        }),
      };
    };
    const bars = (list, value, fmt, max) => ({
      grid: { left: 260, right: 190, top: 330, bottom: 150 },
      xAxis: { type: 'value', max, axisLabel: { color: MUTED }, splitLine: { lineStyle: { color: 'rgba(255,249,216,.08)' } } },
      yAxis: { type: 'category', inverse: true, data: list.map((p) => p[0]), axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 700 } },
      series: [{ id: 'platforms', type: 'bar', barWidth: '56%', universalTransition: { enabled: true, divideShape: 'split' },
        data: list.map((p) => ({ name: p[0], value: p[value], groupId: p[0], itemStyle: { color: p[1], borderRadius: 0 } })),
        label: { show: true, position: 'right', formatter: fmt, color: TEXT, fontWeight: 700 } }],
    });

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        y2025: waffle(2),
        y2020: waffle(3),
        bars: bars(platforms, 2, '{c} / 100', 50),
        hours: bars(platforms.slice().sort((a, b) => b[4] - a[4]), 4, '{c} h', 12),
      },
    });
  });
</script>
