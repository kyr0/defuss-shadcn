<PageHeader slug="text-media" title="Text Media" component="text-media">
  Pairs explanatory text with an image, illustration, or video - side by side, reversed, or with the text card overlapping the picture. Part of the <DocLink href="website-media.html">Media</DocLink> blocks.
</PageHeader>

<SkillPanel component="text-media">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;section&gt;</code>: an eyebrow, a heading, paragraphs, an optional check list and link, beside a <code>&lt;figure&gt;</code> with an <code>&lt;img&gt;</code>, an SVG illustration or a <code>&lt;video&gt;</code>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/figure" title="media with its caption"><code>&lt;figure&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_containment/Container_queries" title="the layout follows the block width, not the viewport"><code>Container queries</code></ApiPill>
