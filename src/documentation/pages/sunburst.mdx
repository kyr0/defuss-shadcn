    const deck = document.getElementById('deck-sunburst');
    const TEXT = '#EEF5FF', MUTED = '#71849B', PANEL = '#06101F';
    const fields = [
      ['Exoplanets', '#FFB45A', [['Transits', 380], ['Radial velocity', 200], ['Direct imaging', 80]], 4.1],
      ['Galaxies', '#6DA7FF', [['High-redshift', 300], ['Nearby', 250]], 2.8],

      ['Solar system', '#FFE096', [['Asteroids', 140], ['Comets', 80]], 1.5],
    ];
    const sum = (kids) => kids.reduce((s, k) => s + k[1], 0);
    // one hierarchy for every state — stable names let sunburst and treemap morph node for node
    const tree = (focus) => fields.map(([n, c, kids]) => {
      const on = !focus || n === focus, col = on ? c : 'rgba(113,132,155,.28)';
      return { name: n, value: sum(kids), itemStyle: { color: col },
        children: kids.map(([k, v]) => ({ name: k, value: v, itemStyle: { color: col, opacity: on ? 0.72 : 1 } })) };
    });
    const rings = (focus) => ({
      series: [{ id: 'hours', type: 'sunburst', data: tree(focus), sort: null, nodeClick: false, center: ['50%', '64%'], radius: ['14%', '66%'],
        emphasis: { focus: 'ancestor' }, itemStyle: { borderColor: PANEL, borderWidth: 3, borderRadius: 6 },
        label: { rotate: 'radial', color: '#020617', fontWeight: 700, minAngle: 8 },
        levels: [{}, { r0: '14%', r: '38%', label: { fontSize: 18 } }, { r0: '38%', r: '66%', label: { fontSize: 16 } }] }],
    });
    const pressure = fields.map((f) => [f[0], f[3], f[1]]).sort((a, b) => b[1] - a[1]);

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        rings: rings(null),
        exo: rings('Exoplanets'),
        blocks: {
          series: [{ id: 'hours', type: 'treemap', data: tree(null), roam: false, nodeClick: false, breadcrumb: { show: false }, sort: false,
            top: 330, left: 150, right: 150, bottom: 110,
            label: { show: true, color: '#020617', formatter: '{b}\n{c} h', fontWeight: 700, fontSize: 17, lineHeight: 22 },
            upperLabel: { show: true, height: 36, color: TEXT, fontWeight: 800, fontSize: 17, formatter: '{b} · {c} h' },
            levels: [
              { itemStyle: { borderColor: PANEL, borderWidth: 0, gapWidth: 8 }, upperLabel: { show: false } },
              { itemStyle: { borderColor: PANEL, borderWidth: 5, gapWidth: 3, borderRadius: 8 } },
              { itemStyle: { borderColor: PANEL, borderWidth: 2, gapWidth: 0, borderRadius: 6 } },
            ] }],
        },
        demand: {
          grid: { left: 310, right: 220, top: 340, bottom: 140 },
