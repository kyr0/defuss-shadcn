<PageHeader slug="progress" title="Progress" component="progress">
  A progress bar showing completion percentage. Built on the native <code>&lt;progress&gt;</code> element.
</PageHeader>

<SkillPanel component="progress">
  <SkillField label="Native basis">
<SkillText><code>&lt;progress&gt;</code> element with CSS styling for the track and indicator bar.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/progress" title="native HTML progress indicator"><code>&lt;progress&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/::-webkit-progress-bar" title="track pseudo-element (WebKit)"><code>::-webkit-progress-bar</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/::-webkit-progress-value" title="indicator pseudo-element (WebKit)"><code>::-webkit-progress-value</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/::-moz-progress-bar" title="indicator pseudo-element (Firefox)"><code>::-moz-progress-bar</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.progress</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Accessibility">
<SkillText dense>• <code>&lt;progress&gt;</code> is natively accessible — screen readers announce the percentage</SkillText>
  <SkillText dense>• The text content inside <code>&lt;progress&gt;</code> is the fallback for non-supporting browsers</SkillText>
  <p class="text-muted-foreground" style="margin:0 0 0;">• Use <code>aria-label</code> if the progress bar lacks a visible label</p>
  </SkillField>
</SkillPanel>
