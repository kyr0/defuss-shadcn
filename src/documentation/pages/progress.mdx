</PageHeader>

<SkillPanel component="progress">
  <SkillField label="Native basis">
<SkillText><code>&lt;progress&gt;</code> element with CSS styling for the track and indicator bar.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">

  <SkillField label="Classes">
<SkillRow>

  </SkillRow>
  </SkillField>

  <SkillField label="Accessibility">
<SkillText dense>• <code>&lt;progress&gt;</code> is natively accessible - screen readers announce the percentage</SkillText>
  <SkillText dense>• The text content inside <code>&lt;progress&gt;</code> is the fallback for non-supporting browsers</SkillText>
  <p class="text-muted-foreground m-0">• Use <code>aria-label</code> if the progress bar lacks a visible label</p>
  </SkillField>

```html example label="Determinate" hint="Shows a specific completion percentage."
<progress class="progress" value="66" max="100">66%</progress>
```

```html example label="Values" hint="Different completion amounts."
<progress class="progress" value="10" max="100">10%</progress>
<progress class="progress" value="40" max="100">40%</progress>
<progress class="progress" value="75" max="100">75%</progress>
```

```html example label="Indeterminate" hint="Animated state when completion amount is unknown. Omit the value attribute."
<progress class="progress" max="100">Loading...</progress>
```
