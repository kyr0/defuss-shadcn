    <h2 id="pb-abstract">Abstract</h2>
    <p>A word error rate is an average; the errors are in the utterances. We show how a reader can browse all of them in the paper: a grid of 100,000 utterances with their speaker, source, duration and errors, and a tree of the corpus they come from. The data here is synthetic - generated deterministically - so that the page stays small and every reader sees the same rows.</p>

<script>
  document.addEventListener('DOMContentLoaded', () => {
    const sources = ['Common Voice', 'Tuda-De', 'Multilingual LibriSpeech'];
    // deterministic synthetic data: utterance i always reads the same
    const rows = Array.from({ length: 100_000 }, (_, i) => {
      const words = 4 + ((i * 7) % 23);
      const errors = (i * 13) % 29 === 0 ? 3 : (i * 11) % 7 === 0 ? 1 : 0;
      return {
        id: 1 + i,
        source: sources[i % 3],
        speaker: 'spk-' + String(1 + ((i * 17) % 3000)).padStart(4, '0'),
        seconds: Math.round((1.5 + words * 0.42 + ((i * 3) % 10) / 10) * 10) / 10,
        words,
        errors,
        wer: Math.round((errors / words) * 1000) / 10,
      };
    });
    df$.shadcn.dataGrid.setSource(document.getElementById('pb-grid'), rows);

    const nodes = [];
    let id = 0;
    for (const source of sources) {
      const s = ++id;
      nodes.push({ id: s, parentId: null, name: source });
      for (let g = 1; g <= 100; g++) {
        const group = ++id;
        nodes.push({ id: group, parentId: s, name: 'Group ' + g });
        for (let k = 1; k <= 10; k++) {
          const speaker = ++id;
          nodes.push({ id: speaker, parentId: group, name: 'Speaker ' + g + '.' + k });
          for (let u = 1; u <= 30; u++) nodes.push({ id: ++id, parentId: speaker, name: 'Utterance ' + g + '.' + k + '.' + u });
        }
      }
    }
    df$.shadcn.dataTree.setSource(document.getElementById('pb-tree'), nodes);
  });
</script>
