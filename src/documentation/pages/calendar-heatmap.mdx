  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-calendar');
    const TEXT = '#43333B', MUTED = '#A08F97', PINK = '#E26C8D';
    // the diary (deterministic): runs on Tue/Thu/Sun, some Wed/Sat, an August
    // break, the marathon on 12 October and a recovery week after it
    const start = Date.UTC(2025, 0, 1);
    const diary = [];
    for (let i = 0; i < 365; i++) {
      const d = new Date(start + i * 864e5), wd = d.getUTCDay(), iso = d.toISOString().slice(0, 10);
      const r = (Math.sin(i * 9.17) + 1) / 2;
      const season = 0.75 + 0.35 * Math.sin(((i - 60) / 365) * Math.PI * 2 + Math.PI / 2);
      let km = wd === 0 ? 16 + 10 * season + r * 4 : wd === 2 || wd === 4 ? 8 + 4 * season + r * 2 : wd === 6 ? (r > 0.35 ? 6 + r * 4 : 0) : wd === 3 ? (r > 0.55 ? 5 + r * 3 : 0) : 0;
      if (d.getUTCMonth() === 7 && d.getUTCDate() >= 4 && d.getUTCDate() <= 17) km = 0;
      if (iso > '2025-10-12' && iso <= '2025-10-19') km = 0;
      km = iso === '2025-10-12' ? 42.2 : km * 0.72;
      diary.push([iso, Math.round(km * 10) / 10]);
    }
    const months = [190, 184, 216, 194, 186, 181, 175, 96, 152, 158, 182, 191];
    // 16 weeks to race day: weekly sums of the diary, Monday-based
    const race = diary.findIndex((d) => d[0] === '2025-10-12');
    const build = Array.from({ length: 16 }, (_, w) => {
      const end = race - (15 - w) * 7;
      return Math.round(diary.slice(Math.max(0, end - 6), end + 1).reduce((s, d) => s + d[1], 0));
    });
    const G = { left: 150, right: 150, top: 330, bottom: 150 };
    const cols = (labels, values, hot) => ({
      grid: G,
      xAxis: { type: 'category', data: labels, axisLabel: { color: TEXT, fontWeight: 700 } },
      yAxis: { type: 'value', axisLabel: { color: MUTED } },
      series: [{ id: 'km', type: 'bar', barWidth: '56%',
        data: values.map((v, i) => ({ value: v, itemStyle: { color: hot(i) ? PINK : '#F8BBD0', borderRadius: [10, 10, 0, 0] } })),
        label: { show: true, position: 'top', color: TEXT, fontWeight: 700 } }],
    });

      base: { tooltip: { trigger: 'item' } },
      states: {

          visualMap: { show: false, min: 0, max: 24, inRange: { color: ['#FCE4EC', '#F48FB1', PINK, '#8E2C4F'] } },
          calendar: { top: 360, left: 150, right: 110, bottom: 190, range: '2025', cellSize: ['auto', 'auto'],
