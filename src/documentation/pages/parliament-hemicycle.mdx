        legend: { top: 262, right: 112, textStyle: { color: TEXT } },
        graphic: [{ type: 'text', left: 'center', top: 650, style: { text: lit ? '211' : '201', fill: lit ? '#8A2E2A' : TEXT, font: "400 72px Baskerville, Georgia, serif" } },
                  { type: 'text', left: 'center', top: 740, style: { text: lit ? 'progressive coalition' : 'needed to govern', fill: MUTED, font: "400 26px Baskerville, Georgia, serif" } }],
        series: groups.map(([n, s, c]) => {
          const data = seats.slice(at, at + s).map((p) => [p.x, p.y]);
          at += s;
          return { id: n, name: n, type: 'scatter', dataGroupId: n, symbolSize: 15, data,
            universalTransition: { enabled: true, divideShape: 'split' },
            itemStyle: { color: !lit || coalition.includes(n) ? c : DIM } };
        }),
      };
    };
    const bars = (list, value, fmt, axis) => ({
      grid: { left: 300, right: 190, top: 330, bottom: 150 },
      xAxis: axis,
      yAxis: { type: 'category', inverse: true, data: list.map((g) => g[0]), axisLine: { show: false }, axisLabel: { color: TEXT } },
      series: [{ id: 'seats', type: 'bar', barWidth: '60%', universalTransition: { enabled: true, divideShape: 'split' },
        data: list.map((g) => ({ name: g[0], value: g[value], groupId: g[0], itemStyle: { color: g[2], borderRadius: 0 }, label: { position: g[value] < 0 ? 'left' : 'right' } })),
        label: { show: true, formatter: fmt, color: TEXT, fontWeight: 700 } }],
    });
    const bySeats = groups.slice().sort((a, b) => b[1] - a[1]);
    const byChange = groups.slice().sort((a, b) => b[3] - a[3]);

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        chamber: arc(false),
        coalition: arc(true),
        parties: bars(bySeats, 1, '{c}', { type: 'value', max: 140, axisLabel: { color: MUTED },
          splitLine: { lineStyle: { color: 'rgba(39,35,29,.1)' } } }),
        change: bars(byChange, 3, (p) => (p.value > 0 ? '+' : '') + p.value, { type: 'value', min: -20, max: 30, axisLabel: { color: MUTED },
          splitLine: { lineStyle: { color: 'rgba(39,35,29,.1)' } } }),
      },
    });
  });
