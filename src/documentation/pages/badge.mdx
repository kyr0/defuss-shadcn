<PageHeader slug="badge" title="Badge" component="badge">
  A small visual indicator for status, labels, or counts. Built on a <code>&lt;span&gt;</code> element with variant styling via <code>data-variant</code>. No JavaScript required.
</PageHeader>

<SkillPanel component="badge">
  <SkillField label="Native basis">
<SkillText><code>&lt;span&gt;</code> element. No interactivity required — pure visual indicator.</SkillText>
  </SkillField>

<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/span" title="inline container for phrasing content"><code>&lt;span&gt;</code></ApiPill>

  </SkillField>
