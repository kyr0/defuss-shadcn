<PageHeader slug="tree-view" title="Tree View" component="tree-view">
  A hierarchical tree for browsing nested data like file systems. Uses native <code>&lt;details&gt;</code> for expand/collapse with ARIA tree roles.
</PageHeader>

<SkillPanel component="tree-view">
  <SkillField label="Native basis">
<SkillText>Nested <code>&lt;ul&gt;</code> elements with <code>role="tree"</code> / <code>role="treeitem"</code> ARIA pattern for hierarchical data.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/ul" title="nested list structure"><code>&lt;ul&gt;</code></ApiPill>
    <ApiPill href="https://www.w3.org/WAI/ARIA/apg/patterns/treeview/" title="ARIA tree roles and keyboard navigation"><code>Tree View WAI-ARIA pattern</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/details" title="native expand/collapse for branches"><code>&lt;details&gt;</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.tree</code>
    <code>.tree-group</code>
    <code>.tree-item</code>
    <code>.tree-branch</code>
    <code>.tree-branch-trigger</code>
    <code>.tree-leaf</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Accessibility">
<SkillText dense>• Root <code>&lt;ul&gt;</code> has <code>role="tree"</code> and <code>aria-label</code></SkillText>
  <SkillText dense>• Branch items have <code>role="treeitem"</code> and <code>aria-expanded</code></SkillText>
  <SkillText dense>• Leaf items have <code>role="treeitem"</code> without <code>aria-expanded</code></SkillText>
  <SkillText dense>• Nested groups use <code>role="group"</code></SkillText>
  <SkillText dense>• <code>data-selectable</code> tree: <code>aria-selected</code> on every operable item, click / Enter / Space selects, <code>tree-select</code> event</SkillText>
  <SkillText dense>• <code>aria-disabled="true"</code> items stay focusable but can't be opened, selected or followed</SkillText>
  <p class="text-muted-foreground m-0">• Keyboard: Arrow keys navigate, Enter/Space toggles branches (and selects in a selectable tree)</p>
  </SkillField>
</SkillPanel>

  <ul class="text-sm text-muted-foreground m-0 mb-3 ps-5">
    <li><code>default</code> - the branch's authored open/closed state (restored from the init snapshot)</li>

  </ul>
