    const G = { left: 150, right: 190, top: 330, bottom: 150 };

      xAxis: { type: 'category', data: weeks.slice(from, to), boundaryGap: false, axisLabel: { color: MUTED, interval: to - from > 20 ? 4 : 1 } },
