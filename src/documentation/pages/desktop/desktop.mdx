<PageHeader slug="desktop" title="Desktop" component="desktop">
  A whole desktop in one element: the <DocLink href="window-manager.html">Window Manager</DocLink> with the <DocLink href="desktop-manager.html">Desktop Manager</DocLink>'s icons under its windows, a <DocLink href="taskbar.html">Taskbar</DocLink> with its <DocLink href="start-button.html">Start Button</DocLink>, <DocLink href="tray.html">Tray</DocLink> and <DocLink href="start-menu.html">Start Menu</DocLink>, and a <DocLink href="logon.html">Logon</DocLink> screen. <code>data-variant</code> gives it a look, the way the old systems let you switch them: the glass of <DocLink href="desktop-windows-7.html">7</DocLink>, the blue bars of <DocLink href="desktop-windows-xp.html">XP</DocLink>, the bevelled grey of <DocLink href="desktop-windows-98.html">98</DocLink> and <DocLink href="desktop-windows-95.html">95</DocLink>, or the <DocLink href="desktop-modern.html">modern</DocLink> look of the system's own tokens. The markup and every component API stay the same - a look is tokens on the desktop and chrome for the components inside, with remade icons (<code>.desktop-glyph</code>), wallpapers and, for 95 and 98, a pixel font.
</PageHeader>

  document.addEventListener('DOMContentLoaded', () => {
    const root = document.querySelector('#dk-desktop');
    const logon = root.querySelector('.logon');
    const open = (value) => {
      if (value === 'logoff' || value === 'shutdown') return df$.shadcn.logon.lock(logon);
      if (value) df$.shadcn.win.open(value);
    };
    root
      .querySelector(':scope > .window-manager > .desktop-manager')
      .addEventListener('desktop-open', (e) => e.detail.values.forEach(open));
    root.querySelector('.start-menu').addEventListener('start-select', (e) => open(e.detail.value));
    logon.addEventListener('logon-submit', () => df$.shadcn.logon.done(logon));
    logon.addEventListener('logon-power', () => df$.shadcn.logon.fail(logon, 'It is now safe to keep reading.'));
    root
      .querySelectorAll('[data-close]')
      .forEach((b) => b.addEventListener('click', () => b.closest('dialog').close()));
    // the calculator: digits and one operation at a time
    const display = root.querySelector('[data-calc-display]');
    let acc = null,
      op = null,
      fresh = true;
    const calc = (a, b) => ({ '+': a + b, '-': a - b, '*': a * b, '/': b ? a / b : NaN })[op] ?? b;
    root.querySelectorAll('[data-calc]').forEach((b) =>
      b.addEventListener('click', () => {
        const c = b.dataset.calc;
        const cur = parseFloat(display.textContent);
        if (/[0-9.]/.test(c)) {
          display.textContent = fresh || display.textContent === '0' ? (c === '.' ? '0.' : c) : display.textContent + c;
          fresh = false;
        } else {
          acc = acc === null ? cur : calc(acc, cur);
          display.textContent = String(+acc.toFixed(10));
          op = c === '=' ? null : c;
          if (c === '=') acc = null;
          fresh = true;
        }
      }),
    );
    // the Look select: data-variant on the desktop - every component inside follows
    root.querySelector('[data-look]').addEventListener('change', (e) => {
      if (e.target.value) root.dataset.variant = e.target.value;
      else delete root.dataset.variant;
    });
  });
</script>
