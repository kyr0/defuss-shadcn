<PageHeader slug="charts-composition" title="Charts · Composition" component="chart">
  Composition patterns on the <DocLink href="chart.html">chart</DocLink> component: small multiples, part-to-whole hierarchies (treemap, sunburst, waffle), and flow diagrams (theme river, sankey, chord). Every demo mounts through <code>df$.shadcn.chart.mount()</code> or declarative <code>data-chart</code> - palette, typography and tooltip chrome come from the active theme's tokens in light and dark.

```html example label="Small multiples" schema="none" height="35" previewStyle="align-items:flex-start;" hint="Repeat the same axes six times and vary only the data - regional shape differences read at a glance, with no legend and no overloaded multi-series chart. Each panel carries its own title, baseline and end value; Metro, the fastest riser, is the only colored line."
<script src="https://cdn.jsdelivr.net/npm/echarts@6.1.0/dist/echarts.min.js"></script>
<figure class="chart-frame" style="margin:0;">
  <figcaption>
    <h3 class="chart-title">Six regions, one scale</h3>
    <p class="chart-dek">Small multiples expose pattern differences that a single overloaded line chart would hide.</p>
  </figcaption>
  <div class="chart" data-size="lg" role="img" aria-label="Small multiples: six regional trend lines on a shared 10 to 60 scale, Metro rising fastest"></div>
  <div class="chart-source">Synthetic data · adapted from echarts-feat 12-small-multiples</div>
</figure>
<script>
window.addEventListener('load', () => {
  const el = document.querySelector('.chart');
  const names = ['North', 'West', 'Central', 'Coast', 'Metro', 'Rural'];
  const vals = [[20, 24, 29, 31, 34, 39], [42, 40, 39, 41, 45, 48], [18, 25, 22, 30, 28, 36], [55, 50, 46, 43, 45, 41], [27, 30, 35, 44, 50, 56], [33, 34, 32, 31, 29, 28]];
  const grid = [], xAxis = [], yAxis = [], series = [], title = [];
  for (let i = 0; i < 6; i++) {
    const col = i % 3, row = Math.floor(i / 3), left = 2 + col * 33, top = 9 + row * 50;
    const hot = names[i] === 'Metro';
    grid.push({ left: left + '%', top: top + '%', width: '26%', height: '32%' });
    xAxis.push({ type: 'category', gridIndex: i, data: ['1', '2', '3', '4', '5', '6'], boundaryGap: false, axisLabel: { show: false } });
    yAxis.push({ type: 'value', gridIndex: i, min: 10, max: 60, interval: 25, axisLabel: { show: col === 0 } });
    title.push({ text: names[i], left: left + '%', top: (top - 8) + '%', textStyle: { fontSize: 13, fontWeight: 600 } });
    series.push({
      type: 'line', xAxisIndex: i, yAxisIndex: i, data: vals[i], smooth: 0.3, showSymbol: false,
      lineStyle: { width: 2.5, color: hot ? 'var(--chart-2)' : 'var(--foreground)' },
      areaStyle: { color: hot ? 'color-mix(in oklch, var(--chart-2) 14%, transparent)' : 'color-mix(in oklch, var(--foreground) 5%, transparent)' },
      endLabel: { show: true, formatter: '{c}', fontWeight: 600, color: hot ? 'var(--chart-2)' : 'var(--muted-foreground)' },
      animationDelay: i * 90,
    });
  }
  df$.shadcn.chart.mount(el, { title, grid, xAxis, yAxis, series });
});
</script>
```

```html example label="Theme river" schema="none" height="27" previewStyle="align-items:flex-start;" hint="A theme river is the right choice when the story is changing prominence among topics rather than precise point comparison. The streams take the theme palette; hovering one dims the rest."

<figure class="chart-frame" style="margin:0;">
  <figcaption>
    <h3 class="chart-title">Streams show how attention shifts</h3>
    <p class="chart-dek">Monthly share of attention across four acquisition channels over 18 months.</p>
  </figcaption>
  <div class="chart" role="img" aria-label="Theme river of attention across search, social, direct and referral channels over 18 months"></div>
  <div class="chart-source">Synthetic data · adapted from echarts-feat 17-theme-river</div>
</figure>
<script>
window.addEventListener('load', () => {
  const el = document.querySelector('.chart');
  const topics = ['Search', 'Social', 'Direct', 'Referral'];
  const data = [];
  for (let m = 0; m < 18; m++) {
    const date = new Date(Date.UTC(2025, m, 1)).toISOString().slice(0, 10);
    topics.forEach((t, i) => data.push([date, Math.max(3, Math.round(20 + 10 * Math.sin(m / (2 + i * 0.25) + i) + i * 3)), t]));
  }
  df$.shadcn.chart.mount(el, {
    legend: { left: 0, data: topics },
    tooltip: { trigger: 'axis', axisPointer: { type: 'line' } },
    singleAxis: { top: 36, bottom: 28, left: 28, right: 28, type: 'time',
      axisLine: { lineStyle: { color: 'color-mix(in oklch, var(--foreground) 28%, transparent)' } },
      axisLabel: { color: 'var(--muted-foreground)' }, splitLine: { show: false } },
    series: [{ type: 'themeRiver', data, label: { show: false }, emphasis: { focus: 'series' },
      itemStyle: { borderColor: 'var(--card)', borderWidth: 1 } }],
  });
});
</script>

```html example label="Waffle dot matrix" schema="none" height="27" previewStyle="align-items:flex-start;" hint="One dot per percent: a square 10×10 grid makes shares countable at a glance. Each category is its own scatter series, so the legend (right) comes for free and the colors are the --chart-* tokens."
<script src="https://cdn.jsdelivr.net/npm/echarts@6.1.0/dist/echarts.min.js"></script>
<figure class="chart-frame" style="margin:0;">
  <figcaption>
    <h3 class="chart-title">Share of effort, one dot per percent</h3>
    <p class="chart-dek">Portfolio split across four investment areas.</p>
  </figcaption>
  <div class="chart" role="img" aria-label="Waffle chart of portfolio share: core 43 percent, growth 27, labs 18, other 12"></div>
  <div class="chart-source">Synthetic data · adapted from echarts-feat 28-waffle-dot-matrix</div>
</figure>
<script>
window.addEventListener('load', () => {
  const el = document.querySelector('.chart');
  const shares = [['Core', 43], ['Growth', 27], ['Labs', 18], ['Other', 12]];
  const colors = ['var(--chart-2)', 'var(--chart-5)', 'var(--chart-1)', 'color-mix(in oklch, var(--foreground) 22%, transparent)'];
  let at = 0;
  const series = shares.map(([name, v], s) => {
    const data = [];
    for (let i = at; i < at + v; i++) data.push([i % 10, Math.floor(i / 10)]);
    at += v;
    return { name: name + ' ' + v + '%', type: 'scatter', data, symbolSize: 18, itemStyle: { color: colors[s] },
      animationDelay: (i) => (data[i][1] * 10 + data[i][0]) * 8 };
  });
  df$.shadcn.chart.mount(el, {
    legend: { orient: 'vertical', left: 312, top: 'middle', icon: 'circle', itemGap: 14, textStyle: { color: 'var(--foreground)', fontWeight: 500 } },
    // a SQUARE plot: fixed width = height, so the dots sit on a true grid
    grid: { left: 16, top: 'middle', width: 264, height: 264 },
    xAxis: { type: 'value', min: -0.5, max: 9.5, show: false },
    yAxis: { type: 'value', min: -0.5, max: 9.5, inverse: true, show: false },
    series,
  });
});
</script>
```

<p class="text-sm text-muted-foreground">Patterns adapted from the ECharts design-language lab (echarts-feat/), re-skinned on the design tokens via the chart theme adapter. Colors are written as <code>var(--token)</code> (or <code>color-mix()</code> over tokens) - the runtime resolves them to sRGB for ECharts and re-resolves them after every theme or dark-mode switch.</p>
