  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-majority');
    const TEXT = '#101319', MUTED = '#6B7280', RED = '#D71920';
    const parties = [['City First', '#1D4ED8'], ['Harbour Greens', '#15803D'], ['Social City', '#DC2626'], ['Liberal Voice', '#EAB308'], ['Neighbourhood lists', '#7C3AED'], ['Others', '#9CA3AF']];
    const early = [27, 14, 15, 8, 4, 3], final = [24, 17, 14, 8, 5, 3];
    const votes = [32.1, 23.4, 19.8, 11.2, 7.4, 6.1];
    const majority = (position) => ({ symbol: 'none', silent: true, lineStyle: { color: RED, width: 5, type: 'solid' },
      label: { formatter: 'MAJORITY 36', position, color: RED, fontWeight: 800 }, data: [{ xAxis: 36 }] });
    const seatBar = (seats) => ({
      legend: { top: 300, left: 112, textStyle: { color: TEXT } },
      grid: { left: 112, right: 112, top: 450, bottom: 300 },
      xAxis: { type: 'value', min: 0, max: 71, show: false },
      yAxis: { type: 'category', data: ['Seats'], show: false },
      series: parties.map(([n, c], i) => ({ id: n, name: n, type: 'bar', stack: 'seats', barWidth: 120,
        data: [seats[i]], itemStyle: { color: c, borderColor: '#F5F6F8', borderWidth: 3, borderRadius: 0 },
        label: { show: seats[i] >= 4, position: 'inside', color: '#FFFFFF', fontWeight: 800, fontSize: 30 },
        markLine: i === 0 ? majority('end') : undefined })),
    });
    const options = [['Greens + Social + Liberal', [0, 17, 14, 8, 0, 0]], ['City First + Social City', [24, 0, 14, 0, 0, 0]], ['City First + Greens', [24, 17, 0, 0, 0, 0]]];

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        early: seatBar(early),
        final: seatBar(final),
        coalitions: {
          legend: { top: 300, left: 112, textStyle: { color: TEXT } },
          grid: { left: 400, right: 190, top: 380, bottom: 170 },
          xAxis: { type: 'value', min: 0, max: 50, axisLabel: { color: MUTED } },
          yAxis: { type: 'category', inverse: true, data: options.map((o) => o[0]), axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 800 } },
          series: parties.map(([n, c], i) => ({ id: n, name: n, type: 'bar', stack: 'seats', barWidth: '58%',
            data: options.map((o) => o[1][i] || '-'), itemStyle: { color: c, borderColor: '#F5F6F8', borderWidth: 3, borderRadius: 0 },
            label: { show: true, position: 'inside', color: '#FFFFFF', fontWeight: 800 },
            markLine: i === 0 ? majority('start') : undefined })),
        },
        votes: {
          legend: { top: 300, right: 112, textStyle: { color: TEXT } },
          grid: { left: 150, right: 112, top: 370, bottom: 150 },
          xAxis: { type: 'category', data: parties.map((p) => p[0]), axisLabel: { color: TEXT, fontWeight: 800, interval: 0, width: 180, overflow: 'break' } },
          yAxis: { type: 'value', max: 40, axisLabel: { color: MUTED, formatter: '{value}%' } },
          series: [
            { id: 'City First', name: '% of votes', type: 'bar', barGap: '10%', data: votes, itemStyle: { color: '#9CA3AF', borderRadius: 0 },
              label: { show: true, position: 'top', formatter: '{c}%', color: MUTED } },
            { id: 'Harbour Greens', name: '% of seats', type: 'bar', data: final.map((s, i) => ({ value: Math.round((s / 71) * 1000) / 10, itemStyle: { color: parties[i][1], borderRadius: 0 } })),
              label: { show: true, position: 'top', formatter: '{c}%', color: TEXT, fontWeight: 800 } },
          ],
        },
      },
    });
  });
