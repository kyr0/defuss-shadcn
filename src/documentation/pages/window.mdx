<PageHeader slug="window" title="Window" component="window">
  A desktop-style window: a title bar with an icon and a title, minimize / maximize / close on the right, dragged by its bar and raised to the front when you press it - several on one page. It is a <strong>non-modal <code>&lt;dialog&gt;</code></strong>, so opening, closing and the <code>close</code> event are the browser's, the × is a <code>&lt;form method="dialog"&gt;</code> button that works without script, and resizing is native CSS. Windows, macOS, Linux and retro chrome; <code>df$.shadcn.win</code> creates and arranges windows from script.

```html example label="Driven from script" schema="none" hint="df$.shadcn.win.create() builds a window (title, icon, content, position, size, chrome) in the first desktop; cascade() and tile() arrange the open ones; close() takes an element, id or selector. <script>
  document.addEventListener('DOMContentLoaded', () => {
    const win = df$.shadcn.win;
    const log = df$('#w-log');
    const desk = df$('#w-desk').get(0);
    const icons = ['file-text', 'image', 'music', 'calendar', 'mail'];
    let n = 0;
    const make = () => {
      n += 1;
      const w = win.create({
        parent: desk, title: `Untitled ${n}`, icon: icons[n % icons.length],
        content: `Window ${n} - created by df$.shadcn.win.create()`, statusbar: `#${n}`, width: 240,
      });
      df$(w).on('close', () => log.text(`closed ${df$(w).find('.window-title').text()}`));
    };
    df$(desk).on('window-focus', (e) => log.text(`front: ${e.detail.title}`));
    df$(desk).on('window-move', (e) => log.text(`moved to ${e.detail.x}, ${e.detail.y}`));
    df$('#w-new').on('click', make);
    df$('#w-cascade').on('click', () => win.cascade(desk));
    df$('#w-tile').on('click', () => win.tile(desk));
    df$('#w-close').on('click', () => { const a = win.active(); if (a) win.close(a); });
    make(); make();
  });
</script>

<script type="module">
  import { codeToTokens } from 'https://esm.sh/shiki@3.0.0';
  // modules run in document order; DOMContentLoaded waits for all of them,
  // so df$ (all.js) is installed by then
  document.addEventListener('DOMContentLoaded', () => {
    const win = df$.shadcn.win;
    const code = df$('#dk-code').get(0);
    const hl = df$('#dk-editor .code-editor-hl').get(0);
    let saved = code.value;

    // the editor: a transparent textarea over a highlighted copy, painted
    // from Shiki's tokens (one coloured span each)
    let pending = 0;
    const paint = async () => {
      const run = ++pending;
      const { tokens } = await codeToTokens(code.value + '\n', { lang: 'html', theme: 'github-dark' });
      if (run !== pending) return; // a newer keystroke already repainted
      const out = [];
      tokens.forEach((line, i) => {
        if (i) out.push('\n');
        line.forEach((tok) => {
          const s = document.createElement('span');
          s.textContent = tok.content;
          if (tok.color) s.style.color = tok.color;
          out.push(s);
        });
      });
      hl.replaceChildren(...out);
      hl.scrollTop = code.scrollTop;
      hl.scrollLeft = code.scrollLeft;
    };
    const preview = () => df$('#dk-frame').prop('srcdoc', code.value);
    const status = () => {
      const before = code.value.slice(0, code.selectionStart).split('\n');
      df$('#dk-pos').text(`Ln ${before.length}, Col ${before[before.length - 1].length + 1}`);
      df$('#dk-saved').text(code.value === saved ? 'Saved' : 'Edited');
    };
    df$(code).on('input', () => { paint(); preview(); status(); });
    df$(code).on('scroll', () => { hl.scrollTop = code.scrollTop; hl.scrollLeft = code.scrollLeft; });
    ['keyup', 'click', 'select'].forEach((type) => df$(code).on(type, status));
    df$(code).on('keydown', (e) => {
      if (e.key !== 'Tab') return;
      e.preventDefault();
      code.setRangeText('  ', code.selectionStart, code.selectionEnd, 'end');
      code.dispatchEvent(new Event('input'));
    });
    paint(); preview();

    // the toolbar - menubar items and tool buttons - drives the windows
    const run = (action) => {
      if (action === 'save') { saved = code.value; status(); }
      else if (action === 'revert') { code.value = saved; code.dispatchEvent(new Event('input')); }
      else if (action === 'close') win.close('dk-editor');
      else if (action === 'select') { code.focus(); code.select(); }
      else if (action === 'copy') navigator.clipboard?.writeText(code.value);
      else if (action === 'preview') win.open('dk-preview');
      else if (action === 'tile') win.tile('#dk');
      else if (action === 'cascade') win.cascade('#dk');
      else if (action === 'maximize') win.toggleMaximize('dk-editor');
      else if (action === 'about') win.open('dk-about');
    };
    df$('#dk-menu').on('dropdown:select', (e) => run(e.detail.value));
    df$('#dk-editor .window-toolbar > .btn').on('click', function () { run(this.dataset.value); });
  });
</script>

  <ul class="text-sm text-muted-foreground m-0 mb-3 ps-5">
