<PageHeader slug="related-item" title="Related Item" component="related-item">
  Links to one related article or resource with a preview - a compact row with a thumbnail, a card, or text with an arrow. Part of the <DocLink href="website-article.html">Article</DocLink> blocks.
</PageHeader>

<SkillPanel component="related-item">
  <SkillField label="Native basis">
<SkillText>An <code>&lt;article&gt;</code>: thumbnail, kind (Article, Guide, Video…), the title link stretched over the item, and meta.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/-webkit-line-clamp" title="two-line titles"><code>line-clamp</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="focus ring on the whole item"><code>:has()</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/translate" title="the arrow nudges on hover"><code>translate</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-related-item</code>
    <code>.mk-related-item-title</code>
    <code>.mk-related-item-media</code>
    <code>.mk-related-item-body</code>
    <code>.mk-related-item-kind</code>
    <code>.mk-related-item-meta</code>
    <code>.mk-related-item-arrow</code>
    <code>.mk-related-item-badge</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant</code> (<code>card</code>, <code>text</code>); <code>data-variant</code> (<code>numbered</code> + <code>data-rank</code>, <code>overlay</code>); <code>.mk-related-item-badge</code> on the thumbnail.</SkillText>
  </SkillField>
</SkillPanel>
