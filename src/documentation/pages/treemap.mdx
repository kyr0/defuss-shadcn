          { itemStyle: { borderColor: INK, borderWidth: 6, gapWidth: 4 } },

          series: [{ id: 'change', type: 'bar', barWidth: '60%',
            data: change.map(([n, v]) => ({ name: n, value: v, itemStyle: { color: color[n], borderColor: INK, borderWidth: 4 }, label: { position: v >= 0 ? 'right' : 'left' } })),

        },
