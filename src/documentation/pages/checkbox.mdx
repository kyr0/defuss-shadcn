```html example label="Indeterminate" hint="The mixed state has no HTML attribute - set it with checkbox.indeterminate = true. Pick a state below; a click on the checkbox clears the mixed state (the browser does that) and the picker follows. For what the state is for - a box that summarises a group - see Select all below."
<div class="flex flex-col gap-4">
  <div class="checkbox-item">
    <input class="checkbox" type="checkbox" id="select-all">
    <label for="select-all">Select all</label>
  </div>
  <fieldset class="radio-group" data-orientation="horizontal" id="select-all-state">
    <legend class="label">State</legend>
    <div class="radio-item">
      <input class="radio" type="radio" name="select-all-state" id="sa-unchecked" value="unchecked">
      <label for="sa-unchecked">Unchecked</label>
    </div>
    <div class="radio-item">
      <input class="radio" type="radio" name="select-all-state" id="sa-checked" value="checked">
      <label for="sa-checked">Checked</label>
    </div>
    <div class="radio-item">
      <input class="radio" type="radio" name="select-all-state" id="sa-indeterminate" value="indeterminate" checked>
      <label for="sa-indeterminate">Indeterminate</label>
    </div>
  </fieldset>
</div>
<script>
  const box = document.getElementById('select-all');
  const picker = document.getElementById('select-all-state');
  const apply = (state) => {
    box.indeterminate = state === 'indeterminate';
    box.checked = state === 'checked';
  };
  const current = () => (box.indeterminate ? 'indeterminate' : box.checked ? 'checked' : 'unchecked');
  apply(picker.querySelector(':checked').value);
  picker.addEventListener('change', (e) => apply(e.target.value));
  box.addEventListener('change', () => { document.getElementById('sa-' + current()).checked = true; });
</script>
```

```html example label="Select all" hint="What the mixed state is for: a parent box above a group. Tick some items and it shows the dash, tick them all and it fills, untick everything and it empties. Clicking it while mixed or empty ticks every item; clicking it while full clears them. The parent names its children with aria-controls, which is also all the small script needs to wire them up."
<fieldset style="border:none;padding:0;margin:0;display:flex;flex-direction:column;gap:0.5rem;">
  <legend style="font-size:0.875rem;font-weight:500;margin-bottom:0.25rem;">Items to display on the desktop:</legend>
  <div class="checkbox-item">
    <input class="checkbox" type="checkbox" id="d-all" aria-controls="d-all-1 d-all-2 d-all-3 d-all-4">
    <label for="d-all">Select all</label>
  </div>
  <div style="display:flex;flex-direction:column;gap:0.5rem;padding-inline-start:1.625rem;">
    <div class="checkbox-item">
      <input class="checkbox" type="checkbox" id="d-all-1" checked>
      <label for="d-all-1">Hard disks</label>
    </div>
    <div class="checkbox-item">
      <input class="checkbox" type="checkbox" id="d-all-2" checked>
      <label for="d-all-2">External disks</label>
    </div>
    <div class="checkbox-item">
      <input class="checkbox" type="checkbox" id="d-all-3">
      <label for="d-all-3">CDs, DVDs, and iPods</label>
    </div>
    <div class="checkbox-item">
      <input class="checkbox" type="checkbox" id="d-all-4">
      <label for="d-all-4">Connected servers</label>
    </div>
  </div>
</fieldset>
<script>
  // a parent box = a .checkbox whose aria-controls lists its children's ids
  document.querySelectorAll('.checkbox[aria-controls]').forEach((parent) => {
    const children = parent.getAttribute('aria-controls').split(' ').map((id) => document.getElementById(id));
    // some ticked = mixed, all = checked, none = empty
    const sync = () => {
      const ticked = children.filter((c) => c.checked).length;
      parent.checked = ticked === children.length;
      parent.indeterminate = ticked > 0 && ticked < children.length;
    };
    // clicking the parent (mixed or empty -> checked, checked -> empty) sets every child
    parent.addEventListener('change', () => {
      children.forEach((c) => { c.checked = parent.checked; });
      sync();
    });
    children.forEach((c) => c.addEventListener('change', sync));
    sync();
  });
</script>
```
