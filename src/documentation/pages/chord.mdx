<script>
  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-chord');
    const TEXT = '#F5F7FF', MUTED = '#8A93C0';
    const regions = [['Europe', '#7CFFEE'], ['North America', '#A78BFA'], ['Asia-Pacific', '#F472B6'], ['Middle East', '#FBBF24'], ['Latin America', '#60A5FA'], ['Africa', '#34D399']];
    const color = Object.fromEntries(regions);
    const links = [['Europe', 'North America', 62], ['Europe', 'Asia-Pacific', 48], ['Europe', 'Middle East', 41], ['Europe', 'Africa', 29], ['Europe', 'Latin America', 18],
      ['North America', 'Asia-Pacific', 44], ['North America', 'Latin America', 51], ['Asia-Pacific', 'Middle East', 36], ['Middle East', 'Africa', 14],
      ['Asia-Pacific', 'Africa', 6], ['North America', 'Middle East', 12], ['Latin America', 'Africa', 2], ['Asia-Pacific', 'Latin America', 5]];
    const total = (r) => links.filter((l) => l[0] === r || l[1] === r).reduce((s, l) => s + l[2], 0);
    const growth = { Africa: 21, 'Middle East': 18, 'Latin America': 11, 'Asia-Pacific': 9, Europe: 4, 'North America': 2 };
    const ring = (focus) => ({
      series: [{ id: 'regions', type: 'chord', center: ['50%', '63%'], radius: ['33%', '38%'], padAngle: 3, clockwise: false,
        itemStyle: { borderRadius: 4 }, label: { show: true, color: TEXT, fontWeight: 650, distance: 16 },
        lineStyle: { color: 'source', opacity: 0.4 },
        data: regions.map(([n, c]) => ({ name: n, itemStyle: { color: c, opacity: !focus || n === focus ? 1 : 0.35 } })),
        links: links.map(([s, t, v]) => ({ source: s, target: t, value: v, lineStyle: { opacity: !focus || s === focus || t === focus ? 0.55 : 0.05 } })) }],
    });
    const bars = (list, value, fmt) => ({
      grid: { left: 290, right: 190, top: 330, bottom: 150 },
      xAxis: { type: 'value', axisLabel: { color: MUTED }, splitLine: { lineStyle: { color: 'rgba(245,247,255,.08)' } } },
      yAxis: { type: 'category', inverse: true, data: list, axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 650 } },
      series: [{ id: 'regions', type: 'bar', barWidth: '58%',
        data: list.map((r) => ({ name: r, value: value(r), itemStyle: { color: color[r], borderRadius: [0, 8, 8, 0], shadowColor: color[r], shadowBlur: 14 } })),
        label: { show: true, position: 'right', formatter: fmt, color: TEXT, fontWeight: 650 } }],
    });
    const byTotal = regions.map((r) => r[0]).sort((a, b) => total(b) - total(a));
    const byGrowth = Object.keys(growth);

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        ring: ring(null),
        europe: ring('Europe'),
        totals: bars(byTotal, total, '{c} M'),
        growth: bars(byGrowth, (r) => growth[r], '+{c}%'),
      },
    });
  });
</script>
