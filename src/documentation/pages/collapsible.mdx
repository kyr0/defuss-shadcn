<PageHeader slug="collapsible" title="Collapsible" component="collapsible">
  An interactive panel that expands/collapses content. Built on native <code>&lt;details&gt;</code> with animated transitions.
</PageHeader>

<SkillPanel component="collapsible">
  <SkillField label="Native basis">
<SkillText><code>&lt;details&gt;</code> element providing native expand/collapse behavior with animated transitions.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/details" title="native disclosure widget"><code>&lt;details&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/summary" title="visible heading/trigger"><code>&lt;summary&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/::details-content" title="pseudo-element for content animation"><code>::details-content</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/@starting-style" title="entry animation starting values"><code>@starting-style</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.collapsible</code>
    <code>.collapsible-trigger</code>
    <code>.collapsible-chevron</code>
    <code>.collapsible-content</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Accessibility">
<SkillText dense>• <code>&lt;details&gt;</code>/<code>&lt;summary&gt;</code> is natively accessible - keyboard and screen reader support built in</SkillText>

  <p class="text-muted-foreground m-0">• Summary text should clearly describe the hidden content</p>
