<PageHeader slug="property-grid" title="Property Grid" component="property-grid">
  The inspector of every RAD tool: a two-column table - key, value - over one JSON object, dense by default. Nested objects and arrays become groups that fold; a click on a value edits it in place with an editor that fits its type (text, number, a switch, a select of options, a color, a date, a textarea), or with the one <code>getEditorFn</code> returns. <code>keyRenderFn</code> and <code>valueRenderFn</code> draw the cells. The JSON object IS the state: every committed edit replaces it in <code>el.store</code> and fires <code>property-grid-change</code>.

<H2 id="render-functions">Render functions</H2>

<DocText><code>keyRenderFn(key, value, ctx)</code> and <code>valueRenderFn(value, key, ctx)</code> are called for every row; each returns text or a node for its cell - or <code>null</code> for the default. <code>ctx</code> carries the <code>path</code>, <code>depth</code>, <code>type</code>, the key's <code>config</code>, the whole <code>source</code> and the grid.</DocText>

```html example label="Badges, units, icons" height="22" hint="Keys get a type glyph, booleans a badge, latencies a unit, dates a relative label - the values stay plain JSON underneath (edit one to see)."
<div class="property-grid" id="pg-render" aria-label="Endpoint health" style="max-width:28rem"></div>
<script>
  document.addEventListener('DOMContentLoaded', () => {
    const grid = document.getElementById('pg-render');
    const GLYPH = { string: 'Aa', number: '#', boolean: '◐', date: '◷', object: '{}' };
    df$.shadcn.propertyGrid.configure(grid, {
      source: { endpoint: '/v2/orders', healthy: true, p95: 182, errors: 0.4, checked: '2026-10-01', owner: { team: 'Payments', oncall: '@mia' } },
      keyRenderFn: (key, value, ctx) => {
        const span = document.createElement('span');
        const glyph = document.createElement('code');
        glyph.className = 'text-muted-foreground';
        glyph.style.cssText = 'font-size:0.7rem;margin-inline-end:0.375rem';
        glyph.textContent = GLYPH[ctx.type] || '·';
        span.append(glyph, key);
        return span;
      },
      valueRenderFn: (value, key) => {
        if (typeof value === 'boolean') {
          const badge = document.createElement('span');
          badge.className = 'badge';
          badge.dataset.variant = value ? 'default' : 'destructive';
          badge.textContent = value ? 'healthy' : 'down';
          return badge;
        }
        if (key === 'p95') return value + ' ms';
        if (key === 'errors') return value + ' %';
        if (key === 'checked') return value + ' · ' + new Intl.RelativeTimeFormat('en', { numeric: 'auto' }).format(Math.round((Date.parse(value) - Date.now()) / 864e5), 'day');
        return null; // the default rendering
      },
    });
  });
</script>

<H2 id="custom-editors">Custom editors</H2>

<DocText><code>getEditorFn(key, value, ctx)</code> picks the editor per key: return an element (an input or a select - its value is read back, a range or number input as a number), an object <code>&#123; el, getValue(), validate?(), focus?(), immediate? &#125;</code> for anything composite, <code>null</code> for the built-in editor or <code>false</code> to make the key read-only. Every committed change fires <code>property-grid-beforechange</code> (cancelable) and <code>property-grid-change</code>.</DocText>

```html example label="A slider, a tag combobox, a guarded value" height="34" hint="opacity edits with a slider; tags with the Combobox tag input (type to filter, Enter or a comma adds, × removes, new tags allowed) - cloned from a template, read back as an array; budget refuses values over 5000 through property-grid-beforechange. Ctrl / Cmd + Enter or a click outside commits the tags; the log lists every change."
<div class="property-grid" id="pg-editors" aria-label="Widget" style="max-width:30rem"></div>
<template id="pg-tags-editor">
  <div class="combobox" data-multiple data-tags data-creatable style="width:100%">
    <div class="combobox-field">
      <input class="combobox-field-input" id="pg-tags-input" type="text" role="combobox" autocomplete="off" aria-label="Tags" aria-expanded="false" aria-controls="pg-tags-list" aria-autocomplete="list" placeholder="Add a tag…">
    </div>
    <div id="pg-tags-pop" class="combobox-content" popover="manual">
      <div id="pg-tags-list" role="listbox" class="combobox-listbox" aria-label="Tags">
        <div class="combobox-empty" hidden>No tag like that.</div>
        <div role="option" id="pg-tag-finance" class="combobox-item" data-value="finance" aria-selected="false">finance</div>
        <div role="option" id="pg-tag-weekly" class="combobox-item" data-value="weekly" aria-selected="false">weekly</div>
        <div role="option" id="pg-tag-board" class="combobox-item" data-value="board" aria-selected="false">board</div>
        <div role="option" id="pg-tag-public" class="combobox-item" data-value="public" aria-selected="false">public</div>
      </div>
    </div>
  </div>
</template>
<p class="text-sm text-muted-foreground mt-2" id="pg-editors-log">No change yet.</p>
<script>
  document.addEventListener('DOMContentLoaded', () => {
    const grid = document.getElementById('pg-editors');
    df$.shadcn.propertyGrid.configure(grid, {
      source: { title: 'Revenue', opacity: 0.8, tags: ['finance', 'weekly'], budget: 1200, id: 'w-42' },
      sourceConfig: { id: { readOnly: true }, tags: { type: 'json' } },
      getEditorFn: (key, value) => {
        if (key === 'opacity') {
          const wrap = document.createElement('div');
          wrap.className = 'flex items-center gap-2 w-full';
          const range = document.createElement('input');
          range.type = 'range';
          range.className = 'slider';
          range.min = '0'; range.max = '1'; range.step = '0.05';
          range.value = String(value);
          const out = document.createElement('output');
          out.className = 'text-xs';
          out.textContent = String(value);
          range.addEventListener('input', () => (out.textContent = range.value));
          wrap.append(range, out);
          return { el: wrap, getValue: () => range.valueAsNumber, focus: () => range.focus() };
        }
        if (key === 'tags') {
          // the Combobox tag input from the template; the current tags are selected
          const el = document.getElementById('pg-tags-editor').content.firstElementChild.cloneNode(true);
          const list = el.querySelector('.combobox-listbox');
          for (const tag of value) {
            let option = list.querySelector('[data-value="' + CSS.escape(tag) + '"]');
            if (!option) {
              option = list.querySelector('.combobox-item').cloneNode(true);
              option.id = 'pg-tag-' + tag.replace(/[^a-z0-9]+/gi, '-');
              option.dataset.value = tag;
              option.textContent = tag;
              list.append(option);
            }
            option.setAttribute('aria-selected', 'true');
          }
          return {
            el,
            ownsEnter: true, // Enter adds a tag; Ctrl / Cmd + Enter or leaving the cell commits
            focus: () => el.querySelector('.combobox-field-input').focus(),
            getValue: () => [...el.querySelectorAll('.combobox-item[aria-selected="true"]')].map((o) => o.dataset.value),
          };
        }
        return null;
      },
    });
    grid.addEventListener('property-grid-beforechange', (e) => {
      if (e.detail.key === 'budget' && e.detail.value > 5000) e.preventDefault();
    });
    grid.addEventListener('property-grid-change', (e) => {
      document.getElementById('pg-editors-log').textContent = e.detail.path + ': ' + JSON.stringify(e.detail.oldValue) + ' → ' + JSON.stringify(e.detail.value);
    });
  });
</script>

<H2 id="validation">Validation</H2>

<DocText>A refused value keeps its editor open, marks it <code>aria-invalid</code> and says why under it (announced). Rules live in <code>sourceConfig</code> - <code>required</code>, <code>min</code> / <code>max</code>, <code>integer</code>, <code>minLength</code> / <code>maxLength</code>, <code>pattern</code>, <code>minItems</code> / <code>maxItems</code>, with an optional <code>message</code> - and <code>validateFn(key, value, ctx)</code> checks anything else, across keys too (<code>ctx.source</code>).</DocText>

```html example label="Rules and a cross-field check" height="30" hint="Try port 80 (below 1024), replicas 2.5 (not whole), an empty name, an e-mail without @, or a budget above the limit (validateFn reads ctx.source.limit). The message shows under the editor until the value is valid or you press Escape."

  </script>
  <script type="application/json" class="property-grid-config">
    {
      "name": { "required": true, "minLength": 3, "maxLength": 24, "pattern": "[a-z][a-z0-9-]*", "message": "3-24 lowercase letters, digits or -" },
      "port": { "min": 1024, "max": 65535, "integer": true },
      "replicas": { "min": 1, "max": 12, "integer": true },

<script>
  document.addEventListener('DOMContentLoaded', () => {
    const pg = df$.shadcn.propertyGrid;
    const grid = document.getElementById('pg-data');
    const json = document.getElementById('pg-data-json');
    const initial = pg.getSource(grid);
    let before = [];
    // the JSON view follows the store; lines that differ from the last view are marked
    const show = (source) => {
      const lines = JSON.stringify(source, null, 2).split('\n');
      json.replaceChildren(...lines.map((line, i) => {
        const pre = document.createElement('pre');
        if (before.length && before[i] !== line) pre.dataset.diff = 'add';
        const code = document.createElement('code');
        code.textContent = line;
        pre.append(code);
        return pre;
      }));
      before = lines;
    };
    show(pg.getSource(grid));
    grid.store.subscribe(({ config }) => show(config.source));
    const out = document.getElementById('pg-data-out');
    document.getElementById('pg-data-read').addEventListener('click', () => (out.textContent = 'replicas = ' + pg.getProperty(grid, 'replicas')));
    document.getElementById('pg-data-scale').addEventListener('click', () => pg.setProperty(grid, 'replicas', 6));
    document.getElementById('pg-data-flip').addEventListener('click', () => pg.setProperty(grid, 'public', !pg.getProperty(grid, 'public')));
    document.getElementById('pg-data-reset').addEventListener('click', () => pg.setSource(grid, initial));
    document.getElementById('pg-data-apply').addEventListener('click', () => {
      const error = document.getElementById('pg-data-error');
      try {
        pg.setSource(grid, JSON.parse(document.getElementById('pg-data-text').value));
        error.textContent = '';
      } catch (e) {
        error.textContent = e.message;
      }
    });
  });
</script>

      const el = e.detail.element;
      if (!el) {
        title.textContent = 'Inspector';

        title.textContent = 'Inspector · ' + e.detail.label;

      if (!selected) return;
      const el = selected.element;
      const { key, value } = e.detail;
      if (selected.kind === 'node') {
        if (key === 'name') el.querySelector('.diagram-node-name').textContent = value;
        if (key === 'meta') el.querySelector('.diagram-node-meta').textContent = value;
        if (key === 'tone') value === 'none' ? el.removeAttribute('data-tone') : el.setAttribute('data-tone', value);
        if (key === 'shape') value === 'box' ? el.removeAttribute('data-shape') : el.setAttribute('data-shape', value);
        if (key === 'name') title.textContent = 'Inspector · ' + value;
      } else {
        if (key === 'label') el.textContent = value;
        if (key === 'line') value === 'solid' ? el.removeAttribute('data-line') : el.setAttribute('data-line', value);
      }
      df$.shadcn.diagram.redraw(fig);
    });
  });
</script>

<StatesSection>
  <p class="text-sm text-muted-foreground m-0 mb-3">Named states via the shared State API, bound on each grid:</p>
