<script>
  document.addEventListener('DOMContentLoaded', () => {
    const chat = df$('#chat').get(0);
    const session = df$.shadcn.session;
    const bubbleRow = (me, inner) => `<div class="message"${me ? ' data-align="end"' : ''}><div class="message-avatar"><span class="avatar" data-size="sm"><img class="avatar-image" src="https://github.com/${me ? 'kyr0' : 'evilrabbit'}.png" alt="${me ? 'You' : 'Sofia Davis'}"><span class="avatar-fallback">${me ? 'AH' : 'SD'}</span></span></div><div class="message-content">${inner}</div></div>`;
    const form = df$('#chat-composer').get(0);
    const text = df$(form).find('.textarea').get(0);
    const list = df$(form).find('.file-drop-list');
    const picker = df$(form).find('[data-files]').get(0);
    const files = []; // { id, file, url }
    const esc = (s) => s.replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' })[c]);

    // send is live only when there is something to send
    const sync = () => df$(form).find('[data-send]').prop('disabled', !text.value.trim() && !files.length);
    // attachment chips: the file-input's own .file-drop-item markup, keyed by id
    const render = () => list.morph(files.map((a) => `
      <li class="file-drop-item" id="${a.id}">
        <span class="file-drop-thumb" aria-hidden="true"><img src="${a.url}" alt=""></span>
        <div class="file-drop-name">${esc(a.file.name)}<span class="file-drop-meta">${Math.ceil(a.file.size / 1024)} KB</span></div>
        <button type="button" class="file-drop-remove" data-remove="${a.id}" aria-label="Remove ${esc(a.file.name)}">✕</button>
      </li>`).join(''));
    const add = (fileList) => {
      for (const file of fileList) {
        if (!file.type.startsWith('image/')) continue; // images only
        files.push({ id: `att-${crypto.randomUUID()}`, file, url: URL.createObjectURL(file) });
      }
      render(); sync();
    };

    df$(text).on('input', sync);
    // Enter sends, Shift+Enter is a new line (never mid-IME composition)
    df$(text).on('keydown', (e) => {
      if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) { e.preventDefault(); form.requestSubmit(); }
    });
    // + opens the native picker
    df$(form).find('[data-attach]').on('click', () => picker.click());
    df$(picker).on('change', () => { add(picker.files); picker.value = ''; });
    df$(list).on('click', (e) => {
      const id = e.target.closest('[data-remove]')?.dataset.remove;
      const i = files.findIndex((a) => a.id === id);
      if (i < 0) return;
      URL.revokeObjectURL(files[i].url);
      files.splice(i, 1); render(); sync();
    });
    // files dropped on the whole message window join the composer (data-drop)
    df$(chat).on('session-drop', (e) => add(e.detail.files));

    df$(form).on('submit', (e) => {
      e.preventDefault();
      if (!text.value.trim() && !files.length) return;
      // your message - text and images - at the end of the thread
      const images = files.map((a) => `<div class="bubble"><div class="bubble-content" data-media style="max-width:14rem;"><img src="${a.url}" alt="${esc(a.file.name)}"></div></div>`).join('');
      const words = text.value.trim() ? `<div class="bubble"><div class="bubble-content">${esc(text.value.trim()).replace(/\n/g, '<br>')}</div></div>` : '';
      session.scrollToEnd(chat);
      session.append(chat, bubbleRow(true, images + words), { id: `me-${Date.now()}` });
      // Sofia types, then answers
      const typing = session.append(chat, bubbleRow(false, '<div class="bubble" data-variant="secondary" data-tail><div class="bubble-content" role="status" aria-label="Sofia is typing"><span class="bubble-typing" aria-hidden="true"><span></span><span></span><span></span></span></div></div>'));
      setTimeout(() => {
        df$(typing).remove();
        session.append(chat, bubbleRow(false, `<div class="bubble" data-variant="secondary"><div class="bubble-content">${images ? 'Nice picture! 📸' : 'Got it 👍'}</div></div>`), { id: `sofia-${Date.now()}` });
      }, 1400);
      files.splice(0); // the sent message keeps the image URLs
      df$(text).val('');
      render(); sync();
      text.focus();
    });
  });
</script>

<StatesSection>
  <p class="text-sm text-muted-foreground m-0 mb-3">Named states via the shared State API, driven per session through the bound <code>api</code>:</p>
  <ul class="text-sm text-muted-foreground m-0 mb-3 ps-5">
    <li><code>default</code> - following the live edge: new messages scroll into view</li>
    <li><code>detached</code> - not following: the reader scrolled away or a turn anchored; <code>\{ to: 'start' | messageId \}</code> scrolls there</li>
    <li><code>streaming</code> - a reply is being written: <code>aria-busy</code> on the log, the transcript follows it at the end</li>
  </ul>
  <p class="text-sm text-muted-foreground">The first example carries <code>data-state-demo</code>; <code>bun run screenshots</code> drives it via <code>el.api.setState(name)</code> and captures <code>screenshots/\{mode\}/session-\{state\}.png</code>.</p>

Machine contract - verified against <a href="../schemas/session.schema.json"><code>session.schema.json</code></a> by <code>bun run verify</code>:

```states
| State | Type | Values | Default | Description |
| --- | --- | --- | --- | --- |
| `detached` | `boolean` | `true`, `false` | `false` | Not following the live edge - the reader scrolled away or a turn anchored. |
| `streaming` | `boolean` | `true`, `false` | `false` | A reply is being written: `aria-busy` on the log, following at the end. |
```
