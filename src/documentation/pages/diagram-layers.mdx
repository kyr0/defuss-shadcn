<H2 id="interactive">Interactive</H2>

Keyboard: Tab to a box, Enter, arrows, Escape.</DocText>

<H2 id="inspect">Inspect and edit</H2>

<DocText>Select, inspect, edit: a click hands the selection to a <a class="link" href="property-grid.html">Property Grid</a> in a closable <a class="link" href="panel.html">Panel</a> - <code>df$.shadcn.diagram.properties()</code> reads the box or the arrow as JSON, <code>propertySchema()</code> gives the grid its options, <code>setProperties()</code> writes an edit back and the diagram redraws. The × closes the panel, the Inspector button toggles it.</DocText>

<script>
  document.addEventListener('DOMContentLoaded', () => {
    const { diagram, propertyGrid } = df$.shadcn;
    const fig = document.getElementById('dg-layers-edit');
    const grid = document.getElementById('dg-layers-edit-grid');
    const title = document.getElementById('dg-layers-edit-title');
    let ref = null;
    // a selection: its properties (and their options) into the grid
    fig.addEventListener('diagram-activate', (e) => {
      ref = e.detail.ref;
      title.textContent = ref ? 'Inspector · ' + e.detail.label : 'Inspector';
      propertyGrid.configure(grid, { sourceConfig: ref ? diagram.propertySchema(fig, ref) : {} });
      propertyGrid.setSource(grid, ref ? diagram.properties(fig, ref) : {});
    });
    // an edit: back into the diagram, which redraws
    grid.addEventListener('property-grid-change', (e) => {
      if (ref) diagram.setProperties(fig, ref, e.detail.source);
    });
  });
</script>
