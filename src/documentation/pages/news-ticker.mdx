<PageHeader slug="news-ticker" title="News Ticker" component="news-ticker">
  Displays a compact stream of linked news headlines - a seamless CSS marquee that pauses on hover and focus, and stops for reduced motion. Part of the <DocLink href="website-news.html">News</DocLink> blocks.
</PageHeader>

<SkillPanel component="news-ticker">
  <SkillField label="Native basis">
<SkillText>A labelled <code>&lt;section&gt;</code> with one list of headline links - repeated once in the same track, <code>inert</code> and <code>aria-hidden</code>, so the loop is seamless while assistive tech and the tab order see each headline once. A CSS animation slides the track; hover and focus pause it.</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="badge.html">Badge</DocLink>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/animation" title="the marquee"><code>CSS animations</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Global_attributes/inert" title="the loop copy leaves tab order and a11y tree"><code>inert</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:focus-within" title="keyboard focus pauses"><code>:focus-within</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/mask-image" title="soft edges"><code>mask-image</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/@media/prefers-reduced-motion" title="stops, scrolls by hand"><code>prefers-reduced-motion</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-news-ticker</code>

    <code>.mk-news-ticker-track</code>
    <code>.mk-news-ticker-items</code>
    <code>.mk-news-ticker-trend</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant</code> (<code>breaking</code>, <code>static</code>), <code>data-size="sm"</code>; <code>--mk-news-ticker-duration</code> (default 40s); <code>data-variant</code> (<code>market</code>, <code>inverse</code>, <code>pill</code>); <code>.mk-news-ticker-trend[data-trend="up|down"]</code>.</SkillText>
  </SkillField>
</SkillPanel>
