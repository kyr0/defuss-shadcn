<PageHeader slug="text-rotate" title="Text Rotate" component="text-rotate">
  Shows one line at a time from a short list and rolls on to the next - an infinite loop that pauses while hovered. Put it inline in a sentence or use it as a headline; it takes the font, size, color and line height of the text around it. CSS only, the same markup as daisyUI's text-rotate.
</PageHeader>

<SkillPanel component="text-rotate">
  <SkillField label="Native basis">
<SkillText>A one-line window (<code>height: 1lh; overflow: hidden</code>) around a stack of 2 - 6 lines; a CSS animation picked by the line count (<code>:has()</code>) moves the stack.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/length#lh" title="a window exactly one line tall"><code>lh</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="picks the keyframes by line count"><code>:has()</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/animation-play-state" title="pauses on hover / focus"><code>animation-play-state</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/@media/prefers-reduced-motion" title="the first line stays"><code>prefers-reduced-motion</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.text-rotate</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-direction="down"</code>, <code>data-effect="fade"</code>, <code>data-align</code> (<code>center</code>, <code>end</code>), <code>data-pause="none"</code>; the duration of one round: <code>--text-rotate-duration</code> (default 10s).</SkillText>
  </SkillField>
</SkillPanel>
