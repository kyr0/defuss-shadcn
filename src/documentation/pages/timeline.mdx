<PageHeader slug="timeline" title="Timeline" component="timeline">
  A vertical timeline for displaying sequential events. Built on <code>&lt;ol&gt;</code> with CSS-generated connector lines.
</PageHeader>

<SkillPanel component="timeline">
  <SkillField label="Native basis">
<SkillText><code>&lt;ol&gt;</code> element with timeline items connected by a line - vertical by default, horizontal with <code>data-orientation="horizontal"</code>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/ol" title="ordered list conveying sequence"><code>&lt;ol&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/::before" title="pseudo-element for connector line and dot"><code>::before</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_grid_layout/Subgrid" title="horizontal: every dot on one line"><code>subgrid</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/scroll-snap-type" title="horizontal: sideways scrolling snaps to items"><code>scroll-snap</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="icon dots re-center the connector"><code>:has()</code></ApiPill>
  </SkillRow>
  </SkillField>
