<PageHeader slug="quick-links" title="Quick Links" component="quick-links">
  Groups shortcuts to frequently needed destinations - icon tiles with a line each, a compact list, or chips. Part of the <DocLink href="website-search-navigation.html">Search & Navigation</DocLink> blocks.
</PageHeader>

<SkillPanel component="quick-links">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;nav aria-label="Quick links"&gt;</code> with a list of links: an icon, a title, a short description and an arrow that slides on hover.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/nav" title="a navigation landmark"><code>&lt;nav&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_containment/Container_queries" title="the layout follows the block width, not the viewport"><code>Container queries</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-quick-links</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant</code>: <code>list</code>, <code>chips</code>.</SkillText>
  </SkillField>
</SkillPanel>
