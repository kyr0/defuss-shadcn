<script>
  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-highlight');
    const INK = '#111111', MUTED = '#8A857B', RED = '#E6392F', GREY = '#CFCAC0';
    const G = { left: 250, right: 150, top: 320, bottom: 140 };
    // [mode, growth %, trips per weekday 2026 (thousands)]
    const modes = [['E-bike', 142, 97], ['Cargo bike', 88, 18], ['Tram', 34, 236], ['Walking', 21, 188], ['Bus', 12, 141], ['Car', -9, 412], ['Taxi & ride-hail', -17, 31]];
    const byVolume = modes.slice().sort((a, b) => b[2] - a[2]);

    const ranking = (rows, value, color, fmt, xAxis) => ({
      grid: G,
      xAxis,
      yAxis: { type: 'category', inverse: true, data: rows.map((m) => m[0]), axisLine: { show: false }, axisLabel: { color: INK, fontWeight: 700 } },
      series: [{ id: 'modes', type: 'bar', barWidth: '62%',
        data: rows.map((m) => ({ name: m[0], value: m[value], itemStyle: { color: color(m) } })),
        label: { show: true, position: 'right', formatter: fmt, color: INK, fontWeight: 700 } }],
    });
    const signed = (p) => (p.value > 0 ? '+' : '−') + Math.abs(p.value) + '%';
    const growthAxis = { type: 'value', min: -40, max: 160, interval: 40, axisLabel: { color: MUTED, formatter: (v) => v + '%' } };

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        growth: ranking(modes, 1, (m) => (m[0] === 'E-bike' ? RED : GREY), signed, growthAxis),
        decline: ranking(modes, 1, (m) => (m[1] < 0 ? RED : GREY), signed, growthAxis),
        volume: ranking(byVolume, 2, (m) => (m[0] === 'E-bike' ? RED : GREY), (p) => p.value + 'k',
          { type: 'value', max: 450, interval: 150, axisLabel: { color: MUTED, formatter: (v) => v + 'k' } }),
        share: {
          series: [{ id: 'modes', type: 'pie', radius: ['30%', '52%'], center: ['50%', '64%'], startAngle: 90,
            data: byVolume.map((m) => ({ name: m[0], value: m[2], itemStyle: { color: m[0] === 'E-bike' ? RED : m[0] === 'Car' ? INK : GREY } })),
            itemStyle: { borderColor: '#F4F1E8', borderWidth: 4 },
            label: { formatter: (p) => p.name + '  ' + Math.round(p.percent) + '%', color: INK, fontWeight: 700 },
            labelLine: { length: 24, length2: 30, lineStyle: { color: MUTED } } }],
        },
      },
    });
  });
</script>
