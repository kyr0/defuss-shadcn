<PageHeader slug="view-switcher" title="View Switcher" component="view-switcher">
  Switches a collection between layouts such as grid and list - icon or labelled segments, and the collection beside them changes with no script. Part of the <DocLink href="website-search-navigation.html">Search & Navigation</DocLink> blocks.
</PageHeader>

<SkillPanel component="view-switcher">
  <SkillField label="Native basis">
<SkillText>A wrapper (<code>.mk-view-switcher</code>) holding a radio group (<code>.mk-view-switcher-options</code>, values <code>grid</code> / <code>list</code>) and the collection (<code>.mk-view-switcher-collection</code>); <code>:has(input[value="list"]:checked)</code> turns the grid into a list.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="parent styles react to a checked input or a state inside"><code>:has()</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/fieldset" title="grouped controls with a legend"><code>&lt;fieldset&gt;</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-view-switcher</code>
    <code>.mk-view-switcher-options</code>
    <code>.mk-view-switcher-collection</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant="labels"</code>; radio values <code>grid</code> / <code>list</code>.</SkillText>
  </SkillField>
</SkillPanel>
