<PageHeader slug="charts-distribution" title="Distribution Charts" component="chart">
  Statistical patterns for showing <strong>spread and density</strong> rather than single values: confidence bands, quadrant scatter, jittered strips, matrix and calendar heatmaps, boxplots, and a custom-series violin. Every demo mounts through the <DocLink href="chart.html">chart component</DocLink>, so palette, typography and chrome come from the active theme's tokens in light and dark.

```html example label="Jittered strip" schema="none" hint="Every observation stays visible - deterministic jitter spreads the points of each group instead of collapsing them into a bar, and one subgroup is highlighted with var(--chart-2) while the rest recede." previewStyle="align-items:flex-start;" height="35"
<script src="https://cdn.jsdelivr.net/npm/echarts@6.1.0/dist/echarts.min.js"></script>
<figure class="chart-frame" style="margin:0;">
  <h3 class="chart-title">Score distribution by group</h3>
  <p class="chart-dek">168 individual scores across four groups; group C is highlighted, the rest fall back to the muted ink.</p>
  <div id="strip-chart" class="chart" data-size="lg" role="img" aria-label="Jittered strip plot of 168 scores across four groups, group C highlighted"></div>
  <figcaption class="chart-source">Synthetic data · adapted from echarts-feat 14-jittered-distribution</figcaption>
</figure>
<script>
  window.addEventListener('load', () => {
    const groups = ['Group A', 'Group B', 'Group C', 'Group D'];
    const data = [];
    for (let g = 0; g < 4; g++) {
      for (let i = 0; i < 42; i++) {
        const base = 35 + g * 9;
        const v = Math.max(5, Math.min(95, base + Math.sin(i * 1.7 + g) * 15 + (i % 7) * 1.1));
        const jitter = (((i * 37) % 17) - 8) / 24;
        data.push({ value: [+v.toFixed(1), g + jitter],
          itemStyle: g === 2 ? { color: 'var(--chart-2)', opacity: 0.85 } : { color: 'var(--muted-foreground)', opacity: 0.35 } });
      }
    }
    df$.shadcn.chart.mount(document.getElementById('strip-chart'), {
      grid: { top: 16 },
      xAxis: { type: 'value', min: 0, max: 100, axisLabel: { formatter: '{value}%' } },
      // ticks sit exactly on the group rows (customValues), not on min + n·interval
      yAxis: { type: 'value', min: -0.6, max: 3.6, inverse: true, splitLine: { show: false },
        axisLabel: { customValues: [0, 1, 2, 3], formatter: (v) => groups[v], color: 'var(--foreground)', fontWeight: 500 } },
      series: [{ type: 'scatter', data, symbolSize: 9, emphasis: { scale: 1.6, itemStyle: { opacity: 1 } } }],
    });
  });
</script>
```

  <p class="chart-dek">Six survey criteria compared pairwise; the diagonal is the self-correlation, labels keep the exact values readable.</p>
  <div id="matrix-chart" class="chart" data-size="lg" role="img" aria-label="Heatmap matrix of pairwise correlations between six criteria, from minus one to plus one"></div>
  <figcaption class="chart-source">Synthetic data · adapted from echarts-feat 15-heatmap-matrix</figcaption>

<script>
  window.addEventListener('load', () => {
    const cats = ['Speed', 'Quality', 'Cost', 'Trust', 'Reach', 'Ease'];
    const data = [];
    for (let y = 0; y < 6; y++) {
      for (let x = 0; x < 6; x++) {
        const v = x === y ? 1 : Math.max(-1, Math.min(1, Math.sin((x + 1) * (y + 2)) * 0.72));

      }

    df$.shadcn.chart.mount(document.getElementById('matrix-chart'), {
      grid: { top: 36 },
      xAxis: { type: 'category', data: cats, axisLine: { show: false } },

        inRange: { color: ['var(--destructive)', 'var(--muted)', 'var(--chart-2)'] }, text: ['+1', '−1'] },
      series: [{
        type: 'heatmap', data,
        label: { show: true, formatter: (p) => p.value[2].toFixed(2), color: 'var(--foreground)', fontWeight: 600 },
        itemStyle: { borderColor: 'var(--card)', borderWidth: 3, borderRadius: 6 },
        emphasis: { itemStyle: { borderColor: 'var(--foreground)', borderWidth: 2 } },

```html example label="Violin (custom series)" schema="none" hint="ECharts has no violin type - a custom renderItem series draws each density polygon via api.coord. The colors are resolved INSIDE renderItem with df$.shadcn.chart.color(), so a theme switch (which re-renders) repaints them too." previewStyle="align-items:flex-start;" height="27"
<script src="https://cdn.jsdelivr.net/npm/echarts@6.1.0/dist/echarts.min.js"></script>
<figure class="chart-frame" style="margin:0;">
  <h3 class="chart-title">Density shape by group</h3>
  <p class="chart-dek">Full distribution silhouettes for four groups - where a boxplot compresses, a violin shows the modes. Group B is highlighted.</p>
  <div id="violin-chart" class="chart" role="img" aria-label="Violin plot of four distributions A to D, each drawn as a custom density polygon"></div>
  <figcaption class="chart-source">Synthetic data · adapted from echarts-feat 30-violin-custom</figcaption>
</figure>
<script>
  window.addEventListener('load', () => {
    const el = document.getElementById('violin-chart');
    const cats = ['Group A', 'Group B', 'Group C', 'Group D'];
    const data = [];
    for (let c = 0; c < 4; c++) {
      const pts = [];
      for (let y = 10; y <= 90; y += 2) {
        const center = 42 + c * 7;
        const spread = 13 + c * 2;
        const dens = Math.exp(-Math.pow(y - center, 2) / (2 * spread * spread)) * (0.65 + 0.35 * Math.cos(y * 0.22 + c));
        pts.push([y, Math.max(0, dens)]);
      }
      data.push(pts);
    }
    df$.shadcn.chart.mount(el, {
      grid: { top: 16 },
      xAxis: { type: 'category', data: cats, axisLabel: { color: 'var(--foreground)', fontWeight: 500 } },
      yAxis: { type: 'value', min: 0, max: 100 },
      series: [{
        type: 'custom',
        data: cats.map((_, i) => [i]),
        renderItem: (p, api) => {
          const pts = data[p.dataIndex];
          const cx = api.coord([p.dataIndex, 0])[0];
          const maxW = api.size([1, 0])[0] * 0.42;
          const left = pts.map(([y, d]) => [cx - d * maxW, api.coord([p.dataIndex, y])[1]]);
          const right = pts.slice().reverse().map(([y, d]) => [cx + d * maxW, api.coord([p.dataIndex, y])[1]]);
          const hot = p.dataIndex === 1;
          const color = df$.shadcn.chart.color(el, hot ? '--chart-2' : '--muted-foreground');
          return {
            type: 'polygon',
            shape: { points: left.concat(right), smooth: 0.3 },
            style: { fill: df$.shadcn.chart.color(el, color, hot ? 0.35 : 0.16), stroke: color, lineWidth: 1.5 },
          };
        },
      }],
    });
  });
</script>
```

<p class="text-sm text-muted-foreground">Patterns adapted from the ECharts design-language lab (echarts-feat/), re-skinned on the design tokens via the chart theme adapter. Colors are written as <code>var(--token)</code> (or <code>color-mix()</code> over tokens) - the runtime resolves them to sRGB for ECharts and re-resolves them after every theme or dark-mode switch.</p>
