<PageHeader slug="search-suggestions" title="Search Suggestions" component="search-suggestions">
  Displays suggested queries or matches while typing - recent searches, matching pages and people grouped in a panel with the match highlighted, or native suggestions. Part of the <DocLink href="website-search-navigation.html">Search & Navigation</DocLink> blocks.
</PageHeader>

<SkillPanel component="search-suggestions">
  <SkillField label="Native basis">
<SkillText>An input with <code>role="combobox"</code>, <code>aria-expanded</code> and <code>aria-controls</code>, and a <code>role="listbox"</code> panel with groups (<code>role="group"</code> + label) of <code>role="option"</code> rows; the matched text is a <code>&lt;mark&gt;</code>. Your script filters and moves <code>aria-activedescendant</code>; the <code>datalist</code> variant needs none.</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="input.html">Input</DocLink>, <DocLink href="kbd.html">Kbd</DocLink>.</SkillText>
  </SkillField>
