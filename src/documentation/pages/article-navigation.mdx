<PageHeader slug="article-navigation" title="Article Navigation" component="article-navigation">
  Links to the previous and next articles - rel="prev" / rel="next" cards with direction and title, or a minimal text row. Part of the <DocLink href="website-article.html">Article</DocLink> blocks.
</PageHeader>

<SkillPanel component="article-navigation">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;nav&gt;</code> with two links - <code>rel="prev"</code> and <code>rel="next"</code> - each a direction label and the title. A lone next link keeps the end side; narrow, they stack.</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="progress.html">Progress</DocLink>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Attributes/rel" title="machine-readable sequence"><code>rel="prev" / "next"</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_containment/Container_queries" title="stack when narrow"><code>Container queries</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:only-child" title="a lone next link keeps its side"><code>:only-child</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-article-nav</code>
    <code>.mk-article-nav-link</code>
    <code>.mk-article-nav-dir</code>
    <code>.mk-article-nav-title</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant="minimal"</code>; <code>rel="prev"</code> / <code>rel="next"</code> on the links; <code>data-variant</code> (<code>media</code>, <code>compact</code>).</SkillText>
  </SkillField>
</SkillPanel>
