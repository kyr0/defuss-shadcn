<p class="text-sm font-medium mb-2 mt-8">Align</p>

<p class="text-muted-foreground text-sm mb-4">Use <code>data-align</code> on the popover to control horizontal alignment: <code>start</code>, <code>center</code> (default), or <code>end</code>.</p>

<Example noCode previewStyle="display:flex;gap:0.5rem;justify-content:center;">
  <button class="btn" data-variant="outline" popovertarget="demo-pop-start">Start</button>
  <div class="popover" id="demo-pop-start" popover data-align="start">
    <div class="popover-header">
      <p class="popover-title">Start aligned</p>
      <p class="popover-description">Popover aligned to the start edge.</p>

  <ul class="text-sm text-muted-foreground" style="margin:0 0 0.75rem;padding-left:1.25rem;">
