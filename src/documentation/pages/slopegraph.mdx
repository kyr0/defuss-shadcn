<script>
  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-slope');
    const TEXT = '#C9FFD8', MUTED = '#5E8B6B', UP = '#43FF76', DOWN = '#FF5F56', DIM = 'rgba(201,255,216,.16)';
    // [language, 2020 %, 2025 %]
    const langs = [['Python', 21, 29], ['TypeScript', 9, 19], ['Java', 24, 17], ['C#', 13, 12], ['Go', 4, 8], ['PHP', 11, 6], ['Rust', 1, 5]];
    const up = (l) => l[2] >= l[1];

    const slopes = (focusRisers) => ({
      grid: { left: 330, right: 330, top: 330, bottom: 140 },
      xAxis: { type: 'category', data: ['2020', '2025'], boundaryGap: false, axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 700 } },
      yAxis: { type: 'value', min: 0, max: 30, show: false },
      series: langs.map((l) => {
        const color = focusRisers && !up(l) ? DIM : up(l) ? UP : DOWN;
        return {
          id: l[0], name: l[0], type: 'line', data: [l[1], l[2]], symbolSize: 16,
          lineStyle: { color, width: 4 }, itemStyle: { color },
          label: { show: true, position: 'left', distance: 16, formatter: (p) => (p.dataIndex === 0 ? l[0] + '  ' + l[1] + '%' : ''), color: focusRisers && !up(l) ? MUTED : TEXT },
          endLabel: { show: true, distance: 16, formatter: l[0] + '  ' + l[2] + '%', color: focusRisers && !up(l) ? MUTED : TEXT },
          labelLayout: { moveOverlap: 'shiftY' },
        };
      }),
    });
    const byDelta = langs.slice().sort((a, b) => (b[2] - b[1]) - (a[2] - a[1]));
    const by2025 = langs.slice().sort((a, b) => b[2] - a[2]);
    const cats = (list) => ({ type: 'category', inverse: true, data: list.map((l) => l[0]), axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 700 } });

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        slope: slopes(false),
        risers: slopes(true),
        delta: {
          grid: { left: 260, right: 190, top: 320, bottom: 150 },
          xAxis: { type: 'value', min: -10, max: 12, axisLabel: { color: MUTED, formatter: (v) => (v > 0 ? '+' : '') + v } },
          yAxis: cats(byDelta),
          series: [{ id: 'bars', type: 'bar', barWidth: '56%',
            data: byDelta.map((l) => ({ name: l[0], value: l[2] - l[1], itemStyle: { color: up(l) ? UP : DOWN } })),
            label: { show: true, position: 'right', formatter: (p) => (p.value > 0 ? '+' : '') + p.value + ' pp', color: TEXT, fontWeight: 700 } }],
        },
        rank: {
          grid: { left: 260, right: 190, top: 320, bottom: 150 },
          xAxis: { type: 'value', max: 32, axisLabel: { color: MUTED, formatter: '{value}%' } },
          yAxis: cats(by2025),
          series: [{ id: 'bars', type: 'bar', barWidth: '56%',
            data: by2025.map((l) => ({ name: l[0], value: l[2], itemStyle: { color: l[0] === 'Python' || l[0] === 'TypeScript' ? UP : 'rgba(67,255,118,.35)' } })),
            label: { show: true, position: 'right', formatter: '{c}%', color: TEXT, fontWeight: 700 } }],
        },
      },
    });
  });
</script>
