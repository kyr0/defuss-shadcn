  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-morph');
    const TEXT = '#FFFFFF', MUTED = '#8A8A8A', ORANGE = '#FF3E00';
    // [channel, tickets (k), average price €, revenue € M, color]
    const ch = [['App', 46, 161, 7.4, ORANGE], ['Web', 38, 166, 6.3, '#FFFFFF'], ['Box office', 17, 129, 2.2, '#3977FF'], ['Resellers', 12, 217, 2.6, '#00D8A7'], ['Partners', 7, 129, 0.9, '#FFE100']];
    // identity contract: one series id, the channel NAME on every data item
    const item = (c, value) => ({ name: c[0], value, itemStyle: { color: c[4] } });
    const bigLabel = { color: TEXT, fontWeight: 900, fontFamily: "'Arial Black', Impact, sans-serif" };
    const G = { left: 150, right: 150, top: 330, bottom: 150 };

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        bars: {
          grid: G,
          xAxis: { type: 'category', data: ch.map((c) => c[0]), axisLabel: { ...bigLabel, fontSize: 26 } },
          yAxis: { type: 'value', max: 50, axisLabel: { color: MUTED, formatter: '{value}k' }, splitLine: { lineStyle: { color: 'rgba(255,255,255,.08)' } } },
          series: [{ id: 'tickets', type: 'bar', barWidth: '54%', data: ch.map((c) => item(c, c[1])), itemStyle: { borderRadius: [10, 10, 0, 0] },
            label: { show: true, position: 'top', formatter: '{c}k', ...bigLabel } }],
        },
        donut: {
          series: [{ id: 'tickets', type: 'pie', radius: ['26%', '48%'], center: ['50%', '64%'], data: ch.map((c) => item(c, c[1])),
            itemStyle: { borderColor: '#111111', borderWidth: 6 },
            label: { formatter: (p) => p.name + '  ' + Math.round(p.percent) + '%', ...bigLabel, fontSize: 26 }, labelLine: { lineStyle: { color: MUTED }, length: 20, length2: 30 } }],
        },
        dots: {
          grid: { left: 190, right: 190, top: 340, bottom: 150 },
          xAxis: { type: 'value', min: 0, max: 50, name: 'tickets (thousands)', nameLocation: 'middle', nameGap: 52, axisLabel: { color: MUTED } },
          yAxis: { type: 'value', min: 100, max: 240, name: 'average price €', nameLocation: 'middle', nameGap: 76, axisLabel: { color: MUTED }, splitLine: { lineStyle: { color: 'rgba(255,255,255,.08)' } } },
          series: [{ id: 'tickets', type: 'scatter', data: ch.map((c) => ({ ...item(c, [c[1], c[2]]), symbolSize: 30 + c[3] * 9 })),
            label: { show: true, formatter: '{b}', position: 'top', distance: 12, ...bigLabel, fontSize: 24 } }],
        },
        revenue: {
          grid: { left: 280, right: 190, top: 330, bottom: 150 },
          xAxis: { type: 'value', max: 8, axisLabel: { color: MUTED, formatter: '€{value}M' }, splitLine: { lineStyle: { color: 'rgba(255,255,255,.08)' } } },
          yAxis: { type: 'category', inverse: true, data: ch.map((c) => c[0]), axisLine: { show: false }, axisLabel: { ...bigLabel, fontSize: 24 } },
          series: [{ id: 'tickets', type: 'bar', barWidth: '58%', data: ch.map((c) => item(c, c[3])), itemStyle: { borderRadius: [0, 10, 10, 0] },
            label: { show: true, position: 'right', formatter: '€{c} M', ...bigLabel } }],
        },
      },
    });
  });
