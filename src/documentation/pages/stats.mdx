<SkillPanel component="stats">
  <SkillField label="Native basis">
<SkillText>CSS Grid columns plus a <code>border-left</code> rule per metric; numbers set <code>font-variant-numeric: tabular-nums</code> so digits align without monospace fonts. No counters, no animation libraries.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/font-variant-numeric#tabular-nums" title="aligned digit columns"><code>font-variant-numeric</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_grid_layout" title="2-up metric grid beside the photo"><code>CSS Grid</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_containment/Container_queries" title="side-by-side only when wide"><code>Container queries</code></ApiPill>
