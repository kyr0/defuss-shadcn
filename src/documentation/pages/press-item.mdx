<PageHeader slug="press-item" title="Press Item" component="press-item">
  Links to one external media mention with publication metadata - the outlet, the date and the headline, a pull quote, or a compact list row. Part of the <DocLink href="website-company.html">Company</DocLink> blocks.
</PageHeader>

<SkillPanel component="press-item">
  <SkillField label="Native basis">
<SkillText>An <code>&lt;article&gt;</code>: the publication and a <code>&lt;time&gt;</code>, an optional <code>&lt;blockquote&gt;</code>, and the headline as an external link (<code>target="_blank" rel="noopener"</code>) that covers the card.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/article" title="a self-contained item"><code>&lt;article&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/time" title="machine-readable dates"><code>&lt;time&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/blockquote" title="a quoted post with its source"><code>&lt;blockquote&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="parent styles react to a checked input or a state inside"><code>:has()</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-press-item</code>
    <code>.mk-press-item-title</code>
    <code>.mk-press-item-meta</code>
    <code>.mk-press-item-publication</code>
    <code>.mk-press-item-quote</code>
    <code>.mk-press-item-more</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant="quote"</code>, <code>data-variant="compact"</code>.</SkillText>
  </SkillField>
</SkillPanel>
