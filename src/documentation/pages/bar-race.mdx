  // df$ exists only after all.js boots (it loads after example scripts in the

  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-race');
    const TEXT = '#FFFFFF', MUTED = '#8A93A6', ORANGE = '#FF4D00', DIM = 'rgba(255,255,255,.22)';
    const days = [5, 10, 15, 20, 25, 30, 34];
    // [player, club color, cumulative goals at the matchdays above, goals per 90]
    const players = [
      ['M. Okoye', '#FF4D00', [3, 7, 11, 15, 19, 24, 28], 0.82],

    ];
    const G = { left: 250, right: 190, top: 320, bottom: 150 };
    const names = players.map((p) => p[0]);
    const bar = (frame, extra = {}) => ({
      grid: G,
      xAxis: { type: 'value', max: 30, axisLabel: { color: MUTED }, splitLine: { lineStyle: { color: DIM } } },
      yAxis: { type: 'category', data: names, inverse: true, max: names.length - 1, animationDuration: 600, animationDurationUpdate: 1100,
               axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 800 } },
      series: [{ id: 'goals', type: 'bar', barWidth: '64%', realtimeSort: true, animationDurationUpdate: 1100, animationEasingUpdate: 'linear',
        data: players.map((p) => ({ name: p[0], value: p[2][frame], itemStyle: { color: p[1], borderRadius: 0 } })),
        label: { show: true, position: 'right', valueAnimation: true, color: TEXT, fontWeight: 800 } }],
      graphic: [{ id: 'md', type: 'text', right: 190, top: 262, style: { text: 'MATCHDAY ' + days[frame], fill: 'rgba(255,77,0,.9)', font: "800 44px 'Arial Narrow', Arial, sans-serif" } }],
      ...extra,
    });
    const last = days.length - 1;
    const byRate = players.slice().sort((a, b) => b[3] - a[3]);

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        race: bar(0),
        final: bar(last, { graphic: [{ id: 'md', type: 'text', right: 190, top: 262, style: { text: 'FULL TIME', fill: 'rgba(255,77,0,.9)', font: "800 44px 'Arial Narrow', Arial, sans-serif" } }] }),
        per90: {
          grid: G,
          xAxis: { type: 'value', max: 1, interval: 0.25, axisLabel: { color: MUTED } },
          yAxis: { type: 'category', inverse: true, data: byRate.map((p) => p[0]), axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 800 } },
          series: [{ id: 'goals', type: 'bar', barWidth: '64%',
            data: byRate.map((p) => ({ name: p[0], value: p[3], itemStyle: { color: p[0] === 'T. Varga' ? p[1] : DIM, borderRadius: 0 } })),
            label: { show: true, position: 'right', formatter: (p) => p.value.toFixed(2), color: TEXT, fontWeight: 800 } }],
        },
        season: {
          legend: { top: 300, right: 190, textStyle: { color: TEXT } },
          grid: { left: 150, right: 230, top: 370, bottom: 150 },
          xAxis: { type: 'category', data: days.map((d) => 'MD ' + d), boundaryGap: false, axisLabel: { color: TEXT, fontWeight: 700 } },
          yAxis: { type: 'value', max: 30, axisLabel: { color: MUTED }, splitLine: { lineStyle: { color: DIM } } },
          series: players.slice(0, 3).map((p) => ({ id: p[0] === 'M. Okoye' ? 'goals' : p[0], name: p[0], type: 'line', data: p[2], symbolSize: 14,
            lineStyle: { color: p[1], width: 6 }, itemStyle: { color: p[1] }, endLabel: { show: true, formatter: '{a}  {c}', color: TEXT, fontWeight: 800 } })),
        },
      },
    });

    // the live replay: while the race slide is on air, tick one matchday frame
    // every 1.2s into the SAME stage instance (merge - realtimeSort re-ranks)
    const stageEl = deck.querySelector('.presentation-stage');
    const raceSlide = deck.querySelector('[data-chart-state="race"]');
    let timer = 0;
    const stop = () => { clearInterval(timer); timer = 0; };
    new MutationObserver(() => {
      stop();
      if (!raceSlide.hasAttribute('data-active')) return;
      let frame = 0;
      timer = setInterval(() => {
        frame += 1;
        const inst = df$.shadcn.chart.instance(stageEl);
        if (!inst || frame > last) return stop();
        inst.setOption({
          series: [{ id: 'goals', data: players.map((p) => ({ name: p[0], value: p[2][frame], itemStyle: { color: p[1], borderRadius: 0 } })) }],
          graphic: [{ id: 'md', type: 'text', right: 190, top: 262, style: { text: 'MATCHDAY ' + days[frame], fill: 'rgba(255,77,0,.9)', font: "800 44px 'Arial Narrow', Arial, sans-serif" } }],
        });
      }, 1200);
    }).observe(raceSlide, { attributes: true, attributeFilter: ['data-active'] });
  });
</script>
