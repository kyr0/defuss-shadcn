<PageHeader slug="service-item" title="Service Item" component="service-item">
  Introduces one service with its description and detail link - as a card with an icon, a card with a picture, or an inline row. Part of the <DocLink href="website-company.html">Company</DocLink> blocks.
</PageHeader>

<SkillPanel component="service-item">
  <SkillField label="Native basis">
<SkillText>An <code>&lt;article&gt;</code> with an icon tile (or a picture), a heading whose link covers the card, a description and a "Learn more" cue (decorative - the title link is the accessible name).</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/article" title="a self-contained item"><code>&lt;article&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="parent styles react to a checked input or a state inside"><code>:has()</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_containment/Container_queries" title="the layout follows the block width, not the viewport"><code>Container queries</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/text-wrap" title="balanced titles, pretty paragraphs"><code>text-wrap</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-service-item</code>

  </SkillField>

</SkillPanel>
