<PageHeader slug="blog-header" title="Blog Header" component="blog-header">
  Introduces a blog with its title and editorial description - actions (subscribe, RSS), category chips, centered or on a banner. Part of the <DocLink href="website-blog.html">Blog</DocLink> blocks.
</PageHeader>

<SkillPanel component="blog-header">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;header&gt;</code> with the title (<code>h1</code>), an editorial description, optional actions (Buttons - subscribe, RSS) and a <code>&lt;nav&gt;</code> of category links; <code>aria-current</code> marks the current category. Each category is a real URL - no filter script.</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="button.html">Button</DocLink>, <DocLink href="input.html">Input</DocLink>, <DocLink href="toggle-group.html">Toggle Group</DocLink>, <DocLink href="select.html">Select</DocLink>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>

```html example label="Default" hint="Eyebrow, title, description, subscribe + RSS actions and the categories - the current one filled (aria-current)." previewStyle="align-items:stretch;"
<header class="mk-blog-header">
  <div class="mk-blog-header-copy">
    <span class="mk-blog-header-eyebrow">The Acme Blog</span>
    <h1 class="mk-blog-header-title">Notes on building calm software</h1>
    <p class="mk-blog-header-desc">Essays on product, engineering and design from the team behind Acme - one post a week, no hot takes.</p>
  </div>
  <div class="mk-blog-header-actions">
    <a class="btn" data-size="sm" href="#"><i data-lucide="mail"></i> Subscribe</a>
    <a class="btn" data-variant="outline" data-size="sm" href="#" type="application/rss+xml"><i data-lucide="rss"></i> RSS</a>
  </div>
  <nav class="mk-blog-header-categories" aria-label="Categories">
    <a href="#" aria-current="page">All posts</a>
    <a href="#">Engineering</a>
    <a href="#">Design</a>
    <a href="#">Product</a>
    <a href="#">Company</a>
  </nav>
</header>
```
