  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-band');
    const TEXT = '#17212B', MUTED = '#6B7580', RED = '#B42025', BLUE = '#2B5C8A';
    const weeks = Array.from({ length: 20 }, (_, i) => 'W' + (i + 1));
    const observed = [2.1, 2.4, 2.9, 3.6, 4.4, 5.5, 6.8, 8.1, 9.6];
    const median = [11.0, 12.1, 12.8, 13.0, 12.6, 11.7, 10.4, 8.9, 7.4, 6.0, 4.8];
    const h50 = [0.6, 1.0, 1.4, 1.8, 2.1, 2.3, 2.4, 2.4, 2.3, 2.1, 1.9];
    const h90 = [1.3, 2.2, 3.1, 3.9, 4.5, 4.9, 5.1, 5.1, 4.9, 4.5, 4.0];
    const realized = [10.7, 13.4, 13.9, 15.1, 13.1, 14.3, 11.8, 11.9, 8.4, 5.1, 2.6];
    const pad = (arr) => [...Array(9).fill('-'), ...arr];
    const r1 = (v) => Math.round(v * 10) / 10;
    const G = { left: 150, right: 150, top: 330, bottom: 150 };
    const axes = {
      grid: G,
      xAxis: { type: 'category', data: weeks, boundaryGap: false, axisLabel: { color: MUTED, interval: 1 } },
      yAxis: { type: 'value', min: 0, max: 20, interval: 5, axisLabel: { color: MUTED } },
    };
    const obs = { id: 'observed', name: 'Observed', type: 'line', data: [...observed, ...Array(11).fill('-')], symbolSize: 12, lineStyle: { color: TEXT, width: 4 }, itemStyle: { color: TEXT } };
    const med = { id: 'median', name: 'Median forecast', type: 'line', data: [...Array(8).fill('-'), 9.6, ...median], showSymbol: false, lineStyle: { color: BLUE, width: 4, type: 'dashed' }, itemStyle: { color: BLUE } };
    const band = (id, half, color) => [
      { id: id + '-lo', type: 'line', stack: id, data: pad(median.map((m, i) => r1(m - half[i]))), showSymbol: false, silent: true, lineStyle: { opacity: 0 }, tooltip: { show: false } },
      { id: id + '-w', type: 'line', stack: id, data: pad(half.map((h) => r1(2 * h))), showSymbol: false, silent: true, lineStyle: { opacity: 0 }, areaStyle: { color }, tooltip: { show: false } },
    ];
    const b90 = band('b90', h90, 'rgba(43,92,138,.16)');
    const b50 = band('b50', h50, 'rgba(43,92,138,.30)');
    const peak = { symbol: 'none', silent: true, lineStyle: { color: MUTED, type: 'dotted', width: 2 }, label: { formatter: 'forecast peak', color: MUTED, position: 'end' }, data: [{ xAxis: 'W13' }] };

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'axis' } },
      states: {
        observed: { ...axes, series: [obs] },
        forecast: { ...axes, series: [...b90, obs, med] },
        bands: { ...axes, series: [...b90, ...b50, obs, { ...med, markLine: peak }] },
        check: { ...axes, series: [...b90, ...b50, obs, med,
          { id: 'realized', name: 'Realized', type: 'scatter', data: pad(realized), symbolSize: 18, itemStyle: { color: RED, borderColor: '#FAFAF8', borderWidth: 3 }, z: 5,
            markPoint: { symbol: 'circle', symbolSize: 1, label: { show: true, formatter: 'peak 15.1', color: RED, fontWeight: 700, offset: [0, -34] }, data: [{ coord: ['W13', 15.1] }] } }] },
      },
    });
  });
</script>
