  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-river');
    const TEXT = '#E8FBFF', MUTED = '#68A4AD', GRID = 'rgba(232,251,255,.08)';
    const formats = [['Print', '#5B8CFF'], ['E-books', '#5AF2C8'], ['Audiobooks', '#F6D365'], ['Board games', '#B983FF']];
    const color = Object.fromEntries(formats);
    // loans per year (thousands) and each format's seasonal rhythm (peak month, amplitude)
    const annual = { Print: [612, 574, 541], 'E-books': [188, 214, 236], Audiobooks: [74, 121, 182], 'Board games': [18, 27, 39] };
    const rhythm = { Print: [6.5, 0.16], 'E-books': [0.5, 0.12], Audiobooks: [6.5, 0.3], 'Board games': [11, 0.55] };
    // the yearly rate interpolated between mid-year points, so the streams flow instead of stepping
    const rate = (n, t) => {
      const y = Math.min(1, Math.max(0, Math.floor((t - 5.5) / 12)));
      const f = (t - 5.5 - y * 12) / 12;
      return annual[n][y] + (annual[n][y + 1] - annual[n][y]) * f;
    };
    const data = [];
    for (let t = 0; t < 36; t++) {
      const date = (2023 + Math.floor(t / 12)) + '/' + String((t % 12) + 1).padStart(2, '0') + '/01';
      for (const [n] of formats) {
        const [peak, amp] = rhythm[n];
        data.push([date, Math.round((rate(n, t) / 12) * (1 + amp * Math.cos(((t % 12) - peak) / 12 * 2 * Math.PI)) * 10) / 10, n]);
      }
    }
    const river = (focus) => ({
      legend: { top: 300, left: 150, textStyle: { color: TEXT }, data: formats.map((f) => f[0]) },
      singleAxis: { top: 370, bottom: 130, left: 150, right: 150, type: 'time', axisLabel: { color: MUTED }, axisLine: { lineStyle: { color: 'rgba(232,251,255,.25)' } }, splitLine: { show: false } },
      series: [{ id: 'loans', type: 'themeRiver', data, label: { show: false }, emphasis: { focus: 'series' },
        color: formats.map(([n, c]) => (!focus || n === focus ? c : 'rgba(104,164,173,.2)')) }],
    });
    const years = ['2023', '2024', '2025'];
    const change = formats.map(([n]) => [n, Math.round((annual[n][2] / annual[n][0] - 1) * 100)]).sort((a, b) => b[1] - a[1]);

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        river: river(null),
        audio: river('Audiobooks'),
        yearly: {
          legend: { top: 300, left: 150, textStyle: { color: TEXT } },
          grid: { left: 190, right: 190, top: 380, bottom: 140 },
          xAxis: { type: 'category', data: years, axisLabel: { color: TEXT, fontWeight: 600 } },
          yAxis: { type: 'value', axisLabel: { color: MUTED, formatter: '{value} k' }, splitLine: { lineStyle: { color: GRID } } },
          series: formats.map(([n, c], i) => ({ id: n, name: n, type: 'bar', stack: 'loans', barWidth: '44%', data: annual[n],
            itemStyle: { color: c, borderRadius: i === formats.length - 1 ? [10, 10, 0, 0] : 0 } })),
        },
        growth: {
          grid: { left: 330, right: 220, top: 340, bottom: 140 },
          xAxis: { type: 'value', axisLabel: { color: MUTED, formatter: '{value}%' }, splitLine: { lineStyle: { color: GRID } } },
          yAxis: { type: 'category', inverse: true, data: change.map((c) => c[0]), axisLine: { show: false }, axisTick: { show: false }, axisLabel: { color: TEXT, fontWeight: 600 } },
          series: [{ id: 'growth', type: 'bar', barWidth: '56%',
            data: change.map(([n, v]) => ({ name: n, value: v, itemStyle: { color: color[n], borderRadius: v >= 0 ? [0, 10, 10, 0] : [10, 0, 0, 10] },
              label: { position: v >= 0 ? 'right' : 'left' } })),
            label: { show: true, color: TEXT, fontWeight: 700, formatter: (p) => (p.value > 0 ? '+' : '') + p.value + '%' } }],
        },
      },
    });
  });
