<PageHeader slug="statistic" title="Statistic" component="statistic">
  A numeric display for key metrics. Supports title, value, description, and trend indicators.
</PageHeader>

<SkillPanel component="statistic">
  <SkillField label="Native basis">
<SkillText><code>&lt;div&gt;</code> containers displaying a numeric value with a label and optional trend indicator.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/div" title="generic container"><code>&lt;div&gt;</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.statistic</code>
    <code>.statistic-title</code>
    <code>.statistic-value</code>
    <code>.statistic-description</code>
    <code>.statistic-trend</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText>• <code>data-trend</code> — values: <code>up</code>, <code>down</code></SkillText>
  </SkillField>

  <SkillField label="Accessibility">
<SkillText dense>• Use semantic text content — values are read inline</SkillText>
  <p class="text-muted-foreground" style="margin:0 0 0;">• Trends are conveyed by text, not color alone ("+12.5%")</p>
  </SkillField>
</SkillPanel>
