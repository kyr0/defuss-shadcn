  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-quadrants');
    const TEXT = '#12304A', MUTED = '#6E8193', TEAL = '#0A7E8C', DIM = 'rgba(18,48,74,.18)';
    // [product, growth %, NRR %, ARR € M, 2026 target growth, 2026 target NRR]
    const p = [
      ['Nimbus', 62, 128, 14, 55, 126], ['Ledgerly', 48, 118, 22, 42, 118], ['Relay', 33, 112, 9, 38, 114], ['Harbor CRM', 18, 104, 41, 16, 106],
      ['Atlas', 12, 109, 31, 14, 110], ['Pathway', 71, 96, 6, 58, 106], ['Sprout', 55, 88, 4, 45, 101], ['Quill', 9, 93, 12, 4, 94],
    ];
    const star = (r) => r[1] >= 30 && r[2] >= 100;
    const G = { left: 190, right: 190, top: 330, bottom: 150 };
    const map = (xi, yi, color) => ({
      grid: G,
      xAxis: { type: 'value', min: 0, max: 80, name: 'growth, year over year', nameLocation: 'middle', nameGap: 52, axisLabel: { color: MUTED, formatter: '{value}%' }, splitLine: { show: false } },
      yAxis: { type: 'value', min: 80, max: 135, name: 'net revenue retention', nameLocation: 'middle', nameGap: 84, axisLabel: { color: MUTED, formatter: '{value}%' } },
      series: [{ id: 'products', type: 'scatter',
        data: p.map((r) => ({ name: r[0], value: [r[xi], r[yi], r[3]], symbolSize: 22 + r[3] * 2.4, itemStyle: { color: color(r), borderColor: '#F3F6FA', borderWidth: 3 } })),
        label: { show: true, formatter: '{b}', position: 'top', distance: 10, color: TEXT, fontWeight: 700 },
        labelLayout: { hideOverlap: false, moveOverlap: 'shiftY' },
        markLine: { symbol: 'none', silent: true, label: { show: false }, lineStyle: { color: MUTED, type: 'dashed', width: 2 }, data: [{ xAxis: 30 }, { yAxis: 100 }] },
        markArea: { silent: true, itemStyle: { color: 'rgba(10,126,140,.08)' }, label: { color: TEAL, fontWeight: 700, position: 'insideTopRight' }, data: [[{ name: 'STARS', xAxis: 30, yAxis: 100 }, { xAxis: 80, yAxis: 135 }]] } }],
    });
    const byArr = p.slice().sort((a, b) => b[3] - a[3]);

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        map: map(1, 2, (r) => (star(r) ? TEAL : TEXT)),
        stars: map(1, 2, (r) => (star(r) ? TEAL : DIM)),
        arr: {
          grid: { left: 250, right: 190, top: 320, bottom: 150 },
          xAxis: { type: 'value', max: 45, axisLabel: { color: MUTED, formatter: '€{value}M' } },
          yAxis: { type: 'category', inverse: true, data: byArr.map((r) => r[0]), axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 700 } },
          series: [{ id: 'products', type: 'bar', barWidth: '60%',
            data: byArr.map((r) => ({ name: r[0], value: r[3], itemStyle: { color: star(r) ? TEAL : TEXT } })),
            label: { show: true, position: 'right', formatter: '€{c} M', color: TEXT, fontWeight: 700 } }],
        },
        plan: map(4, 5, (r) => (r[0] === 'Pathway' || r[0] === 'Sprout' ? TEAL : DIM)),
      },
    });
  });
