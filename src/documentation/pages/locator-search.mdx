<script>
  document.addEventListener('DOMContentLoaded', () => {
    df$('[data-locate]').on('click', () => {
      const out = df$('.mk-locator-search-summary');
      if (!('geolocation' in navigator)) { out.text('Location is not available in this browser.'); return; }
      out.text('Finding your location…');
      navigator.geolocation.getCurrentPosition(
        (p) => out.text('Stores near ' + p.coords.latitude.toFixed(2) + ', ' + p.coords.longitude.toFixed(2)),
        () => out.text('Location was not shared - search by address instead.'),
      );
    });
  });
</script>
