  // df$ exists only after all.js boots (it loads after example scripts in the

  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-shift');
    const TEXT = '#15252D', MUTED = '#7C8A91', UP = '#00897B', DOWN = '#E33F32';
    // [stop, turnout 2021, turnout 2025, share under 30 %] - in line order
    const stops = [['Riverside', 61.2, 66.8, 24], ['Old Mill', 58.4, 55.1, 16], ['Central', 64.0, 70.2, 31], ['Market', 55.3, 57.9, 22], ['University', 48.9, 60.4, 47], ['Parkview', 66.7, 65.9, 18],
      ['Docklands', 51.2, 59.8, 38], ['Hilltop', 69.5, 64.1, 12], ['Northgate', 57.6, 58.2, 21], ['Lakeside', 63.3, 61.0, 15], ['Foundry', 52.8, 57.4, 29], ['Airport Rd', 49.7, 46.5, 19]];
    const ch = (s) => Math.round((s[2] - s[1]) * 10) / 10;
    const G = { left: 150, right: 150, top: 330, bottom: 170 };
    // shaft (bar from 0 to change) + head (triangle at the tip) + station dot on the zero line
    const arrows = (list) => ({
      grid: G,
      xAxis: { type: 'category', data: list.map((s) => s[0]), position: 'bottom', axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 700, interval: 0, rotate: 30 } },
      yAxis: { type: 'value', min: -8, max: 14, interval: 4, axisLabel: { color: MUTED, formatter: (v) => (v > 0 ? '+' : '') + v } },
      series: [
        { id: 'shaft', type: 'bar', barWidth: 8, data: list.map((s) => ({ name: s[0], value: ch(s), itemStyle: { color: ch(s) >= 0 ? UP : DOWN, borderRadius: 0 } })) },
        { id: 'tip', type: 'scatter', symbol: 'triangle', symbolSize: 30,
          data: list.map((s) => ({ name: s[0], value: [s[0], ch(s)], symbolRotate: ch(s) >= 0 ? 0 : 180, itemStyle: { color: ch(s) >= 0 ? UP : DOWN }, label: { position: ch(s) >= 0 ? 'top' : 'bottom' } })),
          label: { show: true, distance: 14, formatter: (p) => (p.value[1] > 0 ? '+' : '') + p.value[1], color: TEXT, fontWeight: 700 } },
        { id: 'line', type: 'line', data: list.map(() => 0), symbol: 'circle', symbolSize: 22, z: 1, silent: true,
          lineStyle: { color: DOWN, width: 12 }, itemStyle: { color: '#FFFFFF', borderColor: TEXT, borderWidth: 5 } },
      ],
    });
    const byChange = stops.slice().sort((a, b) => ch(b) - ch(a));

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        stops: arrows(stops),
        ranked: arrows(byChange),
        levels: {
          grid: G,
          xAxis: { type: 'category', data: byChange.map((s) => s[0]), axisLabel: { color: TEXT, fontWeight: 700, interval: 0, rotate: 30 } },
          yAxis: { type: 'value', min: 0, max: 80, interval: 20, axisLabel: { color: MUTED, formatter: '{value}%' } },
          series: [
            { id: 'shaft', type: 'bar', barWidth: '46%', data: byChange.map((s) => ({ name: s[0], value: s[2], itemStyle: { color: s[0] === 'Central' ? UP : s[0] === 'Airport Rd' ? DOWN : '#9FB3BD', borderRadius: [6, 6, 0, 0] } })),
              label: { show: true, position: 'top', formatter: '{c}%', color: TEXT, fontWeight: 700 } },
          ],
        },
        age: {

          yAxis: { type: 'value', min: -8, max: 14, name: 'turnout change, pts', nameLocation: 'middle', nameGap: 70, axisLabel: { color: MUTED } },
