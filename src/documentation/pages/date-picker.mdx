<PageHeader slug="date-picker" title="Date Picker" component="date-picker">
  A date selection input. Built on native <code>&lt;input type="date"&gt;</code> and <code>&lt;input type="datetime-local"&gt;</code> — the browser provides the calendar UI.
</PageHeader>

<SkillPanel component="date-picker">
  <SkillField label="Native basis">
<SkillText><code>&lt;input type="date"&gt;</code> element with native browser date picker.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/input/date" title="native date picker with calendar UI"><code>&lt;input type="date"&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/input/datetime-local" title="date and time picker"><code>&lt;input type="datetime-local"&gt;</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.input</code>
    <code>.date-input</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Notes">
<SkillText dense>• Reuses the <code>.input</code> class — the native date picker provides the calendar UI.</SkillText>
  <SkillText dense>• No custom calendar implementation needed — the browser handles it.</SkillText>
  <SkillText dense>• The calendar popup is rendered by the OS/browser and cannot be styled.</SkillText>
  <p class="text-muted-foreground" style="margin:0 0 0;">• For a fully custom date picker, a custom calendar component would be needed.</p>
  </SkillField>
</SkillPanel>
