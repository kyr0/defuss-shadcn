<PageHeader slug="toc" title="Table of Contents" component="toc">
  A section-nav block of heading links — the docs rail you're looking at on the right, packaged as a reusable pattern. Native fragment links do the scrolling; the current section is marked with <code>aria-current</code>. No JavaScript shipped — the host's IntersectionObserver moves the marker.
</PageHeader>

<SkillPanel component="toc">
  <SkillField label="Native basis">
    <SkillText><code>&lt;nav&gt;</code> with a title paragraph and one <code>&lt;a&gt;</code> per section heading. Native fragment links do the scrolling; the active link carries <code>aria-current="location"</code>.</SkillText>
  </SkillField>
  <SkillField label="Web Platform APIs">
    <SkillRow>
      <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/nav" title="landmark for major navigation"><code>&lt;nav&gt;</code></ApiPill>
      <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:target" title="native deep links"><code>:target</code></ApiPill>
      <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/Accessibility/ARIA/Reference/Attributes/aria-current" title="marks the link for the section in view"><code>aria-current</code></ApiPill>
      <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/API/IntersectionObserver" title="the host's way to track the visible section"><code>IntersectionObserver</code></ApiPill>
      <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/position" title="the rail follows the scroll without JS"><code>position: sticky</code></ApiPill>
    </SkillRow>
  </SkillField>

  </SkillField>
</SkillPanel>
