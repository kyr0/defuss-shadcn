  A section-nav block of heading links - the docs rail you're looking at on the right, packaged as a reusable pattern. Native fragment links do the scrolling; the current section is marked with <code>aria-current</code>. No JavaScript shipped - the host's IntersectionObserver moves the marker.

<script>
  // host behavior (not shipped): the section entering the top quarter of the
  // scroll box becomes the current one
  (() => {
    const root = document.getElementById('toc-left');
    const main = root.querySelector('main');
    const links = [...root.querySelectorAll('.toc-link')];
    const mark = (id) => links.forEach((l) => {
      if (l.getAttribute('href') === '#' + id) l.setAttribute('aria-current', 'location');
      else l.removeAttribute('aria-current');
    });
    const io = new IntersectionObserver((entries) => {
      for (const e of entries) if (e.isIntersecting) mark(e.target.id);
    }, { root: main, rootMargin: '0px 0px -75% 0px' });
    main.querySelectorAll('section').forEach((s) => io.observe(s));
  })();
</script>
```
