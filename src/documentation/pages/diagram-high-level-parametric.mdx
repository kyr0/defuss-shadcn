<PageHeader slug="diagram-high-level-parametric" title="Illustrative · High-level Parametric" component="diagram">
  The same high-level picture, generated: a spec - <code>&#123; type, cols, phases, groups, nodes, edges &#125;</code> - in a <code>&lt;script type="application/json" class="diagram-spec"&gt;</code> inside an empty figure, or passed to <code>df$.shadcn.diagram.build(el, spec)</code>. The spec becomes exactly the markup an author would write (<code>df$.shadcn.diagram.markup(spec)</code> returns it), so everything else - wires, steps, delta - works unchanged. Same inputs, same picture.
</PageHeader>

<H2 id="parameters">Parameters</H2>

```html example label="Inputs → spec → diagram" component="diagram" hint="The slider and checkboxes feed a function that returns a spec; build() re-renders the figure on every input."
<div class="flex flex-wrap items-center gap-4 mb-3">
  <label class="label" for="dg-sources">Sources</label>
  <input class="slider" id="dg-sources" type="range" min="1" max="5" value="3" style="width:10rem">
  <label class="label"><input class="checkbox" id="dg-stream" type="checkbox"> Streaming</label>
  <label class="label"><input class="checkbox" id="dg-semantic" type="checkbox" checked> Semantic layer</label>
</div>
<figure class="diagram" id="dg-param-inputs" data-type="high-level" aria-label="A generated data stack"></figure>
<script>
  const SOURCES = ['PostgreSQL', 'MySQL', 'SFTP drop', 'REST APIs', 'Mainframe'];
  // the inputs → a spec → df$.shadcn.diagram.build(): same inputs, same picture
  function specFor({ sources, stream, semantic }) {
    const rows = Math.max(sources, 2);
    const nodes = SOURCES.slice(0, sources).map((name, i) => ({ id: 's' + i, name, col: 1, row: i + 2, tone: 'external' }));
    nodes.push({ id: 'ingest', name: stream ? 'Kafka + NiFi' : 'NiFi', meta: stream ? 'stream + batch' : 'batch', col: 2, row: 2, rspan: rows });
    nodes.push({ id: 'lake', name: 'Lake', meta: 'Iceberg', col: 3, row: 2, rspan: rows, shape: 'store', tone: 'accent' });
    nodes.push({ id: 'bi', name: 'Dashboards', col: 4, row: 2, rspan: semantic ? 1 : rows });
    if (semantic) nodes.push({ id: 'cube', name: 'Semantic layer', meta: 'metrics', col: 4, row: 3, rspan: rows - 1 });
    const edges = nodes.filter((n) => n.id.startsWith('s')).map((n) => ({ from: n.id, to: 'ingest' }));
    edges.push({ from: 'ingest', to: 'lake', label: stream ? 'stream' : 'batch', line: stream ? 'dashed' : undefined });
    edges.push(semantic ? { from: 'lake', to: 'cube', label: 'models' } : { from: 'lake', to: 'bi', label: 'SQL' });
    if (semantic) edges.push({ from: 'cube', to: 'bi', label: 'metrics' });
    return { type: 'high-level', cols: 4, phases: ['Sources', 'Ingest', 'Lake', 'Serve'], nodes, edges };
  }
  const read = () => ({
    sources: Number(document.getElementById('dg-sources').value),
    stream: document.getElementById('dg-stream').checked,
    semantic: document.getElementById('dg-semantic').checked,
  });
  const draw = () => df$.shadcn.diagram.build('#dg-param-inputs', specFor(read()));
  document.querySelectorAll('#dg-sources, #dg-stream, #dg-semantic').forEach((input) => input.addEventListener('input', draw));
  addEventListener('load', draw);
</script>
```
