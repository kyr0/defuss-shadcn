  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-boxplot');
    const TEXT = '#202428', MUTED = '#8B8F93', OX = '#5E1B28', GREY = '#9DA2A6';
    // [method, min, q1, median, q3, max, mean, gain]
    const m = [['Tutoring + phonics', 49, 64, 72, 79, 95, 71.3, 14.8], ['Phonics', 41, 58, 66, 74, 92, 65.8, 11.2], ['Mixed', 38, 54, 63, 71, 90, 62.6, 9.9], ['Self-paced app', 22, 45, 61, 76, 97, 60.2, 9.4], ['Whole language', 30, 47, 57, 66, 88, 56.4, 7.1]];
    const G = { left: 330, right: 190, top: 330, bottom: 150 };
    const cats = (list) => ({ type: 'category', inverse: true, data: list.map((r) => r[0]), axisLine: { show: false }, axisLabel: { color: TEXT } });
    const score = { type: 'value', min: 0, max: 100, interval: 20, axisLabel: { color: MUTED } };
    const hot = (r, key) => (r[0] === key ? OX : GREY);
    const bars = (list, value, key, fmt, axis) => ({
      grid: G, xAxis: axis, yAxis: cats(list),
      series: [{ id: 'methods', type: 'bar', barWidth: '46%',
        data: list.map((r) => ({ name: r[0], value: typeof value === 'function' ? value(r) : r[value], itemStyle: { color: hot(r, key), borderRadius: 0 } })),
        label: { show: true, position: 'right', formatter: fmt, color: TEXT, fontWeight: 700 } }],
    });

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        box: {
          grid: G, xAxis: score, yAxis: cats(m),
          series: [{ id: 'methods', type: 'boxplot', boxWidth: [30, 56],
            data: m.map((r) => ({ name: r[0], value: r.slice(1, 6), itemStyle: { color: r[0] === 'Self-paced app' ? 'rgba(94,27,40,.14)' : 'rgba(32,36,40,.06)', borderColor: r[0] === 'Self-paced app' ? OX : TEXT, borderWidth: 3 } })) }],
        },
        means: bars(m, 6, 'Tutoring + phonics', '{c}', score),
        iqr: bars(m.slice().sort((a, b) => (b[4] - b[2]) - (a[4] - a[2])), (r) => r[4] - r[2], 'Self-paced app', '{c} pts', { type: 'value', max: 35, axisLabel: { color: MUTED } }),
        gain: bars(m, 7, 'Tutoring + phonics', '+{c}', { type: 'value', max: 16, axisLabel: { color: MUTED } }),
      },
    });
  });
</script>
