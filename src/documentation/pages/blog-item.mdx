<PageHeader slug="blog-item" title="Blog Item" component="blog-item">
  Previews one post with its image, excerpt, and metadata - a card, a horizontal row, or a featured post with the copy over the image. Part of the <DocLink href="website-blog.html">Blog</DocLink> blocks.
</PageHeader>

<SkillPanel component="blog-item">
  <SkillField label="Native basis">
<SkillText>An <code>&lt;article&gt;</code> - image, category (Badge) + read time, the title link stretched over the item, a clamped excerpt and the byline (Avatar, name, <code>&lt;time&gt;</code>).</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="badge.html">Badge</DocLink>, <DocLink href="avatar.html">Avatar</DocLink>, <DocLink href="swap.html">Swap</DocLink>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/article" title="one post"><code>&lt;article&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/-webkit-line-clamp" title="three-line excerpt"><code>line-clamp</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/aspect-ratio" title="3:2 images"><code>aspect-ratio</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/scale" title="the image zooms on hover"><code>scale</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-blog-item</code>

    <code>.mk-blog-item-media</code>

  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant</code> (<code>horizontal</code>, <code>featured</code>); <code>data-variant</code> (<code>card</code>, <code>minimal</code>, <code>quote</code>, <code>podcast</code>); <code>.mk-blog-item-tags</code>, <code>.mk-blog-item-actions</code> (a bookmark Swap), <code>.mk-blog-item-player</code>.</SkillText>
  </SkillField>
</SkillPanel>
