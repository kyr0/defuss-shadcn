<PageHeader slug="order-summary" title="Order Summary" component="order-summary">
  Displays an order's items and charge breakdown - pictures with quantity badges, the lines, discount, shipping and total; collapsible on small screens. Part of the <DocLink href="website-shop.html">Shop</DocLink> blocks.
</PageHeader>

<SkillPanel component="order-summary">
  <SkillField label="Native basis">
<SkillText>An <code>&lt;aside&gt;</code> (or a <code>&lt;details&gt;</code> for the collapsible variant): the items as a list with a quantity badge on each thumbnail, then a <code>&lt;dl&gt;</code> of subtotal, discount, shipping and the total.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/dl" title="term / value pairs"><code>&lt;dl&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/details" title="native disclosure, no script"><code>&lt;details&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="parent styles react to a checked input or a state inside"><code>:has()</code></ApiPill>
  </SkillRow>
  </SkillField>
