<script>
  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-violin');
    const TEXT = '#171312', MUTED = '#8C8580', RED = '#B92A4A', GREY = '#CFC7C0';
    // [category, median, resale share %, density modes as [centre, spread, weight]]
    const cats = [
      ['Runners', 8, 17, [[6, 14, 1]]],
      ['Skate', 14, 11, [[14, 15, 1]]],
      ['Basketball', 22, 24, [[18, 16, 0.8], [45, 14, 0.3]]],
      ['Retro', 31, 26, [[28, 18, 1], [70, 16, 0.2]]],
      ['Collaborations', 96, 22, [[60, 30, 0.8], [190, 45, 0.45]]],
    ];
    const ys = Array.from({ length: 67 }, (_, i) => -30 + i * 5); // premium grid −30 … +300
    const density = (modes) => {
      const raw = ys.map((y) => modes.reduce((s, [c, sd, w]) => s + w * Math.exp(-((y - c) ** 2) / (2 * sd * sd)), 0));
      const max = Math.max(...raw);
      return raw.map((v) => v / max);
    };
    const shapes = cats.map((c) => density(c[3]));
    const G = { left: 150, right: 150, top: 330, bottom: 150 };
    const violins = (hot) => ({
      grid: G,
      xAxis: { type: 'category', data: cats.map((c) => c[0]), axisLabel: { color: TEXT, fontSize: 26 } },
      yAxis: { type: 'value', min: -50, max: 300, interval: 50, axisLabel: { color: MUTED, formatter: (v) => (v > 0 ? '+' : '') + v + '%' } },
      series: [{ id: 'violins', type: 'custom', data: cats.map((_, i) => [i]),
        renderItem: (p, api) => {
          const i = p.dataIndex, cx = api.coord([i, 0])[0], half = api.size([1, 0])[0] * 0.4;
          const pts = ys.map((y, k) => [cx - shapes[i][k] * half, api.coord([i, y])[1]])
            .concat(ys.slice().reverse().map((y, k) => [cx + shapes[i][ys.length - 1 - k] * half, api.coord([i, y])[1]]));
          const lit = !hot || cats[i][0] === hot;
          const my = api.coord([i, cats[i][1]])[1];
          return { type: 'group', children: [
            { type: 'polygon', shape: { points: pts, smooth: 0.2 }, style: { fill: lit ? (hot ? RED : TEXT) : GREY, opacity: lit ? 0.9 : 0.35 },
              enterFrom: { scaleY: 0, originY: api.coord([i, 0])[1], style: { opacity: 0 } }, transition: ['shape', 'style'] },
            { type: 'line', shape: { x1: cx - half * 0.45, y1: my, x2: cx + half * 0.45, y2: my }, style: { stroke: '#F8F4F0', lineWidth: 4 } },
          ] };
        } }],
    });

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { show: false } },
      states: {
        violins: violins(null),
        collabs: violins('Collaborations'),
        medians: {
          grid: G,
          xAxis: { type: 'category', data: cats.map((c) => c[0]), axisLabel: { color: TEXT, fontSize: 26 } },
          yAxis: { type: 'value', min: 0, max: 120, interval: 20, axisLabel: { color: MUTED, formatter: '+{value}%' } },
          series: [{ id: 'medians', type: 'bar', barWidth: '32%', data: cats.map((c) => ({ name: c[0], value: c[1], itemStyle: { color: c[0] === 'Collaborations' ? RED : TEXT, borderRadius: 0 } })),
            label: { show: true, position: 'top', formatter: '+{c}%', color: TEXT, fontSize: 28 } }],
        },
        volume: {
          series: [{ id: 'medians', type: 'pie', radius: ['30%', '50%'], center: ['50%', '64%'],
            data: cats.map((c) => ({ name: c[0], value: c[2], itemStyle: { color: c[0] === 'Collaborations' ? RED : c[0] === 'Retro' ? TEXT : c[0] === 'Basketball' ? '#5A4E49' : GREY } })),
            itemStyle: { borderColor: '#F8F4F0', borderWidth: 6 },
            label: { formatter: '{b}  {c}%', color: TEXT, fontSize: 26 }, labelLine: { lineStyle: { color: MUTED }, length: 20, length2: 30 } }],
        },
      },
    });
  });
</script>
