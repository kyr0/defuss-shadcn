<PageHeader slug="news-header" title="News Header" component="news-header">
  Introduces a news collection with its title and editorial context - kicker, dateline, topic links, or a newspaper masthead. Part of the <DocLink href="website-news.html">News</DocLink> blocks.
</PageHeader>

<SkillPanel component="news-header">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;header&gt;</code> with the title (<code>h1</code>), a <code>&lt;time&gt;</code> dateline and a <code>&lt;nav&gt;</code> of topic links - <code>aria-current</code> marks the current topic, the row scrolls on its own when it overflows.</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="breadcrumb.html">Breadcrumb</DocLink>, <DocLink href="indicator.html">Indicator</DocLink>, <DocLink href="input.html">Input</DocLink>, <DocLink href="button.html">Button</DocLink>, <DocLink href="toggle-group.html">Toggle Group</DocLink>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/time" title="machine-readable dateline"><code>&lt;time&gt;</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/Accessibility/ARIA/Attributes/aria-current" title="the current topic"><code>aria-current</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/overscroll-behavior" title="the topic row scrolls in place"><code>overscroll-behavior</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_containment/Container_queries" title="the title scales"><code>Container queries</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant="masthead"</code>; <code>aria-current</code> on the current topic link; <code>data-variant="bar"</code>, <code>data-wrap</code> on the topics; <code>.mk-news-header-live</code> (an Indicator dot + the update time), <code>.mk-news-header-tools</code>.</SkillText>
  </SkillField>
</SkillPanel>
