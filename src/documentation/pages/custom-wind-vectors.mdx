<script>
  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-wind');
    const TEXT = '#ECF8FF', MUTED = '#6F93AA', YELLOW = '#FFCE42';
    const cols = 16, rows = 8;
    // a rotating low centred at (cx, cy) with peak speed v (km/h); background westerly flow
    const field = (cx, cy, v) => {
      const pts = [];
      for (let y = 0; y < rows; y++) for (let x = 0; x < cols; x++) {
        const dx = x - cx, dy = y - cy, r = Math.hypot(dx, dy) + 0.6;
        const swirl = v * Math.min(1, 1.6 / r) * Math.exp(-r / 7);
        const u = 18 + swirl * (-dy / r), w = swirl * (dx / r);
        pts.push([x, y, Math.round(Math.hypot(u, w)), Math.atan2(w, u)]);
      }
      return pts;
    };
    const coast = 6.5; // the coastline runs north–south at x = 6.5
    const wind = (cx, cy, v) => ({
      grid: { left: 150, right: 190, top: 330, bottom: 150 },
      xAxis: { type: 'value', min: -0.5, max: cols - 0.5, show: false },
      yAxis: { type: 'value', min: -0.5, max: rows - 0.5, show: false, inverse: true },
      // bound to the arrow field only - the 2-D coastline has no speed dimension
      visualMap: { show: true, seriesIndex: 0, dimension: 2, min: 10, max: 140, orient: 'vertical', right: 40, top: 360, itemHeight: 300, itemWidth: 18,
        text: ['km/h', ''], textStyle: { color: TEXT }, inRange: { color: ['#2A5C7A', '#7CC4E8', YELLOW, '#FF6B3D'] } },
      series: [{ id: 'wind', type: 'custom', data: field(cx, cy, v),
        renderItem: (p, api) => {
          const [px, py] = api.coord([api.value(0), api.value(1)]);
          const a = api.value(3), len = 18 + api.value(2) * 0.42, col = api.visual('color');
          const hx = Math.cos(a) * len / 2, hy = Math.sin(a) * len / 2;
          return { type: 'group', x: px, y: py, children: [
            { type: 'line', shape: { x1: -hx, y1: -hy, x2: hx, y2: hy }, style: { stroke: col, lineWidth: 4, lineCap: 'round' }, transition: ['shape', 'style'] },
            { type: 'polygon', shape: { points: [[hx, hy], [hx - Math.cos(a - 0.5) * 12, hy - Math.sin(a - 0.5) * 12], [hx - Math.cos(a + 0.5) * 12, hy - Math.sin(a + 0.5) * 12]] },
              style: { fill: col }, transition: ['shape', 'style'] },
          ] };
        } },
        { id: 'coast', type: 'line', data: [[coast, -0.5], [coast, rows - 0.5]], showSymbol: false, silent: true, lineStyle: { color: TEXT, width: 3, type: 'dashed', opacity: 0.6 },
          markPoint: { symbol: 'circle', symbolSize: 1, label: { show: true, formatter: 'COAST', color: MUTED, fontWeight: 700, offset: [0, -20] }, data: [{ coord: [coast, -0.5] }] } }],
    });

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { show: false } },
      states: {
        t06: wind(3, 3.5, 90),
        t12: wind(7, 3.5, 150),
        t18: wind(11.5, 3, 95),
        gusts: {
          legend: { top: 300, right: 190, textStyle: { color: TEXT } },
          grid: { left: 150, right: 190, top: 370, bottom: 150 },
          xAxis: { type: 'category', data: ['06', '08', '10', '12', '14', '16', '18', '20', '22'], boundaryGap: false, axisLabel: { color: MUTED, formatter: '{value}:00' } },
          yAxis: { type: 'value', min: 0, max: 160, interval: 40, axisLabel: { color: MUTED }, splitLine: { lineStyle: { color: 'rgba(236,248,255,.08)' } } },
          series: [
            { id: 'harbour', name: 'Harbour Mole', type: 'line', smooth: 0.3, data: [58, 76, 104, 136, 139, 112, 84, 66, 52], lineStyle: { color: YELLOW, width: 5 }, itemStyle: { color: YELLOW }, symbolSize: 12 },
            { id: 'dune', name: 'Dune Point', type: 'line', smooth: 0.3, data: [54, 72, 101, 129, 124, 98, 77, 60, 49], lineStyle: { color: '#7CC4E8', width: 4 }, itemStyle: { color: '#7CC4E8' }, symbolSize: 10 },
            { id: 'inland', name: 'Inland Airfield', type: 'line', smooth: 0.3, data: [31, 40, 57, 78, 96, 101, 83, 61, 44], lineStyle: { color: MUTED, width: 4 }, itemStyle: { color: MUTED }, symbolSize: 10 },
          ],
        },
      },
    });
  });
