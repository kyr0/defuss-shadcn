<script>
  document.addEventListener('DOMContentLoaded', () => {
    const chat = df$('#pr-chat').get(0);
    const session = df$.shadcn.session;
    const turns = [
      { q: 'Let\'s fix the type errors - the compiler ratchet - and measure again.', a: 'All 1,147 component type errors are fixed: 12.5 minutes and 71,958 output tokens to zero, and the 85 shipped minified files are byte for byte the same as before.' },
      { q: 'Should the small-modules guideline become a strong recommendation in AGENTS.md?', a: 'Yes. One owner module removed 707 of the 1,147 errors on its own, and every cross-module failure so far came from an implicit contract - so the rule goes in, and a check where a program can hold it.' },
      { q: 'Let\'s create skills to plan, theme and review with defuss-shadcn.', a: 'Three task skills ship next to the reference skill, generated from the release; the markup checker finds no false alarm on 234 fixtures and 1,553 docs examples.' },
    ];
    const user = (text) => `<div class="message" data-align="end" data-avatar="none"><div class="message-content"><div class="bubble"><div class="bubble-content">${text}</div></div></div></div>`;
    const bot = '<div class="message" data-avatar="none"><div class="message-content"><div class="bubble" data-variant="ghost"><div class="bubble-content"></div></div></div></div>';
    let run = 0;
    // resolves false once a newer replay has started - the older one stops
    const wait = (ms, mine) => new Promise((done) => setTimeout(() => done(mine === run), ms));
    async function replay() {
      const mine = ++run;
      df$('#pr-chat .session-content').html('');
      df$('#pr-status').text('Replaying...');
      for (const [i, turn] of turns.entries()) {
        if (!(await wait(700, mine))) return;
        session.append(chat, user(turn.q), { id: `q${i}`, anchor: true });
        if (!(await wait(600, mine))) return;
        const reply = session.append(chat, bot, { id: `a${i}` });
        const out = df$(reply).find('.bubble-content');
        chat.api.setState('streaming');
        const words = turn.a.split(' ');
        for (let n = 1; n <= words.length; n++) {
          if (!(await wait(45, mine))) return;
          out.text(words.slice(0, n).join(' '));
        }
        chat.api.setState(session.isAtEnd(chat) ? 'default' : 'detached');
      }
      df$('#pr-status').text(`Replayed - ${turns.length} turns.`);
    }
    df$('#pr-replay').on('click', replay);
    replay();
  });
</script>
