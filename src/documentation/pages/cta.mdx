<PageHeader slug="cta" title="CTA" component="cta">
  Combines a focused message with a prominent action - centered, split, on a dark glow or over a picture. Part of the <DocLink href="website-events-promotions.html">Events & Promotions</DocLink> blocks.
</PageHeader>

<SkillPanel component="cta">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;section&gt;</code> named by its heading, one supporting sentence, one primary and at most one secondary action (Buttons) and an optional note.</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="button.html">Button</DocLink>.</SkillText>
  </SkillField>

```html example label="Default" hint="Centered on a muted surface: a heading, one sentence, a primary and a secondary action and a short note." previewStyle="align-items:stretch;"
<section class="mk-cta" aria-labelledby="cta-1">
  <div class="mk-cta-inner">
    <div class="mk-cta-copy">
      <h2 class="mk-cta-title" id="cta-1">Ready to work calmer?</h2>
      <p class="mk-cta-desc">Join 12,000 teams who plan, write and ship in one place.</p>
    </div>
    <div class="mk-cta-actions">
      <a class="btn" data-size="lg" href="#">Start free trial</a>
      <a class="btn" data-variant="outline" data-size="lg" href="#">Talk to sales</a>
    </div>
    <p class="mk-cta-note">Free for 14 days · No credit card required</p>
  </div>
</section>
```

```html example label="Split" hint="data-variant='split': copy left, actions right once the block is 44rem wide." previewStyle="align-items:stretch;"
<section class="mk-cta" data-variant="split" aria-labelledby="cta-2">
  <div class="mk-cta-inner">
    <div class="mk-cta-copy">
      <h2 class="mk-cta-title" id="cta-2">See Acme in 20 minutes</h2>
      <p class="mk-cta-desc">A live walkthrough with a product specialist - bring your questions.</p>
    </div>
    <div class="mk-cta-actions"><a class="btn" data-size="lg" href="#">Book a demo <i data-lucide="arrow-right"></i></a></div>
  </div>
</section>
```

```html example label="Dark with a form" hint="data-variant='dark': inverted, with a glow in the theme's chart colors - here with an inline email form." previewStyle="align-items:stretch;"
<section class="mk-cta" data-variant="dark" aria-labelledby="cta-3">
  <div class="mk-cta-inner">
    <div class="mk-cta-copy">
