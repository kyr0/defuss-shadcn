<script>
  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-lollipop');
    const INK = '#151515', MUTED = '#8C8474', RED = '#E33124', BLUE = '#1B4CA1', YELLOW = '#F2B705';
    const G = { left: 260, right: 160, top: 320, bottom: 140 };
    // [method, serving ml, mg per serving, office cups per day]
    const cups = [['Cold brew', 350, 205, 14], ['Drip', 240, 145, 48], ['Pour-over', 240, 120, 9], ['French press', 240, 107, 5], ['Moka pot', 60, 105, 6], ['Espresso', 30, 63, 36], ['Instant', 240, 62, 4]];
    const per100 = (c) => Math.round((c[2] / c[1]) * 100);

    // a lollipop = hairline stem (bar) + head (scatter); ids 'stem' + 'head' carry the morph
    const lollipop = (rows, value, hot, color, suffix, max) => ({
      grid: G,
      xAxis: { type: 'value', max, axisLabel: { color: MUTED, formatter: '{value}' } },
      yAxis: { type: 'category', inverse: true, data: rows.map((c) => c[0]), axisLine: { show: false }, axisLabel: { color: INK, fontWeight: 700 } },
      series: [
        { id: 'stem', type: 'bar', barWidth: 4, silent: true, data: rows.map((c) => ({ name: c[0], value: value(c) })), itemStyle: { color: INK, borderRadius: 0 } },
        { id: 'head', type: 'scatter', symbolSize: 34, data: rows.map((c) => ({ name: c[0], value: value(c), itemStyle: { color: c[0] === hot ? color : INK } })),
          label: { show: true, position: 'right', distance: 18, formatter: (p) => p.value + suffix, color: INK, fontWeight: 700 } },
      ],
    });
    const bySize = cups.slice().sort((a, b) => b[2] - a[2]);
    const byDensity = cups.slice().sort((a, b) => per100(b) - per100(a));

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        serving: lollipop(bySize, (c) => c[2], 'Cold brew', RED, ' mg', 240),
        density: lollipop(byDensity, per100, 'Espresso', BLUE, ' mg', 240),
        scatter: {
          grid: { left: 190, right: 160, top: 330, bottom: 150 },
          xAxis: { type: 'value', min: 0, max: 400, name: 'serving size (ml)', nameLocation: 'middle', nameGap: 52, axisLabel: { color: MUTED } },
          yAxis: { type: 'value', min: 0, max: 240, name: 'mg per serving', nameLocation: 'middle', nameGap: 84, axisLabel: { color: MUTED } },
          series: [{ id: 'head', type: 'scatter',
            data: cups.map((c) => ({ name: c[0], value: [c[1], c[2]], symbolSize: 18 + per100(c) * 0.16,
              itemStyle: { color: c[0] === 'Espresso' ? BLUE : c[0] === 'Cold brew' ? RED : INK } })),
            label: { show: true, formatter: '{b}', position: 'right', distance: 14, color: INK, fontWeight: 700 } }],
        },
        cups: {
          grid: { left: 150, right: 120, top: 330, bottom: 150 },
          xAxis: { type: 'category', data: cups.map((c) => c[0]), axisLabel: { interval: 0, color: INK, fontWeight: 700 } },
          yAxis: { type: 'value', max: 60, interval: 20, axisLabel: { color: MUTED } },
          series: [{ id: 'head', type: 'bar', barWidth: '46%',
            data: cups.map((c) => ({ name: c[0], value: c[3], itemStyle: { color: c[0] === 'Drip' ? RED : c[0] === 'Espresso' ? BLUE : c[0] === 'Cold brew' ? YELLOW : INK, borderRadius: 0 } })),
            label: { show: true, position: 'top', color: INK, fontWeight: 700 } }],
        },
      },
    });
  });
</script>
