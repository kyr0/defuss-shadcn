<PageHeader slug="tag-cloud" title="Tag Cloud" component="tag-cloud">
  Displays linked tags with optional frequency-based emphasis - five weights from data-weight, or even pills with counts. Part of the <DocLink href="website-search-navigation.html">Search & Navigation</DocLink> blocks.
</PageHeader>

<SkillPanel component="tag-cloud">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;ul&gt;</code> of tag links with <code>data-weight="1"</code>...<code>"5"</code> (frequency buckets) and the count in the link text for screen readers.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/HTML/Element/nav" title="a navigation landmark"><code>&lt;nav&gt;</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-tag-cloud</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-weight="1–5"</code> on links; <code>data-variant="pills"</code>.</SkillText>
  </SkillField>
</SkillPanel>
