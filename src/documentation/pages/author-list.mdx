<PageHeader slug="author-list" title="Author List" component="author-list">
  Lists contributors associated with a publication or article - rows with roles and counts, a card grid, or a stack of overlapping avatars. Part of the <DocLink href="website-article.html">Article</DocLink> blocks.
</PageHeader>

<SkillPanel component="author-list">
  <SkillField label="Native basis">
<SkillText>A <code>&lt;section&gt;</code> with a <code>&lt;ul&gt;</code>: each contributor an Avatar, the name (a link), a role and an optional count. The stack variant overlaps the avatars and keeps the names for assistive tech.</SkillText>
  </SkillField>

  <SkillField label="Built from">
<SkillText><DocLink href="avatar.html">Avatar</DocLink>, <DocLink href="progress.html">Progress</DocLink>, <DocLink href="tooltip.html">Tooltip</DocLink>, <DocLink href="button.html">Button</DocLink>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/clip-path" title="visually hidden names (stack)"><code>clip-path</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/:has" title="focus ring on the avatar"><code>:has()</code></ApiPill>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/grid-template-columns" title="the card grid"><code>auto-fill grid</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.mk-author-list</code>
    <code>.mk-author-list-title</code>
    <code>.mk-author-list-items</code>
    <code>.mk-author-list-item</code>
    <code>.mk-author-list-who</code>
    <code>.mk-author-list-name</code>
    <code>.mk-author-list-role</code>
    <code>.mk-author-list-count</code>
    <code>.mk-author-list-summary</code>
    <code>.mk-author-list-action</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Data attributes">
<SkillText><code>data-variant</code> (<code>grid</code>, <code>stack</code>); <code>data-variant="ranked"</code> (+ <code>data-rank</code> on the items); <code>.mk-author-list-action</code>.</SkillText>
  </SkillField>
</SkillPanel>
