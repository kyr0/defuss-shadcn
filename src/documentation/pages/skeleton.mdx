<SkillPanel component="skeleton">
  <SkillField label="Native basis">
<SkillText>CSS <code>@keyframes</code> pulse animation on placeholder <code>&lt;div&gt;</code> elements. No JavaScript required.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
    <ApiPill href="https://developer.mozilla.org/en-US/docs/Web/CSS/@keyframes" title="CSS pulse animation for shimmer effect"><code>@keyframes</code></ApiPill>
  </SkillRow>
  </SkillField>

  <SkillField label="Classes">
<SkillRow>
    <code>.skeleton</code>
    <code>.skeleton-round</code>
  </SkillRow>
  </SkillField>

  <SkillField label="Accessibility">
<SkillText dense>• Skeleton elements are decorative — screen readers should skip them</SkillText>
  <SkillText dense>• Use <code>aria-hidden="true"</code> on skeleton containers if they sit alongside real content</SkillText>
  <p class="text-muted-foreground" style="margin:0 0 0;">• Announce the loading state separately with <code>role="status"</code> if needed</p>
