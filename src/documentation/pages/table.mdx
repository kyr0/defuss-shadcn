<PageHeader slug="table" title="Table" component="table">
  A responsive table component for displaying tabular data. Built on native <code>&lt;table&gt;</code> with styled headers, rows, and cells.
</PageHeader>

<SkillPanel component="table">
  <SkillField label="Native basis">
<SkillText><code>&lt;table&gt;</code> element with semantic <code>&lt;thead&gt;</code>, <code>&lt;tbody&gt;</code>, <code>&lt;tfoot&gt;</code>, and <code>&lt;caption&gt;</code>.</SkillText>
  </SkillField>

  <SkillField label="Web Platform APIs">
<SkillRow>
