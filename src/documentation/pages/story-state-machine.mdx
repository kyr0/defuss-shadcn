  // df$ exists only after all.js boots (it loads after example scripts in the
  // sandbox) - build the stage in the load handler, never at parse time.
  window.addEventListener('load', () => {
    const deck = document.getElementById('deck-story');
    const TEXT = '#F6F4EF', MUTED = '#7D7A74', ORANGE = '#FF5B35';
    // [service, last year M, this year M, monthly churn %]
    const s = [['Nova+', 46, 54, 3.1], ['Beacon', 43, 41, 4.4], ['Orbit', 22, 29, 5.8], ['Lumen', 19, 18, 6.2], ['Reel', 7, 11, 7.9]];
    const G = { left: 190, right: 190, top: 330, bottom: 150 };
    const yCats = { type: 'category', inverse: true, data: s.map((r) => r[0]), axisLine: { show: false }, axisLabel: { color: TEXT, fontWeight: 700 } };
    const xM = { type: 'value', max: 60, axisLabel: { color: MUTED, formatter: '{value}M' }, splitLine: { lineStyle: { color: 'rgba(246,244,239,.08)' } } };
    // series 'main' carries the identity through all four claims (id + item names)
    const main = (value, color, extra = {}) => ({ id: 'main', name: 'This year', type: 'bar', barWidth: '42%',
      data: s.map((r) => ({ name: r[0], value: r[value], itemStyle: { color: color(r), borderRadius: [0, 6, 6, 0] } })), ...extra });

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        baseline: { grid: G, xAxis: xM, yAxis: yCats,
          series: [main(2, (r) => (r[0] === 'Nova+' ? ORANGE : TEXT), { label: { show: true, position: 'right', formatter: '{c}M', color: TEXT, fontWeight: 700 } })] },
        compare: { grid: G, xAxis: xM, yAxis: yCats, legend: { top: 300, right: 190, textStyle: { color: TEXT } },
          series: [
            main(2, (r) => (r[2] > r[1] ? ORANGE : TEXT), { label: { show: true, position: 'right', formatter: '{c}M', color: TEXT, fontWeight: 700 } }),
            { id: 'then', name: 'Last year', type: 'bar', barWidth: '42%', data: s.map((r) => ({ name: r[0], value: r[1] })), itemStyle: { color: 'rgba(246,244,239,.22)', borderRadius: [0, 6, 6, 0] } },
          ] },
        share: {
          series: [{ id: 'main', type: 'pie', radius: ['30%', '52%'], center: ['50%', '64%'],
            data: s.map((r, i) => ({ name: r[0], value: r[2], itemStyle: { color: [ORANGE, TEXT, '#B8B4AC', '#7D7A74', '#4A4844'][i] } })),
            itemStyle: { borderColor: '#050505', borderWidth: 6 },
            label: { formatter: (p) => p.name + '  ' + Math.round(p.percent) + '%', color: TEXT, fontWeight: 700 }, labelLine: { lineStyle: { color: MUTED }, length: 20, length2: 30 } }],
        },
        churn: { grid: G, yAxis: yCats,
          xAxis: { type: 'value', max: 9, axisLabel: { color: MUTED, formatter: '{value}%' }, splitLine: { lineStyle: { color: 'rgba(246,244,239,.08)' } } },
          series: [main(3, (r) => (r[3] > 5.5 ? ORANGE : TEXT), { label: { show: true, position: 'right', formatter: '{c}%', color: TEXT, fontWeight: 700 } })] },
      },
    });
  });
</script>
