    const byStreams = genres.slice().sort((a, b) => b[3] - a[3]);
    const byGrowth = genres.slice().sort((a, b) => b[4] - a[4]);

    df$.shadcn.chart.deck(deck, {
      base: { tooltip: { trigger: 'item' } },
      states: {
        bump: bump(null),
        climb: bump(['K-pop', 'Latin']),
        streams: bars(byStreams, 3, '{c} B', { type: 'value', max: 11, axisLabel: { color: MUTED, formatter: '{value} B' }, splitLine: { lineStyle: { color: DIM } } }),
        growth: bars(byGrowth, 4, (p) => (p.value > 0 ? '+' : '') + p.value + '%', { type: 'value', min: -100, max: 450, axisLabel: { color: MUTED, formatter: '{value}%' }, splitLine: { lineStyle: { color: DIM } } }),
      },
    });
  });
