      ['Coal', '#6F8393', [42, 37, 30, 28, 21, 16]],
      ['Gas', '#A9C1D3', [10, 13, 15, 14, 15, 14]],
      ['Nuclear', '#E9F7FF', [14, 12, 12, 11, 2, 0]],
      ['Wind', '#7BE0D3', [13, 17, 21, 22, 30, 33]],
      ['Solar', '#FFD166', [6, 7, 8, 10, 13, 17]],
      ['Hydro & bio', '#4FA3E0', [15, 14, 14, 15, 19, 20]],
    ];
    const G = { left: 150, right: 170, top: 370, bottom: 150 };
    const legend = { top: 300, left: 150, textStyle: { color: TEXT }, itemGap: 28 };
    const xCat = { type: 'category', data: years, axisLabel: { color: TEXT, fontWeight: 700 }, axisLine: { lineStyle: { color: MUTED } } };
    const yPct = { type: 'value', max: 100, interval: 25, axisLabel: { color: MUTED, formatter: '{value}%' }, splitLine: { lineStyle: { color: LINE } } };

        split: { legend, grid: G, xAxis: xCat, yAxis: yPct,
          series: [
